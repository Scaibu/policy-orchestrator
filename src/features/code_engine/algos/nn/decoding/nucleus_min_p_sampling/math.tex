\begin{abstract}
Fixed $k$ in Top-$k$ sampling is brittle: when the distribution is flat, $k$ is too small; when the distribution is peaked, $k$ is too large. Nucleus (Top-$p$) sampling dynamically adapts candidate size by selecting the smallest subset whose cumulative probability exceeds $p \in (0, 1]$. Min-$p$ sampling filters out tokens whose probability is below a threshold fraction of the top candidate. This document formalizes both filtering operators and proves probability mass guarantees.
\end{abstract}

\section{Mathematical Formulation}

Let tokens be sorted descending by probability: $p_{(1)} \ge p_{(2)} \ge \dots \ge p_{(V)}$.
\begin{enumerate}
    \item \textbf{Top-$p$ Nucleus:} Find minimum cutoff index $k^*$ such that:
    \begin{equation}
    k^* = \min \left\{ k \;\Big|\; \sum_{i=1}^k p_{(i)} \ge p \right\}
    \end{equation}
    Candidates outside $\{ (1), \dots, (k^*) \}$ are set to zero probability.
    \item \textbf{Min-$p$ Filtering:} Retain tokens satisfying:
    \begin{equation}
    P(v) \ge p_{\min} \cdot p_{(1)}
    \end{equation}
\end{enumerate}
