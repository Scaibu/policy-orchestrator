\begin{abstract}
Autoregressive decoding is vulnerable to degenerative repetition loops. Repetition and frequency penalties penalize token logits based on prior emission frequency $c_v$. This document formalizes multiplicative repetition penalty $\theta$ and additive presence and frequency penalties, establishing monotonic logit attenuation guarantees.
\end{abstract}

\section{Mathematical Formulation}

For each token $v \in \mathcal{V}$ with emission count $c_v$ in history context:
\begin{enumerate}
    \item \textbf{Multiplicative Repetition Penalty ($\theta \ge 1.0$):}
    \begin{equation}
    z_v' = \begin{cases} z_v / \theta, & \text{if } z_v > 0 \text{ and } c_v > 0 \\ z_v \cdot \theta, & \text{if } z_v \le 0 \text{ and } c_v > 0 \\ z_v, & \text{if } c_v = 0 \end{cases}
    \end{equation}
    \item \textbf{Additive Penalties ($\alpha_{\text{freq}}, \alpha_{\text{pres}}$):}
    \begin{equation}
    z_v'' = z_v' - \alpha_{\text{freq}} \cdot c_v - \alpha_{\text{pres}} \cdot \mathbb{I}(c_v > 0)
    \end{equation}
\end{enumerate}
