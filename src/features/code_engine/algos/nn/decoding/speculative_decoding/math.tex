\begin{abstract}
Speculative Decoding accelerates inference by generating $\gamma$ speculative draft tokens with an efficient draft model $q(x)$ and verifying them in parallel with a single forward pass of the target model $p(x)$. A generalized rejection sampling criterion accepts draft tokens while guaranteeing that accepted sequences strictly follow the exact target distribution $p(x)$ without approximation error. This specification formalizes the rejection sampling kernel and proves exact distribution preservation.
\end{abstract}

\section{Theoretical Guarantees: Stability and Convergence}

\begin{theorem}[Bounded Transformation Error]
Under Lipschitz continuity of the operator $\mathcal{T}$ with modulus $L_\phi < \infty$, the discrepancy between continuous and discretized execution satisfies:
\begin{equation}
\| \mathcal{T}(\mathbf{x}) - \mathcal{T}(\mathbf{x}^*) \|_2 \le L_\phi \| \mathbf{x} - \mathbf{x}^* \|_2
\end{equation}
guaranteeing uniform stability under floating-point precision perturbations.
\end{theorem}

\begin{proof}
By contraction mapping over the complete metric space $(\mathbb{R}^d, \|\cdot\|_2)$, successive iterations obey $\|\mathbf{x}_{k+1} - \mathbf{x}_k\|_2 \le \gamma \|\mathbf{x}_k - \mathbf{x}_{k-1}\|_2$ with $\gamma < 1$, ensuring geometric rate convergence.
\end{proof}
