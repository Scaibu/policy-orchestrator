\begin{abstract}
Mixture of Experts (MoE) decouples parameter capacity from computational FLOPs by conditionally routing individual tokens through a small sparse subset $k \ll E$ of feedforward expert networks. This specification formalizes the Top-$k$ discrete gating operator, proves probability simplex normalization over active experts, establishes computational complexity bounds, and details numerical step-by-step verification.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $T \in \mathbb{N}_{\ge 1}$ denote the batch token count, and let $E \in \mathbb{N}_{\ge 2}$ denote the total number of specialized expert networks.

\begin{itemize}
    \item $\mathbf{Z} \in \mathbb{R}^{T \times E}$: Unnormalized router logits, where $Z_{t, e}$ indicates the routing affinity of token $t$ to expert $e$.
    \item $k \in \{1, \dots, E\}$: Sparse gating cardinality (number of active experts per token, typically $k=1$ or $k=2$).
    \item $\mathcal{T}_t = \operatorname{argtopk}(\mathbf{z}_t, k) \subset \{0, \dots, E-1\}$: Set of top-$k$ selected expert indices for token $t$.
    \item $g_{t, e} \in [0, 1]$: Normalized routing weight assigned to expert $e \in \mathcal{T}_t$.
    \item $\mathbf{D} \in \{0, 1\}^{T \times E}$: Binary dispatch routing mask.
\end{itemize}

\section{Problem Definition and Algorithmic Execution}

\subsection{Mathematical Routing Formulation}
For token $t$, the router identifies the $k$ largest logits $\mathcal{T}_t$. Gating weights are normalized exclusively over the active subset via stabilized softmax:
\begin{equation}
g_{t, e} = \begin{cases}
\frac{\exp(Z_{t, e} - \max_{j \in \mathcal{T}_t} Z_{t, j})}{\sum_{j \in \mathcal{T}_t} \exp(Z_{t, j} - \max_{l \in \mathcal{T}_t} Z_{t, l})}, & \text{if } e \in \mathcal{T}_t \\
0, & \text{otherwise}
\end{cases}
\end{equation}
The composite output representation is a linear combination over the activated experts:
\begin{equation}
\mathbf{y}_t = \sum_{e \in \mathcal{T}_t} g_{t, e} \operatorname{Expert}_e(\mathbf{x}_t)
\end{equation}

\begin{algorithm}[H]
\caption{MoE Top-$K$ Gating and Dispatch Mask Evaluation}
\label{alg:moe_topk}
\begin{algorithmic}[1]
\Require Router logits $\mathbf{Z} \in \mathbb{R}^{T \times E}$, active experts count $k$
\Ensure Selected experts $\boldsymbol{\mathcal{T}}$, gating weights $\mathbf{G}$, dispatch mask $\mathbf{D}$
\State $T \gets \text{rows}(\mathbf{Z}), \quad E \gets \text{cols}(\mathbf{Z})$
\State \textbf{Assert} $1 \le k \le E$
\State Initialize $\boldsymbol{\mathcal{T}}, \mathbf{G}, \mathbf{D}$
\For{$t \gets 0 \textbf{ to } T - 1$}
    \State $\mathbf{z} \gets \mathbf{Z}[t]$
    \State Sort expert indices by descending logit value: $\pi_0, \dots, \pi_{E-1}$
    \State $\text{top\_indices} \gets [\pi_0, \dots, \pi_{k-1}]$
    \State $m \gets \max_{j \in \text{top\_indices}} z[j]$
    \State $S \gets \sum_{j \in \text{top\_indices}} \exp(z[j] - m)$
    \State $\mathbf{g}_t \gets [\exp(z[j] - m) / S \text{ for } j \in \text{top\_indices}]$
    \State $\mathbf{d}_t \gets \text{zeros of size } E$
    \For{$j \in \text{top\_indices}$}
        \State $d_t[j] \gets 1$
    \EndFor
    \State Append $\text{top\_indices}$ to $\boldsymbol{\mathcal{T}}$, $\mathbf{g}_t$ to $\mathbf{G}$, $\mathbf{d}_t$ to $\mathbf{D}$
\EndFor
\State \Return $\{\text{selected\_experts}: \boldsymbol{\mathcal{T}}, \text{gating\_weights}: \mathbf{G}, \text{dispatch\_mask}: \mathbf{D}\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees: Probability Simplex Preservation}

\begin{theorem}[Convex Mixture Invariance]
For any token $t$, the active gating weights strictly lie on the $(k-1)$-dimensional probability simplex $\Delta^{k-1}$:
\begin{equation}
\sum_{e \in \mathcal{T}_t} g_{t, e} = 1.0 \quad \text{and} \quad g_{t, e} > 0 \quad \forall e \in \mathcal{T}_t
\end{equation}
bounding output norms by the convex hull of active expert representations: $\|\mathbf{y}_t\| \le \max_{e \in \mathcal{T}_t} \|\operatorname{Expert}_e(\mathbf{x}_t)\|$.
\end{theorem}

\section{Complexity Analysis}
\begin{itemize}
    \item \textbf{Sorting Time Complexity:} Finding top-$k$ out of $E$ experts requires $\Theta(E + k \log E)$ operations per token. Across $T$ tokens, time is $\Theta(T E)$.
    \item \textbf{FLOP Reduction Factor:} Computing $k$ experts instead of $E$ reduces feedforward compute by a factor of $E / k$.
    \item \textbf{Space Complexity:} $\Theta(T E)$ for dispatch masks.
\end{itemize}

\section{Worked Numerical Example}
Let token logits be $\mathbf{z} = [2.0, 0.5, 3.0, 0.1]$ with $E = 4, k = 2$:
\begin{enumerate}
    \item Rank descending: index $2$ ($3.0$), index $0$ ($2.0$), index $1$ ($0.5$), index $3$ ($0.1$).
    \item Top-$2$ indices: $\mathcal{T} = [2, 0]$ with logits $[3.0, 2.0]$.
    \item Max logit: $m = 3.0$.
    \item Exp offsets: $\exp(3.0 - 3.0) = 1.0$, $\exp(2.0 - 3.0) = \exp(-1.0) \approx 0.367879$.
    \item Partition sum: $S = 1.0 + 0.367879 = 1.367879$.
    \item Weights: $g_2 = 1.0 / 1.367879 \approx 0.731059$, $g_0 = 0.367879 / 1.367879 \approx 0.268941$.
    \item Dispatch mask: $\mathbf{d} = [1, 0, 1, 0]$.
\end{enumerate}

\section{Numerical Considerations and Edge Cases}
\begin{itemize}
    \item \textbf{Uniform Fallback on Zero Exponents:} If extreme numerical underflow yields $S = 0$, the algorithm safely falls back to uniform weights $1/k$ to avoid $\text{NaN}$ outputs.
\end{itemize}
