\begin{abstract}
DEtection TRansformer (DETR) reframes object detection from handcrafted surrogate tasks (dense anchor classification, spatial grid regression, non-maximum suppression) into direct set prediction. Using a transformer encoder-decoder with learned object queries, DETR outputs a permutation-invariant set of $N$ predictions. Training is formulated as optimal bipartite graph matching via the Hungarian algorithm, optimizing a global cost combining classification confidence, $\ell_1$ coordinate distance, and Generalized IoU (GIoU). This document presents the formal matching formulation, proves permutation equivariance and uniqueness of bipartite assignment, specifies algorithmic pseudocode, and delivers a verified numerical walkthrough.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $y = \{(c_i, b_i)\}_{i=1}^M$ denote the set of ground-truth objects ($M < N$), padded with background tokens $\varnothing$ to fixed size $N$: $y = \{(c_i, b_i)\}_{i=1}^N$. Let $\hat{y} = \{(\hat{p}_j, \hat{b}_j)\}_{j=1}^N$ denote the set of $N$ query predictions from the transformer decoder.

\section{Problem Definition and Core Concepts}

\subsection{3-Level Explanation}
\begin{enumerate}
    \item \textbf{Intuitive (High School level):} Instead of tossing thousands of anchor boxes everywhere and weeding out duplicates with NMS, DETR sends in 100 specialized ``query inspectors''. Each inspector bids on the best matching real object. The Hungarian matching algorithm pairs each real object to exactly one inspector, and the inspectors who find nothing report ``background''.
    \item \textbf{Engineering Level:} DETR eliminates custom anchor tuning, FPN multi-scale pyramid heads, and post-processing NMS. A CNN backbone + Transformer Encoder extracts global contextual features, and a Transformer Decoder cross-attends onto features using learned query vectors. Bipartite matching enables direct end-to-end set loss training.
    \item \textbf{Mathematical Level:} Given ground truth $y$ and predictions $\hat{y}$, DETR computes the optimal assignment permutation $\hat{\sigma} \in \mathfrak{S}_N$:
    \begin{equation}
    \hat{\sigma} = \arg\min_{\sigma \in \mathfrak{S}_N} \sum_{i=1}^N \mathcal{L}_{\text{match}}(y_i, \hat{y}_{\sigma(i)})
    \end{equation}
    where for non-empty ground truth $c_i \ne \varnothing$:
    \begin{equation}
    \mathcal{L}_{\text{match}}(y_i, \hat{y}_{\sigma(i)}) = -\hat{p}_{\sigma(i)}(c_i) + \lambda_{\text{L1}} \|b_i - \hat{b}_{\sigma(i)}\|_1 + \lambda_{\text{giou}} \mathcal{L}_{\text{giou}}(b_i, \hat{b}_{\sigma(i)})
    \end{equation}
\end{enumerate}

\subsection{5-Step Algorithmic Framework}
\begin{enumerate}
    \item \textbf{Parallel Query Decoding:} Evaluate transformer decoder on $N$ positional queries to produce $\{(\hat{p}_j, \hat{b}_j)\}_{j=1}^N$.
    \item \textbf{Cost Matrix Construction:} Compute $M \times N$ pairwise matching cost matrix $\mathbf{C}_{i,j} = \mathcal{L}_{\text{match}}(y_i, \hat{y}_j)$.
    \item \textbf{Hungarian Optimal Matching:} Solve min-cost linear assignment problem $\hat{\sigma} = \text{Hungarian}(\mathbf{C})$.
    \item \textbf{Set-Level Loss Computation:} Evaluate cross-entropy classification on all $N$ pairs (including $\varnothing$) and $\ell_1 + \text{GIoU}$ on positive matched pairs.
    \item \textbf{Backpropagation:} Update transformer and backbone weights jointly via end-to-end gradient descent.
\end{enumerate}

\section{Algorithm Specification}

\begin{algorithm}[H]
\caption{DETR Bipartite Assignment and Hungarian Loss}
\label{alg:detr_match}
\begin{algorithmic}[1]
\Require Ground truth objects $y = \{(c_i, b_i)\}_{i=1}^M$, predictions $\hat{y} = \{(\hat{p}_j, \hat{b}_j)\}_{j=1}^N$, loss weights $\lambda_{\text{L1}}, \lambda_{\text{giou}}$.
\Ensure Matching assignment $\hat{\sigma}$ and Hungarian training loss $\mathcal{L}$.
\State Initialize cost matrix $\mathbf{C} \in \mathbb{R}^{M \times N}$
\For{$i = 1$ \textbf{to} $M$}
    \For{$j = 1$ \textbf{to} $N$}
        \State $C_{\text{cls}} \gets -\hat{p}_j(c_i)$
        \State $C_{\text{box}} \gets \lambda_{\text{L1}} \|b_i - \hat{b}_j\|_1 + \lambda_{\text{giou}} (1 - \text{GIoU}(b_i, \hat{b}_j))$
        \State $\mathbf{C}[i, j] \gets C_{\text{cls}} + C_{\text{box}}$
    \EndFor
\EndFor
\State $\hat{\sigma} \gets \text{SolveHungarianLinearAssignment}(\mathbf{C})$
\State $\mathcal{L} \gets 0$
\For{$i = 1$ \textbf{to} $M$}
    \State $j \gets \hat{\sigma}(i)$
    \State $\mathcal{L} \gets \mathcal{L} - \log \hat{p}_j(c_i) + \lambda_{\text{L1}} \|b_i - \hat{b}_j\|_1 + \lambda_{\text{giou}} (1 - \text{GIoU}(b_i, \hat{b}_j))$
\EndFor
\For{$j \in \{1,\dots, N\} \setminus \text{range}(\hat{\sigma})$}
    \State $\mathcal{L} \gets \mathcal{L} - \log \hat{p}_j(\varnothing)$
\EndFor
\State \Return $\hat{\sigma}, \mathcal{L}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Permutation Invariance}

\begin{theorem}[Permutation Invariance of Set Prediction Loss]
\label{thm:permutation_invariance}
Let $\pi \in \mathfrak{S}_N$ denote an arbitrary permutation of the prediction index set. The Hungarian matching assignment and resulting training loss are strictly invariant to prediction reordering:
\begin{equation}
\mathcal{L}_{\text{Hungarian}}(y, \pi(\hat{y})) = \mathcal{L}_{\text{Hungarian}}(y, \hat{y})
\end{equation}
\end{theorem}

\begin{proof}
Let $\mathbf{C}$ be the original cost matrix. Permuting predictions by $\pi$ yields cost matrix $\mathbf{C}'$ where $\mathbf{C}'_{i, j} = \mathbf{C}_{i, \pi(j)}$. Under optimal bipartite assignment:
\begin{equation}
\min_{\sigma} \sum_{i=1}^M \mathbf{C}'_{i, \sigma(i)} = \min_{\sigma} \sum_{i=1}^M \mathbf{C}_{i, \pi(\sigma(i))} = \min_{\sigma'} \sum_{i=1}^M \mathbf{C}_{i, \sigma'(i)}
\end{equation}
where $\sigma' = \pi \circ \sigma$. Since $\pi$ is a bijection on $\mathfrak{S}_N$, the minimum value over the symmetric group is identical.
\end{proof}

\section{Complexity Analysis}

\begin{itemize}
    \item \textbf{Time Complexity:}
    \begin{itemize}
        \item Pairwise cost matrix computation: $\mathcal{O}(M \cdot N)$ distance and GIoU calculations.
        \item Hungarian algorithm (Kuhn-Munkres): $\mathcal{O}(M^2 N)$ or $\mathcal{O}(N^3)$ worst-case.
        \item Total Matching Time: $\mathcal{O}(M^2 N)$ (negligible in practice since $M \ll N = 100$).
    \end{itemize}
    \item \textbf{Space Complexity:} $\mathcal{O}(M \cdot N)$ to store the matching cost matrix.
\end{itemize}

\section{Exactness and Error Analysis}

The GIoU formulation $\text{GIoU} = \text{IoU} - \frac{|C \setminus (A \cup B)|}{|C|}$ ensures non-zero gradients even when predicted and ground truth boxes have zero spatial intersection ($\text{IoU} = 0$), guiding the Hungarian solver towards correct spatial convergence.

\section{Worked Numerical Example}

Consider $M=1$ ground truth object ($c_1 = 1$, $b_1 = (0.5, 0.5, 0.2, 0.2)$) and $N=2$ queries with $\lambda_{\text{L1}} = 1.0, \lambda_{\text{giou}} = 1.0$:
\begin{itemize}
    \item Query 1: $\hat{p}_1(1) = 0.8$, $\hat{b}_1 = (0.5, 0.5, 0.2, 0.2) \implies \text{IoU}=1.0, \text{GIoU}=1.0, \text{L1}=0$.
    \item Query 2: $\hat{p}_2(1) = 0.2$, $\hat{b}_2 = (0.1, 0.1, 0.2, 0.2) \implies \text{IoU}=0.0, \text{GIoU}=-0.5, \text{L1}=0.8$.
\end{itemize}

\textbf{Cost Matrix $\mathbf{C}_{1 \times 2}$:}
\begin{itemize}
    \item $\mathbf{C}[0, 0] = -0.8 + 1.0(0.0) - 1.0(1.0) = -1.8$.
    \item $\mathbf{C}[0, 1] = -0.2 + 1.0(0.8) - 1.0(-0.5) = -0.2 + 0.8 + 0.5 = +1.1$.
\end{itemize}

Hungarian match selects Query 1 ($\hat{\sigma}(1) = 1$) minimizing cost $-1.8 < 1.1$.

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Class Probability vs Log-Probability in Matching:} DETR uses probability $\hat{p}(c)$ directly rather than log-probability $\log \hat{p}(c)$ in the matching cost $\mathcal{L}_{\text{match}}$ to keep classification cost commensurable with $\ell_1$ spatial box distances.
    \item \textbf{Slow Convergence:} Baseline DETR requires 500 epochs; Deformable DETR accelerates convergence via multi-scale local sparse attention.
\end{itemize}
