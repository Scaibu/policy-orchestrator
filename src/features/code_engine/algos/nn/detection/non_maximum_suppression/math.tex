\begin{abstract}
Dense object detection systems generate vast numbers of highly redundant, overlapping candidate bounding boxes for each physical instance. Non-Maximum Suppression (NMS) serves as the ubiquitous post-processing arbitration mechanism to eliminate spatial duplicates. While traditional Hard-NMS applies a binary threshold causing missed detections in crowded occlusion scenes, Soft-NMS introduces a continuous score attenuation function decaying confidence proportionally to spatial overlap. This document provides rigorous mathematical formulations of Hard, Linear, and Gaussian Soft-NMS, proves worst-case convergence properties, specifies complete algorithmic pseudocode, and provides verified numerical calculations.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $\mathcal{B} = \{b_1, \dots, b_N\}$ denote the set of candidate bounding boxes where $b_i = (x_1, y_1, x_2, y_2) \in \mathbb{R}^4$, and let $\mathcal{S} = \{s_1, \dots, s_N\}$ denote their initial detection confidence scores where $s_i \in [0, 1]$.

\section{Problem Definition and Core Concepts}

\subsection{3-Level Explanation}
\begin{enumerate}
    \item \textbf{Intuitive (High School level):} When a detector spots a car, it draws 50 boxes around the same car. NMS picks the single highest-rated box and erases the other 49. Soft-NMS is gentler: instead of instantly erasing overlapping boxes, it lowers their score so real objects standing close behind aren't accidentally deleted.
    \item \textbf{Engineering Level:} Hard-NMS sorts candidates descending by score and iteratively discards lower-scoring candidates whose IoU with the current maximum exceeds $N_t$. Soft-NMS replaces hard thresholding with continuous Gaussian score dampening $s_i \gets s_i e^{-\text{IoU}^2/\sigma}$, allowing occluded objects to survive.
    \item \textbf{Mathematical Level:} Given candidate set $\mathcal{B}$ with scores $\mathcal{S}$, Soft-NMS solves an iterative score transformation operator $\mathcal{T}: \mathbb{R}^N \to \mathbb{R}^N$ governed by the recurrence:
    \begin{equation}
    s_i^{(k+1)} = s_i^{(k)} \cdot f(\text{IoU}(M^{(k)}, b_i))
    \end{equation}
    where for Gaussian Soft-NMS $f(u) = \exp(-u^2 / \sigma)$, and for Linear Soft-NMS $f(u) = 1 - u \cdot \mathbb{I}(u \ge N_t)$.
\end{enumerate}

\subsection{5-Step Algorithmic Framework}
\begin{enumerate}
    \item \textbf{Score Sorting / Pivot Selection:} Select box $M$ with highest current confidence score from candidate pool.
    \item \textbf{Transfer to Keep Set:} Move $M$ into retained detection list $\mathcal{D}$.
    \item \textbf{Pairwise Overlap Computation:} Calculate $\text{IoU}(M, b_i)$ for all remaining candidates $b_i$.
    \item \textbf{Score Attenuation / Pruning:} Apply hard removal if $\text{IoU} \ge N_t$ or decay score by $f(\text{IoU})$.
    \item \textbf{Termination and Filtering:} Repeat until candidate pool is exhausted, then discard boxes with score $< \tau$.
\end{enumerate}

\section{Algorithm Specification}

\begin{algorithm}[H]
\caption{Gaussian Soft-NMS Algorithm}
\label{alg:soft_nms}
\begin{algorithmic}[1]
\Require Boxes $\mathcal{B} = \{b_1, \dots, b_N\}$, Scores $\mathcal{S} = \{s_1, \dots, s_N\}$, Gaussian variance $\sigma$, threshold $\tau$.
\Ensure Set of detected boxes and updated scores $\mathcal{D}$.
\State $\mathcal{D} \gets \emptyset$
\While{$\mathcal{B} \ne \emptyset$}
    \State $m \gets \arg\max_{i} \mathcal{S}[i]$
    \State $M \gets \mathcal{B}[m], \quad s_M \gets \mathcal{S}[m]$
    \State $\mathcal{D} \gets \mathcal{D} \cup \{(M, s_M)\}$
    \State $\mathcal{B} \gets \mathcal{B} \setminus \{M\}, \quad \mathcal{S} \gets \mathcal{S} \setminus \{s_M\}$
    \For{$i = 1$ \textbf{to} $|\mathcal{B}|$}
        \State $\text{iou} \gets \text{IoU}(M, \mathcal{B}[i])$
        \State $\mathcal{S}[i] \gets \mathcal{S}[i] \cdot \exp\left( - \frac{\text{iou}^2}{\sigma} \right)$
    \EndFor
\EndWhile
\State $\mathcal{D} \gets \{(b, s) \in \mathcal{D} \mid s \ge \tau\}$
\State \Return $\mathcal{D}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Convergence}

\begin{theorem}[Monotonic Score Decay and Termination]
\label{thm:nms_termination}
For any non-negative initial scores $s_i \ge 0$ and $\sigma > 0$, the sequence of score updates in Gaussian Soft-NMS satisfies $s_i^{(k+1)} \le s_i^{(k)}$ monotonically, and the algorithm terminates in exactly $N$ outer iterations with total time complexity bounded by $\mathcal{O}(N^2)$.
\end{theorem}

\begin{proof}
Because $\text{IoU} \in [0, 1]$ and $\sigma > 0$, the exponent $-\text{IoU}^2 / \sigma \le 0$, yielding $\exp(-\text{IoU}^2 / \sigma) \in (0, 1]$. Thus $s_i^{(k+1)} = s_i^{(k)} \cdot \exp(-\text{IoU}^2 / \sigma) \le s_i^{(k)}$. At each outer step, exactly one element is moved to $\mathcal{D}$, decrementing the pool size from $N$ to $0$ in $N$ steps. The $k$-th iteration evaluates $N - k$ pairwise IoUs, summing to $\sum_{k=1}^N (N - k) = \frac{N(N-1)}{2} = \mathcal{O}(N^2)$.
\end{proof}

\section{Complexity Analysis}

\begin{itemize}
    \item \textbf{Time Complexity:}
    \begin{itemize}
        \item Initial sorting / argmax: $\mathcal{O}(N \log N)$ or $\mathcal{O}(N^2)$.
        \item Pairwise IoU comparisons: $\frac{N(N-1)}{2} \implies \mathcal{O}(N^2)$.
        \item Total Time: $\mathcal{O}(N^2)$ (practical fast batched implementations run in $\mathcal{O}(N)$ parallel tensor cores).
    \end{itemize}
    \item \textbf{Space Complexity:} $\mathcal{O}(N)$ auxiliary memory for index tracking and score arrays.
\end{itemize}

\section{Exactness and Error Analysis}

In crowded environments where two true objects have overlap $\text{IoU} = 0.6$, Hard-NMS with $N_t = 0.5$ produces a false negative rate of $100\%$ on the secondary object. Gaussian Soft-NMS with $\sigma = 0.5$ decays the score by $\exp(-0.36 / 0.5) = e^{-0.72} \approx 0.4868$, preserving the detection above standard thresholds.

\section{Worked Numerical Example}

Consider two boxes:
\begin{itemize}
    \item Box 1: $b_1 = (0, 0, 10, 10)$, score $s_1 = 0.90$.
    \item Box 2: $b_2 = (2, 0, 12, 10)$, score $s_2 = 0.80$.
\end{itemize}

\textbf{Step 1: IoU Computation:}
\begin{itemize}
    \item Intersect: $x_1 = 2, x_2 = 10, y_1 = 0, y_2 = 10 \implies \text{Area}_{\text{inter}} = 8 \times 10 = 80$.
    \item $\text{Area}_1 = 10 \times 10 = 100, \quad \text{Area}_2 = 10 \times 10 = 100$.
    \item $\text{Area}_{\text{union}} = 100 + 100 - 80 = 120$.
    \item $\text{IoU} = \frac{80}{120} = \frac{2}{3} \approx 0.6667$.
\end{itemize}

\textbf{Step 2: Hard-NMS ($N_t = 0.5$):}
$\text{IoU} = 0.6667 \ge 0.5 \implies$ Box 2 is suppressed ($s_2 \gets 0$). Output: $\{b_1\}$.

\textbf{Step 3: Gaussian Soft-NMS ($\sigma = 0.5$):}
\begin{itemize}
    \item $M = b_1, s_1 = 0.90 \to \text{Kept}$.
    \item Decay weight $= \exp\left( -\frac{(0.6667)^2}{0.5} \right) = \exp\left(-\frac{0.4444}{0.5}\right) = \exp(-0.8889) \approx 0.4111$.
    \item Updated score $s_2' = 0.80 \times 0.4111 = 0.3289$.
    \item Output: $\{(b_1, 0.90), (b_2, 0.3289)\}$.
\end{itemize}

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Class-Aware vs Multi-Class NMS:} Performing NMS independently per class prevents cross-category suppression (e.g., a person sitting inside a car being suppressed by the car's bounding box).
    \item \textbf{Degenerate Box Area:} Boxes with $w \le 0$ or $h \le 0$ must be filtered before IoU to prevent division-by-zero.
\end{itemize}
