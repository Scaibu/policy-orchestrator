\begin{abstract}
Traditional object detection frameworks rely heavily on predefined anchor boxes, introducing cumbersome hyperparameter tuning (scales, aspect ratios, IoU assignment thresholds) and significant computation on negative background anchors. Fully Convolutional One-Stage (FCOS) detection eliminates anchor boxes entirely, solving object detection in a per-pixel prediction fashion akin to semantic segmentation. FCOS introduces a 4D distance vector regression target to box boundaries and a novel ``centerness'' branch to suppress low-quality bounding boxes predicted far from object centers. This document establishes the mathematical formulation of per-pixel regression, proves centerness gradient properties, details algorithmic pseudocode, and delivers a complete numerical example.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let feature map level $P_i$ have spatial stride $s$. For each coordinate $(x, y)$ on $P_i$, its corresponding mapped center in input image space is $(s x + s/2, s y + s/2)$.

\section{Problem Definition and Core Concepts}

\subsection{3-Level Explanation}
\begin{enumerate}
    \item \textbf{Intuitive (High School level):} Instead of trying on 9 different sizes of cardboard cutout frames at every spot, each pixel that lands inside an object simply measures 4 rulers: how far to the left, top, right, and bottom edges of the object. Points near the middle are given high ``centerness'' scores to ignore messy edge pixels.
    \item \textbf{Engineering Level:} FCOS converts detection heads into standard per-pixel FCN branches: a classification head ($C$ logits), a box regression head ($4$ distance scalars), and a centerness head ($1$ logit). Scale conflicts between overlapping objects are resolved by distributing target sizes across FPN pyramid levels.
    \item \textbf{Mathematical Level:} A location $(x, y)$ inside bounding box $B = (x_0, y_0, x_1, y_1)$ has target regression vector $\mathbf{t}^* = (l^*, t^*, r^*, b^*)$ defined by:
    \begin{equation}
    l^* = x - x_0, \quad t^* = y - y_0, \quad r^* = x_1 - x, \quad b^* = y_1 - y
    \end{equation}
    and centerness target:
    \begin{equation}
    \text{centerness}^* = \sqrt{\frac{\min(l^*, r^*)}{\max(l^*, r^*)} \times \frac{\min(t^*, b^*)}{\max(t^*, b^*)}}
    \end{equation}
\end{enumerate}

\subsection{5-Step Algorithmic Framework}
\begin{enumerate}
    \item \textbf{Per-Pixel Multi-Head Forward Evaluation:} Predict $C$-class logits, 4 regression distances $(l, t, r, b)$, and centerness logit for every feature pixel.
    \item \textbf{Exponential Distance Scaling:} Compute metric distances $(l, t, r, b) = s \cdot \exp(\mathbf{t}_{\text{reg}})$.
    \item \textbf{Coordinate Box Reconstruction:} Compute corners $(x_1, y_1, x_2, y_2) = (x - l, y - t, x + r, y + b)$.
    \item \textbf{Centerness Confidence Factoring:} Modulate classification probabilities: $\text{Score} = \sqrt{p_{\text{cls}} \cdot \sigma(\text{logit}_{\text{centerness}})}$.
    \item \textbf{Multi-Scale Range Filtering and NMS:} Filter predictions outside pyramid level range $[m_{i-1}, m_i]$ and apply soft/hard NMS.
\end{enumerate}

\section{Algorithm Specification}

\begin{algorithm}[H]
\caption{FCOS Dense Feature Decoding}
\label{alg:fcos}
\begin{algorithmic}[1]
\Require Feature classification logits $P \in \mathbb{R}^{C \times H \times W}$, regression distances $R \in \mathbb{R}^{4 \times H \times W}$, centerness logits $K \in \mathbb{R}^{H \times W}$, stride $s$, threshold $\tau$.
\Ensure Set of detected candidate boxes $\mathcal{D}$.
\State $\mathcal{D} \gets \emptyset$
\For{$i = 0$ \textbf{to} $H-1$}
    \For{$j = 0$ \textbf{to} $W-1$}
        \State $p_y \gets (i + 0.5) \cdot s, \quad p_x \gets (j + 0.5) \cdot s$
        \State $\text{cnt} \gets \sigma(K[i, j])$
        \State $l \gets \exp(R[0, i, j]) \cdot s, \quad t \gets \exp(R[1, i, j]) \cdot s$
        \State $r \gets \exp(R[2, i, j]) \cdot s, \quad b \gets \exp(R[3, i, j]) \cdot s$
        \State $x_1 \gets p_x - l, \quad y_1 \gets p_y - t, \quad x_2 \gets p_x + r, \quad y_2 \gets p_y + b$
        \For{$c = 1$ \textbf{to} $C$}
            \State $p_c \gets \sigma(P[c, i, j])$
            \State $\text{score} \gets \sqrt{p_c \cdot \text{cnt}}$
            \If{$\text{score} \ge \tau$}
                \State $\mathcal{D} \gets \mathcal{D} \cup \{(x_1, y_1, x_2, y_2, \text{score}, c)\}$
            \EndIf
        \EndFor
    \EndFor
\EndFor
\State \Return $\mathcal{D}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Centerness Properties}

\begin{theorem}[Centerness Spatial Gradient Extrema]
\label{thm:centerness}
Let $(x, y) \in [x_0, x_1] \times [y_0, y_1]$. The centerness target function $\mathcal{C}(x, y)$ is strictly maximized with value $1.0$ at the exact geometric centroid $(x^*, y^*) = \left(\frac{x_0+x_1}{2}, \frac{y_0+y_1}{2}\right)$, and decays monotonically to $0.0$ on the bounding box boundary $\partial B$.
\end{theorem}

\begin{proof}
Let $w = x_1 - x_0$. Distance $l = x - x_0$ and $r = x_1 - x = w - l$. The horizontal term $h(l) = \frac{\min(l, w-l)}{\max(l, w-l)}$. For $l \in (0, w/2]$, $h(l) = \frac{l}{w-l}$, whose derivative $\frac{d h}{d l} = \frac{w}{(w-l)^2} > 0$. At $l = w/2$, $h(w/2) = 1.0$. By symmetry, $h(l)$ is concave and maximized at $w/2$. As $l \to 0$ or $l \to w$, $h(l) \to 0$. The product of horizontal and vertical terms is thus uniquely maximized at the box center.
\end{proof}

\section{Complexity Analysis}

\begin{itemize}
    \item \textbf{Time Complexity:} $\mathcal{O}(H \cdot W \cdot (C + 4))$ FLOPs per pyramid level (no anchor iteration multiplier $K$).
    \item \textbf{Space Complexity:} $\mathcal{O}(H \cdot W \cdot (C + 5))$ activation tensor memory.
\end{itemize}

\section{Exactness and Error Analysis}

By predicting direct pixel-to-boundary metrics $(l, t, r, b)$ optimized via IoU / GIoU loss:
\begin{equation}
\mathcal{L}_{\text{IoU}} = -\ln \left( \frac{\min(l, l^*)\min(r, r^*) + \min(t, t^*)\min(b, b^*)}{(l+r)(t+b) + (l^*+r^*)(t^*+b^*) - \text{Inter}} \right)
\end{equation}
the optimization is invariant to arbitrary spatial scaling of images.

\section{Worked Numerical Example}

Consider point at image coordinate $(x, y) = (50.0, 50.0)$.
Ground truth box: $B = (20.0, 10.0, 80.0, 90.0)$.
\begin{itemize}
    \item $l^* = 50.0 - 20.0 = 30.0, \quad r^* = 80.0 - 50.0 = 30.0 \implies \frac{\min(30,30)}{\max(30,30)} = 1.0$.
    \item $t^* = 50.0 - 10.0 = 40.0, \quad b^* = 90.0 - 50.0 = 40.0 \implies \frac{\min(40,40)}{\max(40,40)} = 1.0$.
    \item $\text{centerness}^* = \sqrt{1.0 \times 1.0} = 1.0$ (point is the exact center).
\end{itemize}
If shifted to $(x, y) = (25.0, 50.0)$:
\begin{itemize}
    \item $l^* = 5.0, r^* = 55.0 \implies \frac{5}{55} = 0.0909$.
    \item $\text{centerness}^* = \sqrt{0.0909 \times 1.0} \approx 0.3015$.
\end{itemize}

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Center Sampling:} Restricting positive training samples to a central sub-box (e.g., radius $1.5 \times s$ around GT center) significantly boosts detection precision.
    \item \textbf{Centerness Branch Placement:} Sharing the centerness branch with the regression head rather than the classification head improves stability.
\end{itemize}
