\begin{abstract}
Single-stage object detectors unify bounding box localization and semantic classification into a single dense feedforward pass across multi-scale convolutional grid cells. By eliminating discrete region proposal networks, single-stage architectures achieve real-time inference speeds while resolving severe foreground-background class imbalance via focal loss formulations or objectness priors. This document presents the mathematical foundation of single-stage grid decoders, proves numerical boundedness under sigmoid coordinate parameterization, establishes algorithmic pseudocode, and provides a verified step-by-step numerical calculation.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let the input image be partitioned into an $S_y \times S_x$ spatial grid with stride $s$. Each grid cell $(c_y, c_x)$ predicts parameters for $B$ anchor priors $(p_w, p_h)$.

\section{Problem Definition and Core Concepts}

\subsection{3-Level Explanation}
\begin{enumerate}
    \item \textbf{Intuitive (High School level):} Instead of searching around in two separate guessing steps, a single-stage detector divides the image into a chess board. Every single square immediately shouts out: ``Here is what object is sitting on me, how confident I am, and the exact box around it!''
    \item \textbf{Engineering Level:} Single-stage detectors like YOLO predict a 3D tensor of shape $S \times S \times B(5 + C)$ in one pass. Bounded sigmoid offsets $\sigma(t_x) \in (0, 1)$ prevent grid cells from drifting uncontrollably into neighboring regions, ensuring stable spatial gradients.
    \item \textbf{Mathematical Level:} For grid index $(c_x, c_y)$ and anchor prior $(p_w, p_h)$, the continuous coordinate transformation is given by:
    \begin{align}
    b_x &= (\sigma(t_x) + c_x) \cdot s, \quad b_y = (\sigma(t_y) + c_y) \cdot s \\
    b_w &= p_w \cdot \exp(t_w) \cdot s, \quad b_h = p_h \cdot \exp(t_h) \cdot s \\
    \text{Score}(c) &= \sigma(t_{\text{obj}}) \cdot \sigma(t_{\text{cls}, c})
    \end{align}
\end{enumerate}

\subsection{5-Step Algorithmic Framework}
\begin{enumerate}
    \item \textbf{Multi-Scale Grid Forward Pass:} Evaluate backbone + PANet feature heads to extract dense output tensors across strides $\{8, 16, 32\}$.
    \item \textbf{Sigmoid Offset Bounding:} Apply logistic activation $\sigma(t_x), \sigma(t_y)$ to constrain predicted box center within the activating grid cell.
    \item \textbf{Exponential Dimension Scaling:} Scale prior anchor dimensions by exponential regression factors $p_w e^{t_w}, p_h e^{t_h}$.
    \item \textbf{Joint Confidence Factoring:} Compute joint class detection confidence $\text{conf}_c = \sigma(t_{\text{obj}}) \sigma(t_{\text{cls}, c})$.
    \item \textbf{Coordinate Box Transformation:} Convert center format $(b_x, b_y, b_w, b_h)$ into corner representation $(x_1, y_1, x_2, y_2)$.
\end{enumerate}

\section{Algorithm Specification}

\begin{algorithm}[H]
\caption{Single-Stage Dense Grid Decoder}
\label{alg:yolo_decode}
\begin{algorithmic}[1]
\Require Dense tensor $T \in \mathbb{R}^{S_y \times S_x \times B \times (5 + C)}$, anchor priors $\{(p_{w, b}, p_{h, b})\}_{b=1}^B$, stride $s$, threshold $\tau$.
\Ensure Set of detected objects $\mathcal{D} = \{(x_1, y_1, x_2, y_2, \text{score}, \text{class})\}$.
\State $\mathcal{D} \gets \emptyset$
\For{$c_y = 0$ \textbf{to} $S_y - 1$}
    \For{$c_x = 0$ \textbf{to} $S_x - 1$}
        \For{$b = 1$ \textbf{to} $B$}
            \State $(t_x, t_y, t_w, t_h, t_{\text{obj}}) \gets T[c_y, c_x, b, 0:5]$
            \State $p_{\text{obj}} \gets \sigma(t_{\text{obj}})$
            \State $b_x \gets (\sigma(t_x) + c_x) \cdot s$
            \State $b_y \gets (\sigma(t_y) + c_y) \cdot s$
            \State $b_w \gets p_{w, b} \cdot \exp(t_w) \cdot s$
            \State $b_h \gets p_{h, b} \cdot \exp(t_h) \cdot s$
            \For{$k = 1$ \textbf{to} $C$}
                \State $p_{\text{cls}} \gets \sigma(T[c_y, c_x, b, 5 + k - 1])$
                \State $\text{score} \gets p_{\text{obj}} \cdot p_{\text{cls}}$
                \If{$\text{score} \ge \tau$}
                    \State $x_1 \gets b_x - 0.5 b_w, \quad y_1 \gets b_y - 0.5 b_h$
                    \State $x_2 \gets b_x + 0.5 b_w, \quad y_2 \gets b_y + 0.5 b_h$
                    \State $\mathcal{D} \gets \mathcal{D} \cup \{(x_1, y_1, x_2, y_2, \text{score}, k)\}$
                \EndIf
            \EndFor
        \EndFor
    \EndFor
\EndFor
\State \Return $\mathcal{D}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Spatial Boundedness}

\begin{theorem}[Spatial Center Cell Confinement]
\label{thm:grid_confinement}
Let the predicted center coordinate in grid-space be $u_x = c_x + \sigma(t_x)$. For any logit $t_x \in (-\infty, \infty)$, the decoded coordinate satisfies the strict spatial bound:
\begin{equation}
u_x \in (c_x, c_x + 1)
\end{equation}
preventing numerical gradient instability and resolving anchor assignment ambiguity during early training.
\end{theorem}

\begin{proof}
The standard logistic sigmoid $\sigma: \mathbb{R} \to (0, 1)$ is strictly monotonically increasing with $\lim_{z \to -\infty} \sigma(z) = 0$ and $\lim_{z \to \infty} \sigma(z) = 1$. Adding discrete cell offset $c_x$ yields the open interval $(c_x, c_x + 1)$.
\end{proof}

\section{Complexity Analysis}

\begin{itemize}
    \item \textbf{Time Complexity:} $\mathcal{O}(S_y \cdot S_x \cdot B \cdot C)$ operations for decoding grid cells.
    \item \textbf{Space Complexity:} $\mathcal{O}(S_y \cdot S_x \cdot B \cdot (5 + C))$ memory footprint for output tensors.
\end{itemize}

\section{Exactness and Error Analysis}

The Complete IoU (CIoU) loss incorporates aspect ratio consistency $v = \frac{4}{\pi^2} (\arctan \frac{w^{gt}}{h^{gt}} - \arctan \frac{w}{h})^2$, guaranteeing invariant geometric optimization across scales.

\section{Worked Numerical Example}

Let grid cell $(c_x, c_y) = (2, 3)$, stride $s = 32$, anchor prior $(p_w, p_h) = (2.0, 3.0)$.
Let predictions be $(t_x, t_y, t_w, t_h, t_{\text{obj}}, t_{\text{cls}}) = (0.0, 0.0, 0.0, 0.0, 2.0, 1.5)$.
\begin{itemize}
    \item $\sigma(0.0) = 0.5$.
    \item $b_x = (0.5 + 2) \times 32 = 2.5 \times 32 = 80.0$.
    \item $b_y = (0.5 + 3) \times 32 = 3.5 \times 32 = 112.0$.
    \item $b_w = 2.0 \times e^0 \times 32 = 64.0$.
    \item $b_h = 3.0 \times e^0 \times 32 = 96.0$.
    \item $x_1 = 80.0 - 32.0 = 48.0, \quad y_1 = 112.0 - 48.0 = 64.0$.
    \item $x_2 = 80.0 + 32.0 = 112.0, \quad y_2 = 112.0 + 48.0 = 160.0$.
    \item $p_{\text{obj}} = \sigma(2.0) \approx 0.8808, \quad p_{\text{cls}} = \sigma(1.5) \approx 0.8176$.
    \item Final score $= 0.8808 \times 0.8176 \approx 0.7201$.
\end{itemize}

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Focal Loss Calibration:} When training without objectness priors, focal loss with $\gamma = 2.0, \alpha = 0.25$ is mandatory to prevent negative background gradients from overwhelming rare positive signals.
    \item \textbf{Soft Clamping:} Clamping $t_w, t_h \le 10.0$ avoids floating point overflow in $\exp(t_w)$.
\end{itemize}
