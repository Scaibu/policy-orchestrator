\begin{abstract}
Region Proposal Networks (RPN) introduce translation-invariant convolutional anchor prediction, transforming object detection from external search mechanisms (e.g., Selective Search) into a unified, end-to-end differentiable neural framework. Operating directly on convolutional feature maps, RPN simultaneously predicts objectness probabilities and geometric boundary offsets for $K$ reference anchor boxes per spatial coordinate. This document establishes the scale-invariant coordinate parameterization, proves translation equivariance of the anchor grid, provides complete algorithmic pseudocode, and walks through a verified numerical decoding computation.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $F \in \mathbb{R}^{C \times H \times W}$ denote the shared backbone convolutional feature map. At each spatial grid location $(i, j) \in [0, H-1] \times [0, W-1]$, $K$ anchor templates $\mathcal{A}_{i,j,k} = (x_a, y_a, w_a, h_a)$ are parameterized.

\toprule
\textbf{Symbol} & \textbf{Domain} & \textbf{Semantic Description} \\
\midrule
$s$ & $\mathbb{R}_{> 0}$ & Backbone spatial stride ($s = 16$ typical) \\
$K$ & $\mathbb{N}$ & Number of anchor templates per grid cell ($K = 9$ standard) \\

$p_i^*$ & $\{0, 1\}$ & Ground-truth label ($1$ positive, $0$ negative, $-1$ ignored) \\

\section{Problem Definition and Core Concepts}

\subsection{3-Level Explanation}
\begin{enumerate}
    \item \textbf{Intuitive (High School level):} Imagine laying a grid of differently shaped cookie-cutters (tall, square, wide) over every spot on a photo. A smart mini-network looks through each cutter and answers two questions: ``Is there an object inside?'' and ``How should we stretch and shift this cutter to perfectly frame it?''
    \item \textbf{Engineering Level:} RPN slides a $3 \times 3$ convolutional window over shared feature maps, feeding into two sibling $1 \times 1$ conv heads: a classification branch outputting $2K$ objectness logits (foreground/background) and a regression branch outputting $4K$ coordinate offsets. Top-scoring proposals are filtered with Non-Maximum Suppression before being passed to stage-2 classification and RoI feature extraction.
    \item \textbf{Mathematical Level:} RPN solves a joint classification-regression multi-task optimization:
    \begin{equation}
    \mathcal{L}_{\text{RPN}}(\{p_i\}, \{t_i\}) = \frac{1}{N_{\text{cls}}} \sum_{i} \mathcal{L}_{\text{cls}}(p_i, p_i^*) + \lambda \frac{1}{N_{\text{reg}}} \sum_{i} p_i^* \sum_{c \in \{x,y,w,h\}} \text{Smooth}_{L1}(t_{i,c} - t_{i,c}^*)
    \end{equation}
    under scale-invariant log-space coordinate mapping:
    \begin{equation}
    t_x^* = \frac{x^* - x_a}{w_a}, \quad t_y^* = \frac{y^* - y_a}{h_a}, \quad t_w^* = \log\left(\frac{w^*}{w_a}\right), \quad t_h^* = \log\left(\frac{h^*}{h_a}\right)
    \end{equation}
\end{enumerate}

\subsection{5-Step Algorithmic Framework}
\begin{enumerate}
    \item \textbf{Anchor Field Generation:} Project $K$ canonical scale-ratio templates across the $H \times W$ spatial grid with stride $s$.
    \item \textbf{Sliding Window Feature Mapping:} Convolve feature map $F$ with $3 \times 3$ kernel producing intermediate embedding $Z \in \mathbb{R}^{d \times H \times W}$.
    \item \textbf{Twin Head Prediction:} Evaluate $1 \times 1$ convs for objectness logits ($2K$ channels) and box regression deltas ($4K$ channels).
    \item \textbf{IoU Ground-Truth Assignment:} Assign positive labels ($p_i^* = 1$) to anchors with highest IoU or $\text{IoU} \ge 0.7$; negative labels ($p_i^* = 0$) for $\text{IoU} < 0.3$.
    \item \textbf{Proposal Decoding and Boundary Clipping:} Invert parameterization to produce image-space candidate boxes and clip to image canvas boundaries.
\end{enumerate}

\section{Algorithm Specification}

\begin{algorithm}[H]
\caption{RPN Proposal Decoding and Filtering}
\label{alg:rpn_decode}
\begin{algorithmic}[1]
\Require Anchors $\mathcal{A} \in \mathbb{R}^{N \times 4}$, predicted deltas $\Delta \in \mathbb{R}^{N \times 4}$, objectness scores $S \in \mathbb{R}^N$, image bounds $(H_{\text{img}}, W_{\text{img}})$, score threshold $\tau$.
\Ensure Decoded proposal boxes $\mathcal{B}_{\text{prop}} \in \mathbb{R}^{M \times 4}$.
\State $\mathcal{B}_{\text{prop}} \gets [\,]$
\For{$i = 1$ \textbf{to} $N$}
    \If{$S[i] \ge \tau$}
        \State $(x_a, y_a, w_a, h_a) \gets \mathcal{A}[i]$
        \State $(dx, dy, dw, dh) \gets \Delta[i]$
        \State $x_{\text{pred}} \gets x_a + dx \cdot w_a$
        \State $y_{\text{pred}} \gets y_a + dy \cdot h_a$
        \State $w_{\text{pred}} \gets w_a \cdot \exp(dw)$
        \State $h_{\text{pred}} \gets h_a \cdot \exp(dh)$
        \State $x_1 \gets \max(0, \min(W_{\text{img}}, x_{\text{pred}} - 0.5 w_{\text{pred}}))$
        \State $y_1 \gets \max(0, \min(H_{\text{img}}, y_{\text{pred}} - 0.5 h_{\text{pred}}))$
        \State $x_2 \gets \max(0, \min(W_{\text{img}}, x_{\text{pred}} + 0.5 w_{\text{pred}}))$
        \State $y_2 \gets \max(0, \min(H_{\text{img}}, y_{\text{pred}} + 0.5 h_{\text{pred}}))$
        \State Append $(x_1, y_1, x_2, y_2)$ to $\mathcal{B}_{\text{prop}}$
    \EndIf
\EndFor
\State \Return $\mathcal{B}_{\text{prop}}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Translation Invariance}

\begin{theorem}[Translation Equivariance of Anchor Predictions]
\label{thm:translation_invariance}
Let $\mathcal{T}_{\Delta \mathbf{x}}$ denote a discrete spatial translation operator on the input image $X$ by integer stride multiples $\Delta \mathbf{x} = (s \Delta i, s \Delta j)$. The predicted anchor objectness scores and parameterized box displacements are equivariant:
\begin{equation}
\mathcal{P}(\mathcal{T}_{\Delta \mathbf{x}} X) = \mathcal{T}_{(\Delta i, \Delta j)} \mathcal{P}(X)
\end{equation}
guaranteeing consistent proposal detection irrespective of absolute object position.
\end{theorem}

\begin{proof}
Because all operations in the RPN (feature extraction backbone, $3 \times 3$ sliding window, and $1 \times 1$ sibling prediction heads) consist solely of convolutional operators, convolution commutes with spatial translations: $\text{Conv}(\mathcal{T} X) = \mathcal{T}(\text{Conv} X)$. Hence, shifting the input image by $s$ pixels translates the anchor grid activations identically by 1 feature cell.
\end{proof}

\section{Complexity Analysis}

\begin{itemize}
    \item \textbf{Time Complexity:}
    \begin{itemize}
        \item Intermediate $3 \times 3$ conv: $\mathcal{O}(9 \cdot C \cdot d \cdot H \cdot W)$.
        \item Sibling $1 \times 1$ convs: $\mathcal{O}((2K + 4K) \cdot d \cdot H \cdot W) = \mathcal{O}(6K \cdot d \cdot H \cdot W)$.
        \item Anchor decoding: $\mathcal{O}(K \cdot H \cdot W)$.
        \item Total Time: $\mathcal{O}((9C + 6K) d \cdot H \cdot W)$.
    \end{itemize}
    \item \textbf{Space Complexity:} $\mathcal{O}(K \cdot H \cdot W)$ for storing proposals and objectness logits.
\end{itemize}

\section{Exactness and Error Analysis}

The log-scale parameterization $t_w = \log(w / w_a)$ enforces positivity $w = w_a e^{t_w} > 0$ for all real-valued regression outputs $t_w \in \mathbb{R}$, eliminating degenerately inverted bounding boxes.

\section{Worked Numerical Example}

Let an anchor box be located at center $(x_a, y_a) = (100.0, 100.0)$ with width $w_a = 50.0$ and height $h_a = 50.0$.
Suppose RPN predicts deltas $(dx, dy, dw, dh) = (0.2, -0.1, \ln(1.2), \ln(0.8))$.
\begin{itemize}
    \item $x_{\text{pred}} = 100.0 + 0.2 \times 50.0 = 100.0 + 10.0 = 110.0$.
    \item $y_{\text{pred}} = 100.0 + (-0.1) \times 50.0 = 100.0 - 5.0 = 95.0$.
    \item $w_{\text{pred}} = 50.0 \times e^{\ln(1.2)} = 50.0 \times 1.2 = 60.0$.
    \item $h_{\text{pred}} = 50.0 \times e^{\ln(0.8)} = 50.0 \times 0.8 = 40.0$.
    \item Corners:
    $x_1 = 110.0 - 0.5(60.0) = 80.0$, $y_1 = 95.0 - 0.5(40.0) = 75.0$,
    $x_2 = 110.0 + 0.5(60.0) = 140.0$, $y_2 = 95.0 + 0.5(40.0) = 115.0$.
\end{itemize}
The decoded proposal is $(x_1, y_1, x_2, y_2) = (80.0, 75.0, 140.0, 115.0)$.

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Exponential Clamping:} Very large predicted $dw, dh$ can cause $\exp(dw)$ to overflow floating point representations. A safety clamp $dw \gets \min(dw, 10.0)$ is strictly enforced.
    \item \textbf{Cross-Boundary Anchors:} Anchors extending beyond the image boundaries during training are typically ignored to avoid noisy edge gradients.
\end{itemize}
