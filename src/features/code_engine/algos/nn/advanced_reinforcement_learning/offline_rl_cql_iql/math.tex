\section{Definitions and Cognitive Mental Models}

\subsection{Formal Mathematical Definition}
\textbf{Definition (Distribution Shift and Pessimistic Dynamic Programming):}
In offline reinforcement learning, an agent must learn an optimal policy $\pi^*$ exclusively from a static dataset of previously collected interactions $\mathcal{D} = \{(s, a, r, s')\}$, without active environment exploration. Standard off-policy RL fails catastrophically in this regime: the Bellman target queries Q-values for actions outside the dataset support ($a \notin \operatorname{supp}(\mathcal{D})$), where function approximators suffer from out-of-distribution (OOD) extrapolation errors. The policy greedily exploits these erroneously high Q-values, producing catastrophic policy collapse.

\textbf{Conservative Q-Learning (CQL)} solves this by augmenting the temporal difference objective with an explicit pessimistic regularizer that penalizes Q-values under out-of-distribution actions while rewarding Q-values under the empirical data distribution:
{\boldmath
\begin{equation}
\min_Q \alpha_{\text{CQL}} \left( \mathbb{E}_{s \sim \mathcal{D}} \left[ \ln \sum_{a \in \mathcal{A}} \exp(Q(s, a)) - \mathbb{E}_{a \sim \mathcal{D}}[Q(s, a)] \right] \right) + \frac{1}{2}\mathbb{E}_{(s, a, r, s') \sim \mathcal{D}}\left[ \left( Q(s, a) - \mathcal{B}^{\pi} Q_k(s, a) \right)^2 \right]
\end{equation}
}
Under CQL, the learned value function is mathematically guaranteed to be a lower bound on the true policy value: $Q^{\text{CQL}}(s, a) \le Q^\pi(s, a)$.

\textbf{Implicit Q-Learning (IQL)} takes an alternative, non-pessimistic route: it avoids querying the Q-function on unobserved actions altogether. Instead, it extracts the upper tail of the dataset value distribution using \textbf{Expectile Regression}:
{\boldmath
\begin{equation}
L_2^\tau(u) = |\tau - \mathbb{I}(u < 0)| u^2, \quad \text{where } u = Q(s, a) - V(s)
\end{equation}
}
with expectile parameter $\tau \in (0.5, 1.0)$ (typically $\tau \in [0.7, 0.9]$). Target values for the critic are formed simply as $y = r + \gamma V(s')$, completely bypassing out-of-distribution maximization. The policy is then extracted via Advantage-Weighted Regression (AWR):
{\boldmath
\begin{equation}
\max_\pi \mathbb{E}_{(s, a) \sim \mathcal{D}} \left[ \exp\left( \beta \left( Q(s, a) - V(s) \right) \right) \ln \pi(a | s) \right]
\end{equation}
}

\subsection{Expanded Non-Technical Definition (Intuition for Anyone)}
\textbf{Intuitive Conceptual Definition:}
\begin{enumerate}
    \item \textbf{Learning from History Without New Testing}: In medicine, finance, or nuclear power plants, you cannot let a robot experiment blindly on real patients or live reactors to see what happens. You only have old log files of what human operators previously did.
    \item \textbf{The Trap of the Unknown}: If a neural network has never seen an action in the logs, it might guess: ``Doing a triple backflip in a nuclear control room gives 10 million points!'' Standard RL blindly believes this hallucination and commands the reactor to do a backflip.
    \item \textbf{Conservative Caution (CQL)}: CQL forces the computer to be deeply suspicious: ``Unless I see solid evidence in the old logs that an action works, I assume its value is terribly low!''
    \item \textbf{Implicit Wisdom (IQL)}: IQL refuses to even ask what unknown actions do. It simply looks at the actions that humans actually tried, finds the top 10\% of those historical decisions, and trains the agent to imitate those best moments.
\end{enumerate}

\subsection{Child-Friendly Mental Model (Physical Metaphor)}
Imagine an apprentice chef learning from an old cookbook:
\begin{itemize}
    \item \textbf{The Careless Cook}: Tries to invent bizarre combinations (soup with motor oil) because they dream it might taste amazing, ruining the kitchen!
    \item \textbf{The CQL Cook}: Only cooks recipes verified in the book. If an idea is completely unmentioned, the chef assumes it burns the pan and avoids it.
    \item \textbf{The IQL Cook}: Reads all 500 recipes written in the book, flags the 50 recipes that received five-star reviews from past guests, and masters those exact recipes!
\end{itemize}

\section{Core Mathematical Equations and Definitions}

\subsection{CQL Conservative LogSumExp Penalty}
{\boldmath
\begin{equation}
\operatorname{Gap}_{\text{CQL}}(s) = \ln \left( \sum_{a \in \mathcal{A}} \exp(Q(s, a)) \right) - \mathbb{E}_{a \sim \mathcal{D}}[Q(s, a)]
\end{equation}
}
\begin{itemize}
    \item \textbf{Formal Definition}: The difference between the soft maximum Q-value across candidate actions and the empirical mean dataset Q-value.
    \item \textbf{What It Means}: Pushes down unobserved action values while anchoring observed action values to data evidence.
    \item \textbf{Why It Is Important}: Mathematically guarantees that estimated Q-values form a pessimistic lower bound on real policy performance.
\end{itemize}

\subsection{IQL Asymmetric Expectile Value Loss}
{\boldmath
\begin{equation}
L_2^\tau(Q(s, a) - V(s)) = |\tau - \mathbb{I}(Q(s, a) - V(s) < 0)| \cdot (Q(s, a) - V(s))^2
\end{equation}
}
\begin{itemize}
    \item \textbf{Formal Definition}: Asymmetric quadratic loss that penalizes underestimation with weight $\tau$ and overestimation with weight $1 - \tau$.
    \item \textbf{What It Means}: When $\tau > 0.5$, $V(s)$ estimates the upper envelope of values achieved by dataset actions without computing explicit maximums.
    \item \textbf{Why It Is Important}: Avoids evaluating out-of-distribution actions, completely eliminating OOD extrapolation instability.
\end{itemize}

\subsection{Advantage-Weighted Regression Policy Exponent}
{\boldmath
\begin{equation}
w(s, a) = \exp\left( \beta \left( Q(s, a) - V(s) \right) \right)
\end{equation}
}
\begin{itemize}
    \item \textbf{Formal Definition}: Exponentiated advantage weighting assigned to each dataset transition during policy extraction.
    \item \textbf{What It Means}: Assigns high probability mass to state-action pairs where the action outperformed the state expectile baseline $V(s)$.
    \item \textbf{Why It Is Important}: Enables stable, closed-form policy distillation directly through supervised regression.
\end{itemize}

\section{Rigorous Mathematical Analysis Checklist}

\begin{itemize}
    \item \textbf{Notation}: $\mathcal{D}$ (offline dataset), $\alpha_{\text{CQL}}$ (pessimism scale), $\tau \in (0.5, 1)$ (expectile), $\beta$ (inverse temperature), $V(s)$ (expectile value).
    \item \textbf{Epistemic Status}: Formulated by Kumar et al. (CQL, NeurIPS 2020) and Kostrikov et al. (IQL, ICLR 2022).
    \item \textbf{Dependency Graph}: Offline dataset $\mathcal{D} \to (Q, V) \to \text{Expectile / CQL Loss} \to \text{Advantage } Q - V \to \text{Policy Extraction}$.
    \item \textbf{Dimensions}: Transition batch $N$, candidate actions $A$, state dimension $d_S$.
    \item \textbf{Regimes}: Medical treatment recommendation, quantitative trading, industrial robot operations, web search ranking from logs.
    \item \textbf{Invariants}: If candidate actions equal only the observed dataset action, $\operatorname{Gap}_{\text{CQL}} \equiv 0.0$ identically.
    \item \textbf{Isomorphisms}: IQL expectile regression is isomorphic to asymmetric $L_2$ quantile regression for conditional distribution envelopes.
\end{itemize}

\section{Theoretical Theorems and Formal Proofs}

\begin{theorem}[CQL Pointwise Value Lower-Bound Property]
Under the Conservative Q-Learning objective with sufficiently large regularizer weight $\alpha_{\text{CQL}}$, the expected learned Q-value is a lower bound on the true expected Q-value under the policy $\pi$:
{\boldmath
\begin{equation}
\mathbb{E}_{s \sim d^\pi, a \sim \pi}[Q^{\text{CQL}}(s, a)] \le \mathbb{E}_{s \sim d^\pi, a \sim \pi}[Q^\pi(s, a)]
\end{equation}
}
\end{theorem}

\begin{proof}
Let $\hat{\mathcal{B}}^\pi$ denote the empirical Bellman evaluation operator, and let $\Delta_k(s, a) = Q_k(s, a) - Q^\pi(s, a)$.
The CQL objective updates the Q-function at iteration $k+1$ according to:
{\boldmath
\begin{equation}
Q_{k+1}(s, a) = \hat{\mathcal{B}}^\pi Q_k(s, a) - \alpha_{\text{CQL}} \left( \frac{\pi(a | s)}{\hat{\pi}_\beta(a | s)} - 1 \right)
\end{equation}
}
where $\hat{\pi}_\beta(a | s)$ is the empirical behavior policy of the dataset.
Subtracting the true Bellman identity $Q^\pi(s, a) = \mathcal{B}^\pi Q^\pi(s, a)$:
{\boldmath
\begin{equation}
\Delta_{k+1}(s, a) = \left( \hat{\mathcal{B}}^\pi Q_k(s, a) - \mathcal{B}^\pi Q^\pi(s, a) \right) - \alpha_{\text{CQL}} \left( \frac{\pi(a | s)}{\hat{\pi}_\beta(a | s)} - 1 \right)
\end{equation}
}
Taking expectations under the state-action visitation distribution $d^\pi(s)\pi(a | s)$:
{\boldmath
\begin{align}
\mathbb{E}_{d^\pi, \pi}[\Delta_{k+1}(s, a)] &= \gamma \mathbb{E}_{d^\pi, \pi}[\Delta_k(s, a)] + \mathbb{E}_{d^\pi, \pi}[\hat{\mathcal{B}}^\pi Q_k - \mathcal{B}^\pi Q_k] - \alpha_{\text{CQL}} \mathbb{E}_{s \sim d^\pi}\left[ \sum_a \frac{\pi^2(a|s)}{\hat{\pi}_\beta(a|s)} - 1 \right]
\end{align}
}
Notice that $\sum_a \frac{\pi^2(a|s)}{\hat{\pi}_\beta(a|s)} - 1 = D_{\chi^2}(\pi(\cdot|s) \,||\, \hat{\pi}_\beta(\cdot|s)) \ge 0$.
The sampling error term $|\mathbb{E}[\hat{\mathcal{B}}^\pi Q_k - \mathcal{B}^\pi Q_k]| \le \frac{C}{\sqrt{|\mathcal{D}|}}$ with high probability.
Choosing $\alpha_{\text{CQL}} \ge \frac{C}{(1 - \gamma) \sqrt{|\mathcal{D}|} D_{\chi^2}}$ ensures that:
{\boldmath
\begin{equation}
\mathbb{E}_{d^\pi, \pi}[\Delta_{k+1}(s, a)] \le \gamma \mathbb{E}_{d^\pi, \pi}[\Delta_k(s, a)]
\end{equation}
}
Iterating to the fixed point $k \to \infty$ proves that $\mathbb{E}_{d^\pi, \pi}[Q^{\text{CQL}}(s, a) - Q^\pi(s, a)] \le 0$, establishing the lower bound.
\end{proof}

\begin{theorem}[Upper Envelope Bound of Expectile Value Estimates]
Let $X$ be a random variable representing dataset action values at state $s$, and let $v_\tau = \arg\min_v \mathbb{E}[L_2^\tau(X - v)]$.
Then for any $\tau \in [0.5, 1.0)$:
{\boldmath
\begin{equation}
v_\tau \ge \mathbb{E}[X]
\end{equation}
}
with strict inequality if $\operatorname{Var}(X) > 0$ and $\tau > 0.5$.
\end{theorem}

\begin{proof}
The first-order optimality condition for the expectile loss $L_2^\tau(u) = |\tau - \mathbb{I}(u < 0)| u^2$ requires the derivative to vanish:
{\boldmath
\begin{align}
\frac{d}{dv} \mathbb{E}[L_2^\tau(X - v)] &= -2 \mathbb{E}\left[ |\tau - \mathbb{I}(X < v)| (X - v) \right] = 0 \\
\tau \int_{v}^\infty (x - v) \, dF(x) &= (1 - \tau) \int_{-\infty}^v (v - x) \, dF(x)
\end{align}
}
Rearranging:
{\boldmath
\begin{equation}
v - \mathbb{E}[X] = \frac{2\tau - 1}{1 - \tau} \int_v^\infty (x - v) \, dF(x)
\end{equation}
}
For $\tau > 0.5$, $\frac{2\tau - 1}{1 - \tau} > 0$.
Because $(x - v) > 0$ on the integration domain $(v, \infty)$, the integral is strictly positive if the upper tail has positive measure.
Therefore, $v_\tau - \mathbb{E}[X] > 0$, confirming that $v_\tau$ estimates an upper envelope of values without computing an explicit maximum over out-of-distribution actions.
\end{proof}

\section{Foundational Architectural Questions and Epistemic Meta-Questions}

\subsection{Architectural Question 1: CQL vs IQL Trade-offs}
\textbf{Question:} Under what dataset conditions is IQL preferred over CQL?\\
\textbf{Answer:} CQL requires computing LogSumExp over candidate actions. In high-dimensional continuous action spaces, this requires generating multiple OOD action samples per transition, adding substantial computational overhead. IQL requires only scalar expectile regression on observed actions, making it significantly faster, mathematically simpler, and less prone to hyperparameter sensitivity.

\textbf{Meta-Question:} Why does standard actor-critic fail on offline datasets even when using large networks?\\
\textbf{Answer:} Increasing neural network capacity actually worsens the offline failure mode! Larger networks have higher expressive capacity to fit spurious peaks in unobserved regions of the state-action space, accelerating policy drift into out-of-distribution fantasy regimes.

\subsection{Architectural Question 2: Expectile Parameter Tuning}
\textbf{Question:} How does the expectile parameter $\tau$ control the trade-off between behavior cloning and dynamic programming?\\
\textbf{Answer:} When $\tau = 0.5$, $V(s) = \mathbb{E}[Q(s, a)]$, reducing the algorithm to standard SARSA. As $\tau \to 1.0$, $V(s) \to \max_{a} Q(s, a)$, approaching maximum Q-learning. Setting $\tau \in [0.7, 0.9]$ extracts the best actions present in the data without querying non-existent extremes.

\textbf{Meta-Question:} How is offline RL evaluated safely before real-world deployment?\\
\textbf{Answer:} Production offline policies are evaluated using Off-Policy Evaluation (OPE) techniques, including Doubly Robust estimators, Fitted Q-Evaluation (FQE), and marginalized importance sampling, providing statistical confidence bounds on expected returns before any real traffic is routed to the model.

\section{Algorithmic Implementation Specification}

\begin{algorithm}[H]
\caption{Offline RL Conservative and Expectile Evaluation}
\begin{algorithmic}[1]
\Require Dataset values $\mathbf{q}_{\text{data}}$, candidate actions $\mathbf{Q}_{\text{cands}}$, targets $\mathbf{y}$, values $\mathbf{V}$, $\alpha_{\text{CQL}}, \tau, \beta$
\Ensure Bellman loss $\mathcal{L}_{\text{bell}}$, CQL penalty $\mathcal{L}_{\text{cql}}$, CQL total loss $\mathcal{L}_{\text{critic}}$, IQL value loss $\mathcal{L}_{\text{IQL}}$, weights $\mathbf{w}$
\State Initialize $bell\_err \gets 0.0, \quad cql\_accum \gets 0.0, \quad iql\_accum \gets 0.0$
\State Allocate $\mathbf{w} \in \mathbb{R}^N$
\For{$i = 0$ \textbf{to} $N-1$}
    \State $diff \gets \mathbf{q}_{\text{data}}[i] - \mathbf{y}[i]$
    \State $bell\_err \gets bell\_err + diff^2$
    \State $m \gets \max(\mathbf{Q}_{\text{cands}}[i])$
    \State $lse \gets m + \ln\left( \sum_{c \in \mathbf{Q}_{\text{cands}}[i]} \exp(c - m) \right)$
    \State $cql\_accum \gets cql\_accum + (lse - \mathbf{q}_{\text{data}}[i])$
    \State $u \gets \mathbf{q}_{\text{data}}[i] - \mathbf{V}[i]$
    \State $wt \gets \tau$ \textbf{if} $u \ge 0$ \textbf{else} $(1 - \tau)$
    \State $iql\_accum \gets iql\_accum + wt \cdot u^2$
    \State $\mathbf{w}[i] \gets \exp(\operatorname{clamp}(\beta \cdot u, -100.0, 100.0))$
\EndFor
\State $\mathcal{L}_{\text{bell}} \gets bell\_err / N, \quad \mathcal{L}_{\text{cql}} \gets cql\_accum / N$
\State $\mathcal{L}_{\text{critic}} \gets \mathcal{L}_{\text{bell}} + \alpha_{\text{CQL}} \cdot \mathcal{L}_{\text{cql}}$
\State $\mathcal{L}_{\text{IQL}} \gets iql\_accum / N$
\State \Return $(\mathcal{L}_{\text{bell}}, \mathcal{L}_{\text{cql}}, \mathcal{L}_{\text{critic}}, \mathcal{L}_{\text{IQL}}, \mathbf{w})$
\end{algorithmic}
\end{algorithm}

\section{Big-$\Theta$ Complexity Analysis}

\begin{itemize}
    \item \textbf{CQL LogSumExp Time}: $\Theta(N \cdot |\mathcal{A}|)$ operations across candidates.
    \item \textbf{IQL Expectile Loss Time}: $\Theta(N)$ asymmetric quadratic steps.
    \item \textbf{Advantage Weighting Time}: $\Theta(N)$ exponential evaluations.
    \item \textbf{Space Complexity}: $\Theta(N \cdot |\mathcal{A}|)$ buffer for candidate evaluations.
    \item \textbf{Parallel Reduction Depth}: $\mathcal{D} = \Theta(\log N)$ for mean loss reductions.
\end{itemize}
