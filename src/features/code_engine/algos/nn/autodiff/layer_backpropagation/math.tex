\begin{abstract}
Layer-wise backpropagation applies the multi-variable matrix chain rule to propagate error gradients across sequential affine transformations $\mathbf{Y} = \mathbf{X} \mathbf{W}^\top + \mathbf{1} \mathbf{b}^\top$. Given upstream adjoints $\frac{\partial \mathcal{L}}{\partial \mathbf{Y}}$, we derive the exact vector-Jacobian products yielding weight gradients $\frac{\partial \mathcal{L}}{\partial \mathbf{W}} = \left(\frac{\partial \mathcal{L}}{\partial \mathbf{Z}}\right)^\top \mathbf{X}$, input adjoints $\frac{\partial \mathcal{L}}{\partial \mathbf{X}} = \frac{\partial \mathcal{L}}{\partial \mathbf{Z}} \mathbf{W}$, and bias updates. We prove correctness, derive operation-by-operation complexity, and provide full pseudocode matching the reference implementation.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\mathbf{X} \in \mathbb{R}^{B \times D_{\mathrm{in}}}$ denote batch activations, $\mathbf{W} \in \mathbb{R}^{D_{\mathrm{out}} \times D_{\mathrm{in}}}$ weight parameters, $\mathbf{b} \in \mathbb{R}^{D_{\mathrm{out}}}$ bias parameters, and $\mathbf{Z} = \mathbf{X} \mathbf{W}^\top + \mathbf{1}_B \mathbf{b}^\top \in \mathbb{R}^{B \times D_{\mathrm{out}}}$ the pre-activation tensor.

\subsection{Matrix Calculus and Adjoint Dynamics}
\paragraph{Intuitive Meaning:} When backpropagating through a linear layer, the gradient with respect to weights is formed by outer products between incoming error signals and input features, while input gradients are formed by multiplying errors by transposed weights.

\paragraph{Formal Mathematical Definition:}
Let $\mathbf{\Delta} = \frac{\partial \mathcal{L}}{\partial \mathbf{Z}} = \frac{\partial \mathcal{L}}{\partial \mathbf{Y}} \odot \phi'(\mathbf{Z}) \in \mathbb{R}^{B \times D_{\mathrm{out}}}$.
The analytic parameter and activation gradients are:
\begin{align}
\label{eq:grad_w}
\frac{\partial \mathcal{L}}{\partial \mathbf{W}} &= \mathbf{\Delta}^\top \mathbf{X} \in \mathbb{R}^{D_{\mathrm{out}} \times D_{\mathrm{in}}}, \\
\label{eq:grad_x}
\frac{\partial \mathcal{L}}{\partial \mathbf{X}} &= \mathbf{\Delta} \mathbf{W} \in \mathbb{R}^{B \times D_{\mathrm{in}}}, \\
\label{eq:grad_b}
\frac{\partial \mathcal{L}}{\partial \mathbf{b}} &= \mathbf{\Delta}^\top \mathbf{1}_B = \sum_{i=1}^B \mathbf{\Delta}_{i, :} \in \mathbb{R}^{D_{\mathrm{out}}}.
\end{align}

\subsection{Summary of Symbols}
\begin{itemize}
    \item $B \in \mathbb{N}$: Mini-batch size.
    \item $D_{\mathrm{in}} \in \mathbb{N}$: Input feature dimensionality.
    \item $D_{\mathrm{out}} \in \mathbb{N}$: Output feature dimensionality.
    \item $\mathbf{X} \in \mathbb{R}^{B \times D_{\mathrm{in}}}$: Forward input activation matrix.
    \item $\mathbf{W} \in \mathbb{R}^{D_{\mathrm{out}} \times D_{\mathrm{in}}}$: Layer weight parameter matrix.
    \item $\mathbf{dY} \in \mathbb{R}^{B \times D_{\mathrm{out}}}$: Upstream incoming gradient matrix.
    \item $\mathbf{dX} \in \mathbb{R}^{B \times D_{\mathrm{in}}}$: Downstream propagated input gradient.
    \item $\mathbf{dW} \in \mathbb{R}^{D_{\mathrm{out}} \times D_{\mathrm{in}}}$: Weight parameter gradient matrix.
    \item $\mathbf{db} \in \mathbb{R}^{D_{\mathrm{out}}}$: Bias parameter gradient vector.
\end{itemize}

\section{Problem Definition and Core Concepts}
\label{sec:problem_definition}

\subsection{Activation Derivative Formulations}
\begin{itemize}
    \item \textbf{Linear ($\phi(z) = z$):} $\phi'(z) = 1 \implies \mathbf{\Delta} = \mathbf{dY}$.
    \item \textbf{ReLU ($\phi(z) = \max(0, z)$):} $\phi'(z) = \mathbb{I}_{\{z > 0\}} \implies \mathbf{\Delta} = \mathbf{dY} \odot \mathbb{I}_{\{\mathbf{Z} > 0\}}$.
    \item \textbf{Sigmoid ($\phi(z) = \sigma(z)$):} $\phi'(z) = \sigma(z)(1 - \sigma(z))$.
    \item \textbf{Tanh ($\phi(z) = \tanh(z)$):} $\phi'(z) = 1 - \tanh^2(z)$.
\end{itemize}

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:layer_backprop} details the exact matrix multiplication routine matching \texttt{impl.py}.

\begin{algorithm}[H]
\caption{Layer-by-Layer Affine Backpropagation}
\label{alg:layer_backprop}
\begin{algorithmic}[1]
\Require Inputs $\mathbf{X} \in \mathbb{R}^{B \times D_{\mathrm{in}}}$, weights $\mathbf{W} \in \mathbb{R}^{D_{\mathrm{out}} \times D_{\mathrm{in}}}$, upstream gradients $\mathbf{dY} \in \mathbb{R}^{B \times D_{\mathrm{out}}}$, activation $\mathrm{act}$, pre-activations $\mathbf{Z}$.
\Ensure Input gradients $\mathbf{dX}$, weight gradients $\mathbf{dW}$, bias gradients $\mathbf{db}$.
\State Initialize $\mathbf{\Delta} \gets [0.0]^{B \times D_{\mathrm{out}}}$
\For{$i = 1$ to $B$}
    \For{$j = 1$ to $D_{\mathrm{out}}$}
        \State $dy \gets dY_{i, j}$
        \If{$\mathrm{act} = \text{linear}$}
            \State $\Delta_{i, j} \gets dy$
        \ElsIf{$\mathrm{act} = \text{relu}$}
            \State $\Delta_{i, j} \gets dy \cdot \mathbb{I}_{\{Z_{i, j} > 0\}}$
        \ElsIf{$\mathrm{act} = \text{sigmoid}$}
            \State $s \gets \sigma(Z_{i, j}), \quad \Delta_{i, j} \gets dy \cdot s \cdot (1.0 - s)$
        \ElsIf{$\mathrm{act} = \text{tanh}$}
            \State $th \gets \tanh(Z_{i, j}), \quad \Delta_{i, j} \gets dy \cdot (1.0 - th^2)$
        \EndIf
    \EndFor
\EndFor
\State $\mathbf{dX} \gets \mathbf{\Delta} \cdot \mathbf{W}$ \Comment{Shape $(B, D_{\mathrm{in}})$}
\State $\mathbf{dW} \gets \mathbf{\Delta}^\top \cdot \mathbf{X}$ \Comment{Shape $(D_{\mathrm{out}}, D_{\mathrm{in}})$}
\State $\mathbf{db} \gets \sum_{i=1}^B \mathbf{\Delta}_{i, :}$ \Comment{Shape $(D_{\mathrm{out}},)$}
\State \Return $\mathbf{dX}, \quad \mathbf{dW}, \quad \mathbf{db}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Exactness}
\label{sec:guarantees}

\begin{theorem}[Exact Differential Form Invariance]
\label{thm:matrix_chain_rule}
\textbf{Assumptions:} Let affine transformation $\mathbf{Z} = \mathbf{X} \mathbf{W}^\top + \mathbf{b}$ be continuously differentiable with scalar loss $\mathcal{L}$.
\textbf{Guarantee:} The differential forms evaluate exactly as $d\mathcal{L} = \operatorname{Tr}(\mathbf{\Delta}^\top d\mathbf{Z}) = \operatorname{Tr}((\mathbf{\Delta}^\top \mathbf{X}) d\mathbf{W}^\top) + \operatorname{Tr}((\mathbf{W}^\top \mathbf{\Delta}^\top) d\mathbf{X}) + \mathbf{\Delta}^\top \mathbf{1} d\mathbf{b}$, yielding gradient expressions~\eqref{eq:grad_w}--\eqref{eq:grad_b} without approximation error.
\textbf{Proof:}
Substituting $d\mathbf{Z} = (d\mathbf{X})\mathbf{W}^\top + \mathbf{X}(d\mathbf{W})^\top + \mathbf{1}(d\mathbf{b})^\top$ into the trace formulation:
\begin{equation}
\operatorname{Tr}(\mathbf{\Delta}^\top d\mathbf{Z}) = \operatorname{Tr}(\mathbf{\Delta}^\top (d\mathbf{X})\mathbf{W}^\top) + \operatorname{Tr}(\mathbf{\Delta}^\top \mathbf{X} (d\mathbf{W})^\top) + \operatorname{Tr}(\mathbf{\Delta}^\top \mathbf{1} (d\mathbf{b})^\top).
\end{equation}
Using the cyclic trace property $\operatorname{Tr}(\mathbf{A}\mathbf{B}) = \operatorname{Tr}(\mathbf{B}\mathbf{A})$:
$\operatorname{Tr}(\mathbf{\Delta}^\top (d\mathbf{X})\mathbf{W}^\top) = \operatorname{Tr}(\mathbf{W}^\top \mathbf{\Delta}^\top d\mathbf{X}) = \operatorname{Tr}((\mathbf{\Delta}\mathbf{W})^\top d\mathbf{X}) \implies \frac{\partial \mathcal{L}}{\partial \mathbf{X}} = \mathbf{\Delta}\mathbf{W}$.
$\operatorname{Tr}(\mathbf{\Delta}^\top \mathbf{X} (d\mathbf{W})^\top) = \operatorname{Tr}(( \mathbf{\Delta}^\top \mathbf{X} ) d\mathbf{W}^\top) \implies \frac{\partial \mathcal{L}}{\partial \mathbf{W}} = \mathbf{\Delta}^\top \mathbf{X}$.
\textbf{Limitations:} Memory footprint requires caching activations $\mathbf{X}$ and $\mathbf{Z}$ across all active layers until the backward pass reaches them.
\end{theorem}

\section{Complexity Analysis}
\label{sec:complexity}

\begin{itemize}
    \item \textbf{Time Complexity:} $\mathcal{O}(B \cdot D_{\mathrm{in}} \cdot D_{\mathrm{out}})$ multiply-accumulate operations for $\mathbf{dX}$ and $\mathbf{dW}$.
    \item \textbf{Space Complexity:} $\mathcal{O}(B \cdot D_{\mathrm{in}} + D_{\mathrm{out}} \cdot D_{\mathrm{in}} + D_{\mathrm{out}})$ for output gradient matrices.
\end{itemize}

\section{Worked Numerical Example}
\label{sec:worked_example}

Let $B = 1, D_{\mathrm{in}} = 2, D_{\mathrm{out}} = 2, \mathbf{X} = [[1.0, 2.0]], \mathbf{W} = [[0.5, -0.5], [1.0, 0.0]], \mathbf{dY} = [[1.0, -1.0]]$ (linear activation):
\begin{enumerate}
    \item $\mathbf{\Delta} = [[1.0, -1.0]]$.
    \item $\mathbf{dX} = \mathbf{\Delta} \mathbf{W} = [[1.0(0.5) + (-1.0)(1.0), \quad 1.0(-0.5) + (-1.0)(0.0)]] = [[-0.5, -0.5]]$.
    \item $\mathbf{dW} = \mathbf{\Delta}^\top \mathbf{X} = \begin{bmatrix} 1.0 \\ -1.0 \end{bmatrix} \begin{bmatrix} 1.0 & 2.0 \end{bmatrix} = \begin{bmatrix} 1.0 & 2.0 \\ -1.0 & -2.0 \end{bmatrix}$.
    \item $\mathbf{db} = [1.0, -1.0]$.
\end{enumerate}
