\begin{abstract}
This document presents the complete theoretical derivation and algorithmic specification of Adafactor. In billion-parameter transformer architectures, maintaining uncompressed second-moment matrices requires auxiliary memory equal to the parameter count ($\mathcal{O}(R \times C)$). Adafactor compresses second moments by storing low-rank non-negative row and column factor vectors ($\mathcal{O}(R + C)$) through generalized I-divergence projection. We derive the rank-1 reconstruction equation, formalize the full algorithmic execution matching the reference implementation, prove optimal divergence properties, detail step-by-step Big-O complexity, calculate a complete numerical example, and discuss floating-point numerical considerations.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $W \in \mathbb{R}^{R \times C}$ denote a 2D weight matrix parameter tensor with $R$ rows and $C$ columns. Let $G \triangleq \nabla_W \mathcal{L}(W) \in \mathbb{R}^{R \times C}$ denote the corresponding gradient matrix.

\begin{itemize}
    \item $W_{ij} \in \mathbb{R}$: Parameter element at row $i \in \{1, \dots, R\}$ and column $j \in \{1, \dots, C\}$.
    \item $G_{ij} \in \mathbb{R}$: Gradient element at coordinate $(i, j)$.
    \item $v_{\text{row}} \in \mathbb{R}^R$: Row factor vector tracking mean squared gradients across columns.
    \item $v_{\text{col}} \in \mathbb{R}^C$: Column factor vector tracking mean squared gradients across rows.
    \item $\hat{V} \in \mathbb{R}^{R \times C}$: Factored rank-1 reconstructed second-moment matrix.
    \item $\beta_2 \in [0, 1)$: Exponential smoothing coefficient for second-moment factors.
    \item $\eta > 0$: Learning rate step size.
    \item $\epsilon > 0$: Denominator stabilization constant.
\end{itemize}

\section{Problem Definition and Core Concepts}

\subsection{Intuition and Motivation}
In large-scale deep learning models (e.g., GPT and T5 models exceeding 10B parameters), storing 32-bit floating point optimizer states for Adam (both first and second moments $m$ and $v$) consumes $2 \times 4 \times P = 8P$ bytes of memory—double the memory footprint of the weights themselves. Adafactor eliminates the dense second moment tensor $V \in \mathbb{R}^{R \times C}$ by tracking only marginal row and column averages $v_{\text{row}} \in \mathbb{R}^R$ and $v_{\text{col}} \in \mathbb{R}^C$, reducing memory from $\mathcal{O}(RC)$ down to $\mathcal{O}(R + C)$ (e.g., from 16,000,000 floats down to 8,000 floats for a $4096 \times 4096$ linear projection layer).

\subsection{Formal Mathematical Definition}
\begin{definition}[Rank-1 Factored Second-Moment Estimation]
Given empirical row and column mean squared gradients:
\begin{align}
\bar{g}_{\text{row}, i} &= \frac{1}{C} \sum_{j=1}^C G_{ij}^2 \quad \forall i \in \{1, \dots, R\} \label{eq:row_mean} \\
\bar{g}_{\text{col}, j} &= \frac{1}{R} \sum_{i=1}^R G_{ij}^2 \quad \forall j \in \{1, \dots, C\} \label{eq:col_mean}
\end{align}
The factored moving averages evolve via:
\begin{align}
v_{\text{row}, i}^{(t)} &= \beta_2 v_{\text{row}, i}^{(t-1)} + (1 - \beta_2) \bar{g}_{\text{row}, i} \label{eq:adafactor_vrow} \\
v_{\text{col}, j}^{(t)} &= \beta_2 v_{\text{col}, j}^{(t-1)} + (1 - \beta_2) \bar{g}_{\text{col}, j} \label{eq:adafactor_vcol}
\end{align}
The reconstructed coordinate second moment $\hat{V}_{ij}$ is synthesized as:
\begin{equation}
\hat{V}_{ij} = \frac{v_{\text{row}, i}^{(t)} \cdot v_{\text{col}, j}^{(t)}}{\sum_{k=1}^R v_{\text{row}, k}^{(t)} + \epsilon} \label{eq:adafactor_recon}
\end{equation}
and the parameter matrix is updated as:
\begin{equation}
W_{ij}^{(t+1)} = W_{ij}^{(t)} - \frac{\eta}{\sqrt{\hat{V}_{ij}} + \epsilon} G_{ij} \label{eq:adafactor_update}
\end{equation}
\end{definition}

\section{Algorithmic Specification}

The computational pipeline implemented in \texttt{NnAlgoAdafactorOptimizer} is presented below.

\begin{algorithm}[H]
\caption{Adafactor Factored Second-Moment Optimization Step}
\begin{algorithmic}[1]
\Require Weight matrix $W \in \mathbb{R}^{R \times C}$, gradient matrix $G \in \mathbb{R}^{R \times C}$
\Require Row factor $v_{\text{row}} \in \mathbb{R}^R$, column factor $v_{\text{col}} \in \mathbb{R}^C$
\Require Learning rate $\eta > 0$, decay factor $\beta_2 \in [0, 1)$, stability constant $\epsilon > 0$
\Ensure Updated matrix $W_{\text{next}} \in \mathbb{R}^{R \times C}$, factors $v_{\text{row}}^{\text{next}} \in \mathbb{R}^R, v_{\text{col}}^{\text{next}} \in \mathbb{R}^C$
\State $R \gets \text{rows}(W), \quad C \gets \text{cols}(W)$
\If{$R = 0$ \textbf{or} $\text{rows}(G) \ne R$ \textbf{or} $\text{cols}(G) \ne C$ \textbf{or} $\text{length}(v_{\text{row}}) \ne R$ \textbf{or} $\text{length}(v_{\text{col}}) \ne C$}
    \State \textbf{throw} PreconditionFailureException(``Matrix dimension mismatch'')
\EndIf
\State $r_{\text{means}} \gets \text{new Array of size } R$
\For{$i \gets 1 \textbf{ to } R$}
    \State $r_{\text{means}}[i] \gets \frac{1}{C} \sum_{j=1}^C (G[i][j])^2$
\EndFor
\State $c_{\text{means}} \gets \text{new Array of size } C$
\For{$j \gets 1 \textbf{ to } C$}
    \State $c_{\text{means}}[j] \gets \frac{1}{R} \sum_{i=1}^R (G[i][j])^2$
\EndFor
\State $v_{\text{row}}^{\text{next}} \gets \text{new Array of size } R$, \quad $v_{\text{col}}^{\text{next}} \gets \text{new Array of size } C$
\For{$i \gets 1 \textbf{ to } R$}
    \State $v_{\text{row}}^{\text{next}}[i] \gets \beta_2 \cdot v_{\text{row}}[i] + (1.0 - \beta_2) \cdot r_{\text{means}}[i]$
\EndFor
\For{$j \gets 1 \textbf{ to } C$}
    \State $v_{\text{col}}^{\text{next}}[j] \gets \beta_2 \cdot v_{\text{col}}[j] + (1.0 - \beta_2) \cdot c_{\text{means}}[j]$
\EndFor
\State $S_{\text{row}} \gets \sum_{i=1}^R v_{\text{row}}^{\text{next}}[i] + \epsilon$
\State $W_{\text{next}} \gets \text{new Matrix of shape } (R, C)$
\For{$i \gets 1 \textbf{ to } R$}
    \For{$j \gets 1 \textbf{ to } C$}
        \State $\hat{V}_{ij} \gets \frac{v_{\text{row}}^{\text{next}}[i] \cdot v_{\text{col}}^{\text{next}}[j]}{S_{\text{row}}}$
        \State $u_{ij} \gets \frac{G[i][j]}{\sqrt{\hat{V}_{ij}} + \epsilon}$
        \State $W_{\text{next}}[i][j] \gets W[i][j] - \eta \cdot u_{ij}$
    \EndFor
\EndFor
\State \Return $\{\text{updated\_weights}: W_{\text{next}}, \text{updated\_row\_factor}: v_{\text{row}}^{\text{next}}, \text{updated\_col\_factor}: v_{\text{col}}^{\text{next}}\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and I-Divergence Minimization}

\begin{theorem}[Generalized I-Divergence Optimality of Rank-1 Reconstruction]
Let $V \in \mathbb{R}_{\ge 0}^{R \times C}$ be any non-negative matrix with row marginals $r_i = \sum_{j=1}^C V_{ij}$ and column marginals $c_j = \sum_{i=1}^R V_{ij}$. The rank-1 matrix $\hat{V}_{ij} = \frac{r_i c_j}{\sum_k r_k}$ is the unique non-negative rank-1 matrix that minimizes the generalized I-divergence (Kullback-Leibler divergence for unnormalized measures):
\begin{equation}
D_{I}(V \parallel \hat{V}) = \sum_{i=1}^R \sum_{j=1}^C \left( V_{ij} \ln \frac{V_{ij}}{\hat{V}_{ij}} - V_{ij} + \hat{V}_{ij} \right)
\end{equation}
subject to matching exact marginal row and column sums.
\end{theorem}

\begin{proof}
Formulating the Lagrangian with constraints $\sum_j \hat{V}_{ij} = r_i$ and $\sum_i \hat{V}_{ij} = c_j$, the maximum entropy / minimum divergence distribution under independence of coordinate marginals factors as the product distribution normalized by the total sum $\sum_{i, j} V_{ij} = \sum_k r_k$. Substituting $\hat{V}_{ij} = \frac{r_i c_j}{\sum_k r_k}$ satisfies $\sum_{j=1}^C \hat{V}_{ij} = r_i \frac{\sum_j c_j}{\sum_k r_k} = r_i (1) = r_i$ and $\sum_{i=1}^R \hat{V}_{ij} = c_j \frac{\sum_i r_i}{\sum_k r_k} = c_j$, establishing uniqueness and exact marginal preservation.
\end{proof}

\subsection{Limitations}
\begin{itemize}
    \item \textbf{Rank-1 Assumption}: If gradients across rows and columns have high mutual covariance (e.g., block diagonal correlation), rank-1 factorization underestimates individual corner variances.
\end{itemize}

\section{Complexity Analysis}

\subsection{Time Complexity}
\begin{itemize}
    \item \textbf{Marginal Reductions}: Row sums take $\mathcal{O}(RC)$; column sums take $\mathcal{O}(RC)$ FLOPs.
    \item \textbf{Factor Updates}: Row/col EMA updates take $\mathcal{O}(R + C)$ FLOPs.
    \item \textbf{Weight Reconstruction and Update}: Iterating over $R \times C$ takes $\mathcal{O}(RC)$ FLOPs.
    \item \textbf{Total Time}: $\Theta(RC)$ FLOPs (same as standard Adam).
\end{itemize}

\subsection{Space Complexity}
\begin{itemize}
    \item \textbf{Auxiliary Memory}: Requires storing $v_{\text{row}} \in \mathbb{R}^R$ and $v_{\text{col}} \in \mathbb{R}^C$.
    \item \textbf{Total Space}: $\Theta(R + C)$ auxiliary floats, compared to $\Theta(RC)$ for Adam ($\approx 99.9\%$ memory savings for $R, C \ge 4096$).
\end{itemize}

\section{Worked Numerical Example}

Consider a $2 \times 2$ weight matrix $W$, gradient matrix $G$, row factors $v_{\text{row}}$, and column factors $v_{\text{col}}$:
\begin{equation}
W = \begin{bmatrix} 1.0 & 2.0 \\ 3.0 & 4.0 \end{bmatrix}, \quad G = \begin{bmatrix} 0.2 & 0.4 \\ 0.6 & 0.8 \end{bmatrix}, \quad v_{\text{row}} = \begin{bmatrix} 0.0 \\ 0.0 \end{bmatrix}, \quad v_{\text{col}} = \begin{bmatrix} 0.0 \\ 0.0 \end{bmatrix}
\end{equation}
with parameters $\eta = 0.01$, $\beta_2 = 0.5$, $\epsilon = 10^{-8}$.

\begin{enumerate}
    \item \textbf{Squared Gradient Matrix}:
    \begin{equation}
    G^2 = \begin{bmatrix} 0.04 & 0.16 \\ 0.36 & 0.64 \end{bmatrix}
    \end{equation}
    \item \textbf{Marginal Means}:
    \begin{align}
    r_{\text{means}} &= \left[ \frac{0.04 + 0.16}{2}, \frac{0.36 + 0.64}{2} \right]^T = [0.10, 0.50]^T \\
    c_{\text{means}} &= \left[ \frac{0.04 + 0.36}{2}, \frac{0.16 + 0.64}{2} \right]^T = [0.20, 0.40]^T
    \end{align}
    \item \textbf{Factor Updates} ($\beta_2 = 0.5$):
    \begin{align}
    v_{\text{row}}^{\text{next}} &= 0.5[0, 0]^T + 0.5[0.10, 0.50]^T = [0.05, 0.25]^T \\
    v_{\text{col}}^{\text{next}} &= 0.5[0, 0]^T + 0.5[0.20, 0.40]^T = [0.10, 0.20]^T
    \end{align}
    \item \textbf{Row Normalizer}:
    \begin{equation}
    S_{\text{row}} = 0.05 + 0.25 + 10^{-8} = 0.30000001
    \end{equation}
    \item \textbf{Reconstructed Second Moments}:
    \begin{equation}
    \hat{V}_{11} = \frac{0.05 \times 0.10}{0.30} = \frac{0.005}{0.30} \approx 0.01666667, \quad \sqrt{\hat{V}_{11}} \approx 0.12909944
    \end{equation}
    \item \textbf{Update on $W_{11}$}:
    \begin{equation}
    W_{11}^{\text{next}} = 1.0 - 0.01 \times \frac{0.2}{0.12909944 + 10^{-8}} = 1.0 - 0.01(1.5491933) \approx 0.984508
    \end{equation}
\end{enumerate}

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Non-2D Tensors}: For 1D tensors (e.g., biases), Adafactor falls back to standard uncompressed second moment tracking.
    \item \textbf{Underflow in Small Marginal Products}: When both row and column factors are extremely small ($\approx 10^{-19}$), their product $v_{\text{row}, i} v_{\text{col}, j}$ can underflow to zero in FP32. Adding $\epsilon$ to both the numerator and denominator prevents irregular scaling.
\end{itemize}
