\begin{abstract}
This document presents the complete theoretical foundation and algorithmic specification of Second-Order Preconditioning (e.g., Shampoo / K-FAC). Standard diagonal optimizers ignore off-diagonal Hessian and Fisher information matrix (FIM) cross-correlations. Second-order preconditioning captures directional curvature by approximating the full covariance matrix through Kronecker products of left ($L = G G^T \in \mathbb{R}^{M \times M}$) and right ($R = G^T G \in \mathbb{R}^{N \times N}$) statistics, computing inverse matrix roots $(L + \epsilon I)^{-1/4} G (R + \epsilon I)^{-1/4}$. We formalize Kronecker curvature factorization, provide the complete algorithmic pseudo-code, derive exact computational complexities, demonstrate a worked numerical example matching the reference implementation, and discuss numerical damping stability.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $W \in \mathbb{R}^{M \times N}$ denote a 2D weight matrix parameter with gradient matrix $G \triangleq \nabla_W \mathcal{L}(W) \in \mathbb{R}^{M \times N}$.

\begin{itemize}
    \item $M, N \in \mathbb{N}_{\ge 1}$: Input and output dimensional ranks of the linear transformation.
    \item $L_t \in \mathbb{R}^{M \times M}$: Accumulated left spatial covariance matrix ($L_t = \sum_{\tau=1}^t G_\tau G_\tau^T$).
    \item $R_t \in \mathbb{R}^{N \times N}$: Accumulated right spatial covariance matrix ($R_t = \sum_{\tau=1}^t G_\tau^T G_\tau$).
    \item $P_t \in \mathbb{R}^{M \times N}$: Preconditioned gradient matrix update.
    \item $\otimes$: Kronecker product operator.
    \item $\epsilon > 0$: Tikhonov damping regularization factor added to diagonals.
    \item $\eta > 0$: Base optimization learning rate.
\end{itemize}

\section{Problem Definition and Core Concepts}

\subsection{Intuition and Motivation}
First-order optimizers treat all coordinates independently or scale step sizes solely based on diagonal variance $\operatorname{diag}(H)^{-1/2}$. However, in neural networks with correlated features, the true curvature is dominated by off-diagonal interactions in the Hessian $H \in \mathbb{R}^{MN \times MN}$. Inverting an $MN \times MN$ matrix is computationally intractable ($\mathcal{O}((MN)^3)$). Shampoo and K-FAC approximate the full curvature matrix $F \approx L \otimes R$ via Kronecker factorization, enabling full 2D curvature preconditioning via decoupled matrix roots on $M \times M$ and $N \times N$ matrices:
\begin{equation}
(L \otimes R)^{-1/2} \operatorname{vec}(G) = \operatorname{vec}\left( L^{-1/4} G R^{-1/4} \right)
\end{equation}

\subsection{Formal Mathematical Formulation}
\begin{definition}[Kronecker Preconditioner Update]
At iteration $t$, given gradient matrix $G_t \in \mathbb{R}^{M \times N}$, historical preconditioners $L_{t-1} \in \mathbb{R}^{M \times M}$ and $R_{t-1} \in \mathbb{R}^{N \times N}$, and damping $\epsilon > 0$:
\begin{align}
L_t &= L_{t-1} + G_t G_t^T \label{eq:shampoo_l} \\
R_t &= R_{t-1} + G_t^T G_t \label{eq:shampoo_r}
\end{align}
Under diagonal approximation / fourth-root scaling:
\begin{equation}
P_{t, ij} = (L_{t, ii} + \epsilon)^{-1/4} \cdot G_{t, ij} \cdot (R_{t, jj} + \epsilon)^{-1/4} \label{eq:shampoo_p}
\end{equation}
\end{definition}

\section{Algorithmic Specification}

The computational pipeline implemented in \texttt{NnAlgoSecondOrderPreconditioning} is shown below.

\begin{algorithm}[H]
\caption{Second-Order Kronecker Preconditioning Step}
\begin{algorithmic}[1]
\Require Gradient matrix $G \in \mathbb{R}^{M \times N}$
\Require Left preconditioner $L \in \mathbb{R}^{M \times M}$, right preconditioner $R \in \mathbb{R}^{N \times N}$
\Require Base step size $\eta > 0$, damping parameter $\epsilon > 0$
\Ensure Updated buffers $L_{\text{next}} \in \mathbb{R}^{M \times M}, R_{\text{next}} \in \mathbb{R}^{N \times N}$, preconditioned gradient $P \in \mathbb{R}^{M \times N}$
\State $M \gets \text{rows}(G), \quad N \gets \text{cols}(G)$
\If{$M = 0$ \textbf{or} $\text{rows}(L) \ne M$ \textbf{or} $\text{cols}(L) \ne M$ \textbf{or} $\text{rows}(R) \ne N$ \textbf{or} $\text{cols}(R) \ne N$}
    \State \textbf{throw} PreconditionFailureException(``Preconditioner dimension mismatch'')
\EndIf
\State $L_{\text{next}} \gets \text{new Matrix of shape } (M, M)$
\For{$i \gets 1 \textbf{ to } M$}
    \For{$j \gets 1 \textbf{ to } M$}
        \State $L_{\text{next}}[i][j] \gets L[i][j] + \sum_{k=1}^N G[i][k] \cdot G[j][k]$
    \EndFor
\EndFor
\State $R_{\text{next}} \gets \text{new Matrix of shape } (N, N)$
\For{$i \gets 1 \textbf{ to } N$}
    \For{$j \gets 1 \textbf{ to } N$}
        \State $R_{\text{next}}[i][j] \gets R[i][j] + \sum_{k=1}^M G[k][i] \cdot G[k][j]$
    \EndFor
\EndFor
\State $P \gets \text{new Matrix of shape } (M, N)$
\For{$i \gets 1 \textbf{ to } M$}
    \State $l_{\text{scale}} \gets (L_{\text{next}}[i][i] + \epsilon)^{-1/4}$
    \For{$j \gets 1 \textbf{ to } N$}
        \State $r_{\text{scale}} \gets (R_{\text{next}}[j][j] + \epsilon)^{-1/4}$
        \State $P[i][j] \gets l_{\text{scale}} \cdot G[i][j] \cdot r_{\text{scale}}$
    \EndFor
\EndFor
\State \Return $\{\text{updated\_left}: L_{\text{next}}, \text{updated\_right}: R_{\text{next}}, \text{preconditioned\_gradient}: P\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Curvature Invariance}

\begin{theorem}[Affine Transformation Invariance under Kronecker Preconditioning]
Let linear layer $y = W x$ undergo invertible linear reparameterizations of inputs $x \leftarrow S_x x$ and outputs $y \leftarrow S_y y$. Full Kronecker preconditioned updates $(L^{-1/4} G R^{-1/4})$ are invariant under affine coordinates transformations, matching natural gradient descent on the manifold.
\end{theorem}

\begin{proof}
Under transformations $S_x, S_y$, transformed gradients transform as $G' = S_y G S_x^T$. Covariances transform as $L' = S_y L S_y^T$ and $R' = S_x R S_x^T$. The inverse fourth roots transform as $(L')^{-1/4} = S_y^{-1} L^{-1/4}$ and $(R')^{-1/4} = R^{-1/4} S_x^{-T}$. Multiplying yields $(L')^{-1/4} G' (R')^{-1/4} = S_y^{-1} L^{-1/4} (S_y G S_x^T) R^{-1/4} S_x^{-T} = L^{-1/4} G R^{-1/4}$, establishing exact parameterization invariance.
\end{proof}

\subsection{Limitations}
\begin{itemize}
    \item \textbf{Matrix Inversion Overhead}: When full dense matrix roots are computed instead of diagonal approximations, SVD or Schur decompositions scale as $\mathcal{O}(M^3 + N^3)$, typically computed asynchronously every $k \approx 50$ steps.
\end{itemize}

\section{Complexity Analysis}

\subsection{Time Complexity}
\begin{itemize}
    \item \textbf{Left Covariance Accumulation} ($G G^T$): Matrix multiplication of $(M \times N)$ by $(N \times M)$ requires $\mathcal{O}(M^2 N)$ FLOPs.
    \item \textbf{Right Covariance Accumulation} ($G^T G$): Matrix multiplication of $(N \times M)$ by $(M \times N)$ requires $\mathcal{O}(M N^2)$ FLOPs.
    \item \textbf{Diagonal Preconditioning}: Applying fourth-roots and element-wise products takes $\mathcal{O}(MN)$ FLOPs.
    \item \textbf{Total Time}: $\Theta(MN(M + N))$ FLOPs per step.
\end{itemize}

\subsection{Space Complexity}
\begin{itemize}
    \item \textbf{Covariance Memory}: Stores $L \in \mathbb{R}^{M \times M}$ and $R \in \mathbb{R}^{N \times N}$, requiring $M^2 + N^2$ floating point words ($\Theta(M^2 + N^2)$ auxiliary space vs. $(MN)^2$ for full Fisher).
\end{itemize}

\section{Worked Numerical Example}

Consider $M=2, N=2$, $G = \begin{bmatrix} 1.0 & 2.0 \\ 3.0 & 4.0 \end{bmatrix}$, $L_0 = \begin{bmatrix} 0 & 0 \\ 0 & 0 \end{bmatrix}$, $R_0 = \begin{bmatrix} 0 & 0 \\ 0 & 0 \end{bmatrix}$, $\epsilon = 1.0$.

\begin{enumerate}
    \item \textbf{Left Covariance $L = G G^T$}:
    \begin{equation}
    L = \begin{bmatrix} 1(1) + 2(2) & 1(3) + 2(4) \\ 3(1) + 4(2) & 3(3) + 4(4) \end{bmatrix} = \begin{bmatrix} 5 & 11 \\ 11 & 25 \end{bmatrix}
    \end{equation}
    \item \textbf{Right Covariance $R = G^T G$}:
    \begin{equation}
    R = \begin{bmatrix} 1(1) + 3(3) & 1(2) + 3(4) \\ 2(1) + 4(3) & 2(2) + 4(4) \end{bmatrix} = \begin{bmatrix} 10 & 14 \\ 14 & 20 \end{bmatrix}
    \end{equation}
    \item \textbf{Diagonal Scaling Factors} ($\epsilon = 1.0$):
    \begin{align}
    l_{\text{scale}, 1} &= (L_{11} + 1)^{-1/4} = (5 + 1)^{-1/4} = 6^{-0.25} \approx 0.638943 \\
    l_{\text{scale}, 2} &= (L_{22} + 1)^{-1/4} = (25 + 1)^{-1/4} = 26^{-0.25} \approx 0.442864 \\
    r_{\text{scale}, 1} &= (R_{11} + 1)^{-1/4} = (10 + 1)^{-1/4} = 11^{-0.25} \approx 0.549280 \\
    r_{\text{scale}, 2} &= (R_{22} + 1)^{-1/4} = (20 + 1)^{-1/4} = 21^{-0.25} \approx 0.466986
    \end{align}
    \item \textbf{Preconditioned Gradient $P_{ij} = l_{\text{scale}, i} \cdot G_{ij} \cdot r_{\text{scale}, j}$}:
    \begin{align}
    P_{11} &= 0.638943 \times 1.0 \times 0.549280 \approx 0.35096 \\
    P_{12} &= 0.638943 \times 2.0 \times 0.466986 \approx 0.59676 \\
    P_{21} &= 0.442864 \times 3.0 \times 0.549280 \approx 0.72975 \\
    P_{22} &= 0.442864 \times 4.0 \times 0.466986 \approx 0.82725
    \end{align}
\end{enumerate}

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Diagonal Damping $\epsilon$}: Tikhonov regularization $\epsilon I$ prevents singularity and indefinite negative eigenvalues caused by finite precision roundoff in outer product sums.
    \item \textbf{Coupled Block-Diagonal Structure}: For multi-head attention projection layers, Kronecker factors are computed per head to keep tensor dimensions $M, N$ small.
\end{itemize}
