\begin{abstract}
Contiguous memory pre-allocation for KV caches produces up to $80\%$ internal and external memory fragmentation in serving systems. PagedAttention adapts virtual memory principles from operating systems by partitioning sequence KV caches into fixed-size physical blocks and indexing them through logical page tables. This specification formalizes non-contiguous block indexing, physical memory utilization, complexity derivations, and worked numerical verification.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $B_{\text{size}} \in \mathbb{N}_{\ge 1}$ denote block size (number of tokens per physical block, e.g. 16).

\begin{itemize}
    \item $\mathcal{T} = [b_0, b_1, \dots, b_{M-1}] \in \mathbb{N}^M$: Block table mapping logical block index $m$ to physical memory frame index $b_m$.
    \item $\mathcal{P} \in \mathbb{R}^{K \times B_{\text{size}} \times d}$: Physical block memory pool storing $K$ pre-allocated frames.
    \item $\mathbf{q} \in \mathbb{R}^d$: Incoming decoding query vector.
\end{itemize}

\section{Problem Definition and Algorithmic Execution}

Logical token index $t$ maps to physical block $b = \mathcal{T}[\lfloor t / B_{\text{size}} \rfloor]$ and intra-block offset $o = t \pmod{B_{\text{size}}}$.

\begin{algorithm}[H]
\caption{PagedAttention Block Gathering and Forward Attention}
\label{alg:paged_attn}
\begin{algorithmic}[1]
\Require Block table $\mathcal{T}$, physical blocks $\mathcal{P}$, query $\mathbf{q}$, block size $B_{\text{size}}$
\Ensure Attention output $\mathbf{o}$, gathered sequence length $L$
\State $\mathbf{K}_{\text{virtual}} \gets \text{empty list}$
\For{$b \in \mathcal{T}$}
    \State Append $\mathcal{P}[b]$ to $\mathbf{K}_{\text{virtual}}$
\EndFor
\State $L \gets \text{rows}(\mathbf{K}_{\text{virtual}})$
\State $\mathbf{s} \gets \mathbf{K}_{\text{virtual}} \mathbf{q} / \sqrt{d}$
\State $\mathbf{a} \gets \operatorname{Softmax}(\mathbf{s})$
\State $\mathbf{o} \gets \sum_{i=1}^L a_i \mathbf{K}_{\text{virtual}}[i]$
\State \Return $\{\text{output}: \mathbf{o}, \text{gathered\_length}: L\}$
\end{algorithmic}
\end{algorithm}

\section{Complexity Analysis}
\begin{itemize}
    \item \textbf{Memory Waste Bound:} At most $B_{\text{size}} - 1$ slots wasted per request (near zero fragmentation).
    \item \textbf{Time Complexity:} $\Theta(L \cdot d)$ gathering and attention FLOPs.
\end{itemize}
