\begin{abstract}
Structured State Space sequence models (S4) map continuous 1D signals $u(t) \mapsto y(t)$ through linear ordinary differential equations $h'(t) = \mathbf{A}h(t) + \mathbf{B}u(t), y(t) = \mathbf{C}h(t) + \mathbf{D}u(t)$ governed by High-order Polynomial Projection Operators (HiPPO). Discretization via bilinear (Tustin) transforms admits dual computational modes: linear recurrent step updates for $\mathcal{O}(1)$ inference and non-circular FFT convolutions for parallel $\mathcal{O}(L \log L)$ training. This document specifies the discretization operators, stability guarantees, and numerical verification.
\end{abstract}

\section{Continuous Formulation and Discretization}

Given step size $\Delta > 0$, the continuous matrices $(\mathbf{A}, \mathbf{B})$ discretize via bilinear transform to:
\begin{equation}
\bar{\mathbf{A}} = \left(\mathbf{I} - \frac{\Delta}{2}\mathbf{A}\right)^{-1}\left(\mathbf{I} + \frac{\Delta}{2}\mathbf{A}\right), \quad \bar{\mathbf{B}} = \left(\mathbf{I} - \frac{\Delta}{2}\mathbf{A}\right)^{-1}\Delta \mathbf{B}
\end{equation}
Discrete state updates: $h_k = \bar{\mathbf{A}} h_{k-1} + \bar{\mathbf{B}} u_k$, with output $y_k = \mathbf{C} h_k + \mathbf{D} u_k$.
