\begin{abstract}
Linear Time-Invariant (LTI) State Space Models cannot perform content-based memory filtering because their transition operators are fixed across tokens. Mamba introduces a Selective State Space mechanism where step size $\Delta_t$, input projection $\mathbf{B}_t$, and output projection $\mathbf{C}_t$ are continuous parameterized functions of the current input $x_t$. This specification formalizes the selective discretization, proving linear time scaling and numerical parity.
\end{abstract}

\section{Selective Discretization}

Given input $x_t$, selection parameters are dynamically computed:
\begin{align}
\mathbf{B}_t &= \text{Linear}_B(x_t), \quad \mathbf{C}_t = \text{Linear}_C(x_t), \quad \Delta_t = \operatorname{softplus}(\text{Linear}_\Delta(x_t)) \\
\bar{\mathbf{A}}_t &= \exp(\Delta_t \mathbf{A}), \quad \bar{\mathbf{B}}_t = (\Delta_t \mathbf{A})^{-1}(\exp(\Delta_t \mathbf{A}) - \mathbf{I}) \cdot \Delta_t \mathbf{B}_t \approx \Delta_t \mathbf{B}_t
\end{align}
State update: $\mathbf{h}_t = \bar{\mathbf{A}}_t \mathbf{h}_{t-1} + \bar{\mathbf{B}}_t x_t, \quad y_t = \mathbf{C}_t \mathbf{h}_t$.
