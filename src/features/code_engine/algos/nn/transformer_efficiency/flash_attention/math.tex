\begin{abstract}
Standard self-attention materializes $N \times N$ attention matrices to High Bandwidth Memory (HBM), incurring $\mathcal{O}(N^2)$ memory reads and writes. FlashAttention tiles query, key, and value matrices into SRAM-sized blocks and applies online softmax rescaling, evaluating exact attention in a single fused pass with $\mathcal{O}(N)$ HBM accesses and zero intermediate attention matrix storage. This document specifies the online softmax recurrence, IO complexity, and verified step-by-step numerical execution.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $\mathbf{Q}, \mathbf{K}, \mathbf{V} \in \mathbb{R}^{N \times d}$. Let $B_r, B_c \in \mathbb{N}_{\ge 1}$ denote block sizes along query and key dimensions.

\begin{itemize}
    \item $m_i \in \mathbb{R}$: Running maximum logit for row $i$.
    \item $l_i \in \mathbb{R}$: Running unnormalized softmax partition sum for row $i$.
    \item $\mathbf{O}_i \in \mathbb{R}^d$: Running unnormalized accumulated output vector for row $i$.
\end{itemize}

\section{Online Softmax Recurrence Relations}

When processing a new block of key-values with local scores $\mathbf{S}^{(j)}$, let local maximum be $\tilde{m} = \max_k S_k^{(j)}$. The updated global statistics satisfy:
\begin{align}
m_{\text{new}} &= \max(m_{\text{old}}, \tilde{m}) \\
l_{\text{new}} &= l_{\text{old}} \cdot e^{m_{\text{old}} - m_{\text{new}}} + \sum_{k} e^{S_k^{(j)} - m_{\text{new}}} \\
\mathbf{O}_{\text{new}} &= \mathbf{O}_{\text{old}} \cdot e^{m_{\text{old}} - m_{\text{new}}} + \sum_{k} e^{S_k^{(j)} - m_{\text{new}}} \mathbf{v}_k
\end{align}
Final normalization yields $\mathbf{Y}_i = \mathbf{O}_i / l_i$.

\section{Algorithmic Specification}

\begin{algorithm}[H]
\caption{FlashAttention Blocked Online Softmax Forward Pass}
\label{alg:flash_attn}
\begin{algorithmic}[1]
\Require Queries $\mathbf{Q}$, keys $\mathbf{K}$, values $\mathbf{V} \in \mathbb{R}^{N \times d}$, block size $B$
\Ensure Output matrix $\mathbf{Y} \in \mathbb{R}^{N \times d}$
\State Partition $\mathbf{Q}, \mathbf{K}, \mathbf{V}$ into blocks of size $B$
\State Initialize $m_i \gets -\infty, \quad l_i \gets 0.0, \quad \mathbf{O}_i \gets \mathbf{0} \quad \forall i \in \{1, \dots, N\}$
\For{each block $K_j, V_j$}
    \For{each block $Q_i$}
        \State Compute local scores $S_{i, j} = Q_i K_j^\top / \sqrt{d}$
        \State Compute updated $m_{\text{new}}, l_{\text{new}}, \mathbf{O}_{\text{new}}$ using online softmax formulas
    \EndFor
\EndFor
\State $\mathbf{Y}_i \gets \mathbf{O}_i / l_i$ for all $i$
\State \Return $\{\text{output}: \mathbf{Y}\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees: IO Complexity Bound}

\begin{theorem}[IO Complexity Reduction (Dao et al., 2022)]
Let fast SRAM memory capacity be $M$ with $d \le M \le Nd$. Standard attention requires $\Theta(N d + N^2)$ HBM accesses. FlashAttention executes with $\Theta(N^2 d^2 / M)$ HBM accesses, yielding up to a $5\times$ wall-clock speedup.
\end{theorem}
