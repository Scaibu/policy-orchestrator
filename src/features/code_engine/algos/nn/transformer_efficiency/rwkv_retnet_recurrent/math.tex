\begin{abstract}
Retentive Network (RetNet) and Receptance Weighted Key Value (RWKV) combine the parallel training speed of Transformers with the $\mathcal{O}(1)$ inference latency and constant memory of recurrent networks. Retentive attention weights relative distances with explicit exponential decay $\gamma^{i - j}$ ($\gamma \in (0, 1)$). This document derives the dual parallel and recurrent forms, proves mathematical equivalence, and details numerical verification.
\end{abstract}

\section{Recurrent and Parallel Dual Formulations}

\subsection{Recurrent Representation (Inference Mode)}
At step $n$, state $\mathbf{S}_n \in \mathbb{R}^{d \times d}$ updates recurrently:
\begin{align}
\mathbf{S}_n &= \gamma \mathbf{S}_{n-1} + \mathbf{k}_n \mathbf{v}_n^\top \\
\mathbf{o}_n &= \mathbf{q}_n^\top \mathbf{S}_n
\end{align}

\subsection{Parallel Representation (Training Mode)}
For sequence length $N$:
\begin{equation}
\mathbf{O} = (\mathbf{Q}\mathbf{K}^\top \odot \mathbf{D}) \mathbf{V}, \quad D_{i, j} = \begin{cases} \gamma^{i - j}, & \text{if } i \ge j \\ 0, & \text{if } i < j \end{cases}
\end{equation}
