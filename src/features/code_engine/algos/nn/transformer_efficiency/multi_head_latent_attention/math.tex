\begin{abstract}
Multi-Head Latent Attention (MLA, introduced in DeepSeek architectures) projects key and value representations into a shared low-rank latent vector $\mathbf{c}_{kv} \in \mathbb{R}^{d_c}$ where $d_c \ll H \cdot d_k$. During inference, only the low-rank latent vectors are cached, with multi-head keys and values reconstructed dynamically. This document specifies the low-rank factorized projections, memory savings, complexity derivations, and numerical verification.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $\mathbf{H} \in \mathbb{R}^{N \times d_{\text{model}}}$ denote input hidden states, $d_c \in \mathbb{N}_{\ge 1}$ denote the compressed latent dimension, $H \in \mathbb{N}_{\ge 1}$ denote attention heads count, and $d_k \in \mathbb{N}_{\ge 1}$ denote per-head dimension.

\begin{itemize}
    \item $\mathbf{W}_{DKV} \in \mathbb{R}^{d_{\text{model}} \times d_c}$: Down-projection compression matrix.
    \item $\mathbf{c}_{kv} = \mathbf{H} \mathbf{W}_{DKV} \in \mathbb{R}^{N \times d_c}$: Compressed latent KV representation.
    \item $\mathbf{W}_{UK} \in \mathbb{R}^{d_c \times (H d_k)}, \mathbf{W}_{UV} \in \mathbb{R}^{d_c \times (H d_k)}$: Up-projection decompression matrices.
\end{itemize}

\section{Problem Definition and Algorithmic Execution}

\begin{algorithm}[H]
\caption{Multi-Head Latent Attention Forward Execution}
\label{alg:mla}
\begin{algorithmic}[1]
\Require Hidden states $\mathbf{H} \in \mathbb{R}^{N \times d_{\text{model}}}$, latent dim $d_c$, heads $H$, head dim $d_k$
\Ensure Attended output $\mathbf{O}$, compressed cache $\mathbf{c}_{kv}$
\State $\mathbf{c}_{kv} \gets \text{ProjectDown}(\mathbf{H}, d_c)$
\State $\mathbf{K} \gets \text{ProjectUp}(\mathbf{c}_{kv}, H \cdot d_k)$
\State $\mathbf{V} \gets \text{ProjectUp}(\mathbf{c}_{kv}, H \cdot d_k)$
\State $\mathbf{Q} \gets \text{ProjectQuery}(\mathbf{H}, H \cdot d_k)$
\State $\mathbf{O} \gets \operatorname{MultiHeadAttention}(\mathbf{Q}, \mathbf{K}, \mathbf{V}, H)$
\State \Return $\{\text{output}: \mathbf{O}, \text{compressed\_kv}: \mathbf{c}_{kv}, \text{compression\_ratio}: \frac{2 H d_k}{d_c}\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees: Low-Rank Matrix Equivalence}

\begin{theorem}[Low-Rank Factorization Rank Bound]
The effective key-value projection matrices $\mathbf{W}_K = \mathbf{W}_{DKV} \mathbf{W}_{UK}$ have rank strictly upper-bounded by $\operatorname{rank}(\mathbf{W}_K) \le \min(d_c, d_{\text{model}}, H d_k) = d_c$.
\end{theorem}
