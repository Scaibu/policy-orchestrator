\begin{abstract}
The canonical Transformer block stacks Multi-Head Self-Attention (MHA) and Position-wise Feedforward Networks (FFN) coupled with residual skip connections and normalization layers. Pre-Layer-Normalization (Pre-LN) places normalization operators within the residual branches, preserving an unimpeded identity gradient highway throughout the network depth. This document specifies the complete forward algebraic mapping, proves identity gradient preservation, derives computational complexities, and details numerical verification.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $\mathbf{X} \in \mathbb{R}^{N \times d}$ denote the input hidden state tensor of sequence length $N$ and hidden dimension $d$.

\begin{itemize}
    \item $h \in \mathbb{N}_{\ge 1}$: Number of attention heads, with head dimension $d_k = d / h$.
    \item $d_{\text{ff}} \in \mathbb{N}_{\ge 1}$: Intermediate feedforward expansion dimension (typically $4d$).
    \item $\mathbf{W}_1 \in \mathbb{R}^{d \times d_{\text{ff}}}, \mathbf{W}_2 \in \mathbb{R}^{d_{\text{ff}} \times d}$: Feedforward linear projection weights.
    \item $\operatorname{LN}(\mathbf{z}) = \frac{\mathbf{z} - \mu}{\sqrt{\sigma^2 + \epsilon}} \odot \boldsymbol{\gamma} + \boldsymbol{\beta}$: Layer normalization operator.
    \item $\mathbf{X}^{(1)} = \mathbf{X} + \operatorname{MHA}(\operatorname{LN}(\mathbf{X}))$: First attentional residual state.
    \item $\mathbf{Y} = \mathbf{X}^{(1)} + \operatorname{FFN}(\operatorname{LN}(\mathbf{X}^{(1)}))$: Block output representation.
\end{itemize}

\section{Problem Definition and Algorithmic Execution}

\begin{algorithm}[H]
\caption{Pre-LN Transformer Block Forward Pass}
\label{alg:transformer_block}
\begin{algorithmic}[1]
\Require Input representations $\mathbf{X} \in \mathbb{R}^{N \times d}$, heads $h$, MLP dimension $d_{\text{ff}}$, mask $\mathbf{M}$
\Ensure Output representations $\mathbf{Y} \in \mathbb{R}^{N \times d}$
\State $\tilde{\mathbf{X}} \gets \operatorname{LayerNorm}(\mathbf{X})$
\State $\mathbf{A} \gets \operatorname{MultiHeadAttention}(\tilde{\mathbf{X}}, \tilde{\mathbf{X}}, \tilde{\mathbf{X}}, h, \mathbf{M})$
\State $\mathbf{X}^{(1)} \gets \mathbf{X} + \mathbf{A}$ \Comment{First residual highway}
\State $\tilde{\mathbf{X}}^{(1)} \gets \operatorname{LayerNorm}(\mathbf{X}^{(1)})$
\State $\mathbf{F} \gets \operatorname{ReLU}(\tilde{\mathbf{X}}^{(1)} \mathbf{W}_1) \mathbf{W}_2$
\State $\mathbf{Y} \gets \mathbf{X}^{(1)} + \mathbf{F}$ \Comment{Second residual highway}
\State \Return $\{\text{output}: \mathbf{Y}\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees: Identity Gradient Highway}

\begin{theorem}[Identity Gradient Flow in Pre-LN]
In a network of $L$ stacked Pre-LN Transformer blocks, the gradient of the loss $\mathcal{L}$ with respect to layer $l$ representation contains an unattenuated direct identity term:
\begin{equation}
\frac{\partial \mathcal{L}}{\partial \mathbf{X}_l} = \frac{\partial \mathcal{L}}{\partial \mathbf{X}_L} \left( \mathbf{I} + \sum_{j=l}^{L-1} \frac{\partial \mathcal{F}_j}{\partial \mathbf{X}_l} \right)
\end{equation}
preventing vanishing and exploding gradients without requiring warmup heuristics.
\end{theorem}

\section{Complexity Analysis}
\begin{itemize}
    \item \textbf{Time Complexity:} $\mathcal{O}(N^2 d + N d \cdot d_{\text{ff}})$ FLOPs.
    \item \textbf{Space Complexity:} $\mathcal{O}(N^2 + N d_{\text{ff}})$ auxiliary activation memory.
\end{itemize}

\section{Worked Numerical Example}
Let $N = 1, d = 2, d_{\text{ff}} = 2, h = 1$, with simplified identity projections:
Input $\mathbf{x} = [1.0, 3.0]$. Mean $\mu = 2.0$, variance $\sigma^2 = 1.0$.
Normalized $\tilde{\mathbf{x}} = [-1.0, 1.0]$.
Self-attention maps $\tilde{\mathbf{x}}$ to itself: residual $\mathbf{x}^{(1)} = [1.0 - 1.0, 3.0 + 1.0] = [0.0, 4.0]$.
Subsequent MLP evaluation adds forward transformation, preserving shape $(1, 2)$.
