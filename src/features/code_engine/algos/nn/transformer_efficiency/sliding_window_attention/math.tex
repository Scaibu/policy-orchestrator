\begin{abstract}
Full self-attention scales quadratically $\mathcal{O}(N^2)$ with sequence length $N$. Sliding Window Attention restricts attention connectivity to a local temporal band of width $W$, yielding linear $\mathcal{O}(N \cdot W)$ computational and memory complexity while achieving an effective receptive field of $L \cdot W$ across $L$ stacked layers. This specification details the banded masking formulation, complexity derivations, and worked numerical example.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $N \in \mathbb{N}_{\ge 1}$ denote sequence length, and let $W \in \mathbb{N}_{\ge 1}$ denote window radius.

\begin{itemize}
    \item $\mathbf{M}_{i, j} \in \{0, -\infty\}$: Banded attention mask:
    \begin{equation}
    M_{i, j} = \begin{cases} 0, & \text{if } |i - j| \le W \\ -\infty, & \text{otherwise} \end{cases}
    \end{equation}
\end{itemize}

\section{Complexity Analysis}
\begin{itemize}
    \item \textbf{Time Complexity:} $\Theta(N \cdot W \cdot d)$ scalar operations.
    \item \textbf{Space Complexity:} $\Theta(N \cdot W + N \cdot d)$ memory footprint.
\end{itemize}
