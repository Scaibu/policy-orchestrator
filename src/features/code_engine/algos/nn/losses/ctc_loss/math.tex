\begin{abstract}
Sequence-to-sequence alignment without frame-level segmentation is addressed via Connectionist Temporal Classification (CTC). By introducing a blank symbol $\epsilon$ and a many-to-one collapsible mapping $\mathcal{B}$, CTC marginalizes over all valid frame-level alignments in polynomial time $\mathcal{O}(T \cdot L)$ using the forward-backward lattice algorithm. We formalize the collapse operator, state transition dynamics, log-space sum-product stability, step-by-step complexity derivation, and complete pseudocode matching the reference implementation.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\Sigma$ denote the alphabet of output labels and $\Sigma' = \Sigma \cup \{\epsilon\}$ the augmented vocabulary containing the CTC blank symbol $\epsilon$. Let $T \in \mathbb{N}$ denote the length of the input frame sequence and $L \in \mathbb{N}$ the length of the target label sequence ($L \le T$).

\subsection{Lattice State Sequence Construction}
\paragraph{Intuitive Meaning:} Because speech and handwriting frames can repeat characters or emit silence, CTC inserts blanks between every pair of characters and at the boundaries, expanding target sequence $\mathbf{l}$ of length $L$ into modified sequence $\mathbf{l}'$ of length $S = 2L + 1$.

\paragraph{Formal Mathematical Definition:}
Given target sequence $\mathbf{l} = (l_1, l_2, \dots, l_L) \in \Sigma^L$:
\begin{equation}
\label{eq:l_prime_def}
\mathbf{l}' = (\epsilon, l_1, \epsilon, l_2, \dots, \epsilon, l_L, \epsilon) \in (\Sigma')^{2L+1}.
\end{equation}
The collapse operator $\mathcal{B}: (\Sigma')^T \to \Sigma^{\le T}$ sequentially removes adjacent duplicate tokens and then deletes all blank symbols $\epsilon$.

\paragraph{Worked Numerical Example:}
Let target $\mathbf{l} = (\text{c}, \text{a}, \text{t})$ with blank $\epsilon = 0$.
The augmented sequence is $\mathbf{l}' = (0, \text{c}, 0, \text{a}, 0, \text{t}, 0)$ with length $S = 2(3) + 1 = 7$.
A path $\pi = (\text{c}, \text{c}, 0, \text{a}, \text{t}, \text{t})$ collapses under $\mathcal{B}$ as:
$\text{c} \text{c} 0 \text{a} \text{t} \text{t} \to \text{c} 0 \text{a} \text{t} \to \text{c} \text{a} \text{t} = \mathbf{l}$.

\subsection{Summary of Symbols}
\begin{itemize}
    \item $T \in \mathbb{N}$: Temporal frame length.
    \item $L \in \mathbb{N}$: Target sequence length ($L \le T$).
    \item $S = 2L + 1$: Augmented lattice state count.
    \item $C \in \mathbb{N}$: Vocabulary size $|\Sigma'|$.
    \item $\mathbf{y} \in \mathbb{R}^{T \times C}$: Frame emission log-probability matrix.
    \item $\alpha_t(s) \in \mathbb{R}$: Forward log-probability at time $t$ for lattice state $s$.
\end{itemize}

\section{Problem Definition and Core Concepts}
\label{sec:problem_definition}

\subsection{Forward Lattice Recurrence}
The forward variable $\alpha_t(s)$ computes the log-probability of all sub-paths $\pi_{1:t}$ whose collapsed prefix maps to $\mathbf{l}'_{1:s}$:
\begin{equation}
\label{eq:ctc_alpha_recurrence}
\alpha_t(s) = \log\left[ \exp(\alpha_{t-1}(s)) + \exp(\alpha_{t-1}(s-1)) + \mathbb{I}_{\{l'_s \neq \epsilon \land l'_s \neq l'_{s-2}\}} \exp(\alpha_{t-1}(s-2)) \right] + \log y_{l'_s}^t.
\end{equation}

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:ctc_loss} specifies the complete dynamic programming forward pass matching \texttt{impl.py}.

\begin{algorithm}[H]
\caption{Connectionist Temporal Classification (CTC) Forward Lattice Loss}
\label{alg:ctc_loss}
\begin{algorithmic}[1]
\Require Log emissions $\mathbf{Y} \in \mathbb{R}^{T \times C}$, target sequence $\mathbf{l} \in \mathbb{Z}^L$, blank index $b \in [0, C-1]$.
\Ensure Scalar loss $\mathcal{L}$, forward lattice table $\boldsymbol{\alpha} \in \mathbb{R}^{T \times (2L+1)}$.
\State Construct $\mathbf{l}' \gets [b, l_1, b, l_2, \dots, b, l_L, b]$ of length $S = 2L + 1$
\State Initialize lattice table $\boldsymbol{\alpha} \gets [-\infty]^{T \times S}$
\State $\alpha_0(0) \gets Y_{0, b}$
\If{$S > 1$}
    \State $\alpha_0(1) \gets Y_{0, l_1}$
\EndIf
\For{$t = 1$ to $T-1$}
    \For{$s = 0$ to $S-1$}
        \State $term_1 \gets \alpha_{t-1}(s)$
        \State $term_2 \gets \alpha_{t-1}(s-1)$ if $s > 0$ else $-\infty$
        \If{$s \ge 2$ and $l'_s \neq b$ and $l'_s \neq l'_{s-2}$}
            \State $term_3 \gets \alpha_{t-1}(s-2)$
        \Else
            \State $term_3 \gets -\infty$
        \EndIf
        \State $sum\_prev \gets \mathrm{LogSumExp}(term_1, term_2, term_3)$
        \State $\alpha_t(s) \gets sum\_prev + Y_{t, l'_s}$
    \EndFor
\EndFor
\State $total\_log\_prob \gets \mathrm{LogSumExp}(\alpha_{T-1}(S-1), \alpha_{T-1}(S-2))$
\State $\mathcal{L} \gets -total\_log\_prob$
\State \Return $\mathcal{L}, \quad \boldsymbol{\alpha}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Polynomial Exactness}
\label{sec:guarantees}

\begin{theorem}[Exact Marginalization via Lattice Dynamic Programming]
\label{thm:ctc_polynomial_exactness}
\textbf{Assumptions:} Let frame emission probabilities satisfy conditional independence given input sequence $\mathbf{x}$, and input length $T \ge L$.
\textbf{Guarantee:} The forward algorithm evaluates the exact marginal likelihood $P(\mathbf{l} \mid \mathbf{x}) = \sum_{\pi \in \mathcal{B}^{-1}(\mathbf{l})} P(\pi \mid \mathbf{x})$ in $\mathcal{O}(T \cdot L)$ time, resolving an exponential path space of size $(2L+1)^T$.
\textbf{Proof:}
The set of valid alignments $\mathcal{B}^{-1}(\mathbf{l})$ is isomorphic to all directed paths traversing lattice $\mathbf{l}'$ from states $\{0, 1\}$ at $t=0$ to states $\{S-2, S-1\}$ at $t=T-1$. Because state transitions partition the path space across mutually exclusive, collectively exhaustive sub-alignments at time $t$, dynamic programming summation computes the exact measure without overcounting.
\textbf{Limitations:} Conditional frame independence prevents CTC from conditioning label transitions on previously emitted non-blank tokens (resolved by RNN-T or Autoregressive Attention).
\end{theorem}

\section{Complexity Analysis}
\label{sec:complexity}

\begin{itemize}
    \item \textbf{Time Complexity:} $\mathcal{O}(T \cdot L)$ state transition evaluations, each requiring at most 3 lookups and 1 Log-Sum-Exp.
    \item \textbf{Space Complexity:} $\mathcal{O}(T \cdot L)$ for the forward dynamic programming lattice table.
\end{itemize}

\section{Worked Numerical Example}
\label{sec:worked_example}

Let $T = 2, L = 1$, target $\mathbf{l} = [1]$, blank $b = 0$. $\mathbf{l}' = [0, 1, 0]$ ($S = 3$).
Log emission matrix $\mathbf{Y}$ of shape $(2, 2)$:
$t = 0: Y_{0, 0} = -0.5, Y_{0, 1} = -0.9$.
$t = 1: Y_{1, 0} = -0.2, Y_{1, 1} = -1.6$.
\begin{enumerate}
    \item $t = 0$: $\alpha_0(0) = -0.5$, $\alpha_0(1) = -0.9$, $\alpha_0(2) = -\infty$.
    \item $t = 1, s = 1$: transitions from $s=1$ (self-loop $\alpha_0(1)$) and $s=0$ (transition $\alpha_0(0)$).
    $sum\_prev = \log(e^{-0.9} + e^{-0.5}) = \log(0.40657 + 0.60653) = \log(1.01310) = 0.01301$.
    $\alpha_1(1) = 0.01301 + (-1.6) = -1.58699$.
    \item Total loss: $\mathcal{L} = -\mathrm{LSE}(\alpha_1(2), \alpha_1(1))$.
\end{enumerate}
