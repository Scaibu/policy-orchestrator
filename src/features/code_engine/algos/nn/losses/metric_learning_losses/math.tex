\begin{abstract}
Metric learning transforms representations such that semantically similar samples cluster together while dissimilar samples are separated by a margin. We analyze Euclidean Triplet Loss alongside hyperspherical angular margin classifiers (ArcFace and CosFace). We provide the formal geometric proofs of decision boundaries, derive operation-by-operation complexity, and present full pseudocode matching the reference implementation.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\mathbb{R}^D$ denote the $D$-dimensional feature embedding space and $\mathbb{S}^{D-1} = \{\mathbf{u} \in \mathbb{R}^D \mid \|\mathbf{u}\|_2 = 1\}$ the unit hypersphere.

\subsection{Coordinate and Angular Distance Mechanics}
\paragraph{Intuitive Meaning:} On the unit hypersphere, Euclidean distance squared is linearly related to cosine similarity: $\|\mathbf{u} - \mathbf{v}\|_2^2 = 2 - 2 \cos\theta$. Angular margin losses penalize the cosine angle $\theta_y$ between an embedding and its ground truth class weight vector, forcing intra-class compact clusters.

\paragraph{Formal Mathematical Definition:}
For normalized feature $\hat{\mathbf{x}} = \frac{\mathbf{x}}{\|\mathbf{x}\|_2}$ and class vector $\hat{\mathbf{w}}_j = \frac{\mathbf{w}_j}{\|\mathbf{w}_j\|_2}$, the cosine angle is $\cos\theta_j = \hat{\mathbf{w}}_j^\top \hat{\mathbf{x}}$.
\begin{itemize}
    \item \textbf{Triplet Loss:}
    \begin{equation}
    \mathcal{L}_{\mathrm{triplet}}(\mathbf{a}, \mathbf{p}, \mathbf{n}; m) = \max\left(0, \|\mathbf{a} - \mathbf{p}\|_2^2 - \|\mathbf{a} - \mathbf{n}\|_2^2 + m\right).
    \end{equation}
    \item \textbf{ArcFace (Additive Angular Margin):}
    \begin{equation}
    \mathcal{L}_{\mathrm{ArcFace}} = -\log \frac{\exp(s \cos(\theta_y + m))}{\exp(s \cos(\theta_y + m)) + \sum_{j \neq y} \exp(s \cos\theta_j)}.
    \end{equation}
    \item \textbf{CosFace (Large Margin Cosine Loss):}
    \begin{equation}
    \mathcal{L}_{\mathrm{CosFace}} = -\log \frac{\exp(s (\cos\theta_y - m))}{\exp(s (\cos\theta_y - m)) + \sum_{j \neq y} \exp(s \cos\theta_j)}.
    \end{equation}
\end{itemize}

\subsection{Summary of Symbols}
\begin{itemize}
    \item $B \in \mathbb{N}$: Batch size.
    \item $D \in \mathbb{N}$: Embedding dimension.
    \item $C \in \mathbb{N}$: Total number of classes.
    \item $m \in \mathbb{R}_{\ge 0}$: Metric or angular margin parameter.
    \item $s \in \mathbb{R}_{>0}$: Inverse temperature scale parameter ($s = 30.0$).
    \item $\mathbf{a}, \mathbf{p}, \mathbf{n} \in \mathbb{R}^D$: Anchor, positive, and negative embeddings for Triplet mode.
    \item $\mathbf{X} \in \mathbb{R}^{B \times D}$: Feature embeddings for ArcFace/CosFace.
    \item $\mathbf{W} \in \mathbb{R}^{C \times D}$: Class weight matrix.
\end{itemize}

\section{Problem Definition and Core Concepts}
\label{sec:problem_definition}

\subsection{Hyperspherical Margin Geometry}
In ArcFace, adding margin $m$ directly to the geodesic angle $\theta_y$ on $\mathbb{S}^{D-1}$ enforces an exact angular decision boundary $\theta_1 + m < \theta_2$, producing constant geometric angular margins across all hypersphere coordinates.

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:metric_learning} details the complete computation matching \texttt{impl.py}.

\begin{algorithm}[H]
\caption{Unified Metric Learning Loss Forward Evaluation}
\label{alg:metric_learning}
\begin{algorithmic}[1]
\Require Mode $\mathrm{mode} \in \{\text{triplet}, \text{arcface}, \text{cosface}\}$, Triplet tensors $(\mathbf{A}, \mathbf{P}, \mathbf{N})$ or Angular tensors $(\mathbf{X}, \mathbf{W}, \mathbf{y})$, margin $m \ge 0$, scale $s > 0$.
\Ensure Loss $\mathcal{L}$, auxiliary metrics.
\If{$\mathrm{mode} = \text{triplet}$}
    \State $L_{\mathrm{total}} \gets 0.0, \quad N_{\mathrm{active}} \gets 0$
    \For{$i = 1$ to $B$}
        \State $d_{ap} \gets \sum_{k=1}^D (A_{i, k} - P_{i, k})^2, \quad d_{an} \gets \sum_{k=1}^D (A_{i, k} - N_{i, k})^2$
        \State $val \gets d_{ap} - d_{an} + m$
        \If{$val > 0$}
            \State $L_{\mathrm{total}} \gets L_{\mathrm{total}} + val, \quad N_{\mathrm{active}} \gets N_{\mathrm{active}} + 1$
        \EndIf
    \EndFor
    \State \Return $\frac{L_{\mathrm{total}}}{B}, \quad N_{\mathrm{active}}$
\Else
    \State Normalize each row of $\mathbf{W}$: $\hat{\mathbf{W}}_j \gets \frac{\mathbf{W}_j}{\|\mathbf{W}_j\|_2 + \epsilon}$
    \State $L_{\mathrm{total}} \gets 0.0, \quad \mathbf{S} \gets []$
    \For{$i = 1$ to $B$}
        \State $\hat{\mathbf{x}}_i \gets \frac{\mathbf{X}_i}{\|\mathbf{X}_i\|_2 + \epsilon}$
        \State $\mathbf{cos\_row} \gets [\hat{\mathbf{x}}_i \cdot \hat{\mathbf{W}}_j]_{j=1}^C$
        \State Clamp each $\cos\theta \in [-1, 1]$
        \State $\mathbf{z}_i \gets [s \cdot \cos\theta_j]_{j=1}^C$
        \If{$\mathrm{mode} = \text{arcface}$}
            \State $\theta_y \gets \arccos(\cos\theta_{y_i})$
            \State $z_{i, y_i} \gets s \cdot \cos(\theta_y + m)$
        \Else
            \State $z_{i, y_i} \gets s \cdot (\cos\theta_{y_i} - m)$
        \EndIf
        \State $m_{\mathrm{max}} \gets \max_j z_{i, j}$
        \State $\mathrm{lse} \gets m_{\mathrm{max}} + \log\sum_j \exp(z_{i, j} - m_{\mathrm{max}})$
        \State $L_{\mathrm{total}} \gets L_{\mathrm{total}} + (\mathrm{lse} - z_{i, y_i})$
        \State Append $\mathbf{z}_i$ to $\mathbf{S}$
    \EndFor
    \State \Return $\frac{L_{\mathrm{total}}}{B}, \quad \mathbf{S}$
\EndIf
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Angular Boundaries}
\label{sec:guarantees}

\begin{theorem}[Exact Angular Margin Separation on $\mathbb{S}^{D-1}$]
\label{thm:arcface_boundary}
\textbf{Assumptions:} Let feature $\mathbf{x} \in \mathbb{S}^{D-1}$, normalized class weights $\mathbf{w}_1, \mathbf{w}_2 \in \mathbb{S}^{D-1}$, scale $s > 0$, and margin $m \in (0, \pi)$.
\textbf{Guarantee:} ArcFace requires $\theta_1 + m < \theta_2$ to classify $\mathbf{x}$ as class 1, enforcing a minimal angular buffer of $m$ radians between classes.
\textbf{Proof:}
A sample is assigned to class 1 if and only if $z_1 > z_2 \iff s \cos(\theta_1 + m) > s \cos(\theta_2)$.
Since $s > 0$ and cosine is strictly monotonically decreasing on $[0, \pi]$, $\cos(\theta_1 + m) > \cos(\theta_2) \iff \theta_1 + m < \theta_2 \implies \theta_2 - \theta_1 > m$.
Thus the angular decision boundary is displaced by exactly $m$ radians away from class 1.
\textbf{Limitations:} If $\theta_y + m > \pi$, the cosine function ceases to be monotonic, requiring threshold boundary clipping.
\end{theorem}

\section{Complexity Analysis}
\label{sec:complexity}

\begin{itemize}
    \item \textbf{Time Complexity:} $\mathcal{O}(B \cdot D)$ for Triplet loss; $\mathcal{O}(B \cdot C \cdot D)$ for ArcFace/CosFace matrix multiplications and normalization.
    \item \textbf{Space Complexity:} $\mathcal{O}(B \cdot C)$ for cosine logit matrices.
\end{itemize}

\section{Worked Numerical Example}
\label{sec:worked_example}

Let $\mathbf{a} = [1.0, 0.0], \mathbf{p} = [0.9, 0.1], \mathbf{n} = [0.0, 1.0], m = 0.5$:
\begin{enumerate}
    \item $d_{ap} = (1.0 - 0.9)^2 + (0.0 - 0.1)^2 = 0.01 + 0.01 = 0.02$.
    \item $d_{an} = (1.0 - 0.0)^2 + (0.0 - 1.0)^2 = 1.0 + 1.0 = 2.0$.
    \item Value: $d_{ap} - d_{an} + m = 0.02 - 2.0 + 0.5 = -1.48 \le 0 \implies \mathcal{L} = 0.0$ (inactive triplet).
\end{enumerate}
