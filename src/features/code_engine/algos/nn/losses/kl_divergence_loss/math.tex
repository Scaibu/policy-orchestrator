\begin{abstract}
Kullback-Leibler (KL) divergence measures the relative entropy between a target reference distribution $P$ and an approximating student model $Q$. In knowledge distillation, variational autoencoding, and reinforcement learning policy regularization (RLHF), KL divergence serves as the fundamental constraint functional. This specification presents the log-space numerical stabilization, derives forward versus reverse divergence asymmetry, proves Gibbs' inequality with all intermediate steps, provides full pseudocode matching the reference implementation, and establishes complexity bounds.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\Delta^{C-1} = \{\mathbf{p} \in \mathbb{R}^C \mid p_c \ge 0, \sum_{c=1}^C p_c = 1\}$ denote the probability simplex over $C$ discrete categories.

\subsection{Coordinate and Divergence Formulations}
\paragraph{Intuitive Meaning:} KL divergence measures the expected information loss when distribution $Q$ is used to approximate true distribution $P$. It is non-negative and zero if and only if $P = Q$.

\paragraph{Formal Mathematical Definition:}
For discrete distributions $\mathbf{p}, \mathbf{q} \in \Delta^{C-1}$:
\begin{equation}
\label{eq:kl_def}
D_{\mathrm{KL}}(P \parallel Q) = \sum_{c=1}^C p_c \log\left(\frac{p_c}{q_c}\right) = \sum_{c=1}^C p_c (\log p_c - \log q_c).
\end{equation}

\paragraph{Worked Numerical Example:}
Let $\mathbf{p} = [0.7, 0.3]$ and $\mathbf{q} = [0.6, 0.4]$:
$D_{\mathrm{KL}} = 0.7(\log(0.7) - \log(0.6)) + 0.3(\log(0.3) - \log(0.4)) = 0.7(0.154151) + 0.3(-0.287682) = 0.107905 - 0.086305 = 0.021600$.

\subsection{Summary of Symbols}
\begin{itemize}
    \item $B \in \mathbb{N}$: Batch size.
    \item $C \in \mathbb{N}$: Number of categories.
    \item $\log \mathbf{Q} \in \mathbb{R}^{B \times C}$: Predicted log-probabilities.
    \item $\mathbf{P} \in \Delta^{C-1 \times B}$: Reference target probability distribution.
    \item $\mathbf{G} \in \mathbb{R}^{B \times C}$: Analytic gradient with respect to predicted log-probabilities ($\frac{\partial \mathcal{L}}{\partial \log Q_{i, c}} = -P_{i, c}$).
\end{itemize}

\section{Problem Definition and Core Concepts}
\label{sec:problem_definition}

\subsection{Forward vs. Reverse KL Asymmetry}
\begin{itemize}
    \item \textbf{Forward KL ($D_{\mathrm{KL}}(P \parallel Q)$):} Mode-covering. $Q$ must place non-zero mass everywhere $P > 0$ to avoid infinite loss. Standard in supervised distillation.
    \item \textbf{Reverse KL ($D_{\mathrm{KL}}(Q \parallel P)$):} Mode-seeking. $Q$ concentrates on a single high-probability mode of $P$. Standard in variational inference and policy distillation.
\end{itemize}

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:kl_div} specifies the exact forward pass and gradient calculation matching \texttt{impl.py}.

\begin{algorithm}[H]
\caption{Kullback-Leibler Divergence Loss Forward and Gradient Evaluation}
\label{alg:kl_div}
\begin{algorithmic}[1]
\Require Log predictions $\log\mathbf{Q} \in \mathbb{R}^{B \times C}$, target probabilities $\mathbf{P} \in \mathbb{R}^{B \times C}$, log target flag $\mathrm{log\_target}$, reduction $\mathrm{red} \in \{\text{batchmean}, \text{mean}, \text{sum}, \text{none}\}$.
\Ensure Loss $\mathcal{L}$, gradients $\mathbf{G} \in \mathbb{R}^{B \times C}$.
\State Initialize $\mathbf{l} \gets [], \mathbf{G} \gets []$
\For{$i = 1$ to $B$}
    \State $kl_i \gets 0.0, \quad \mathbf{g}_i \gets []$
    \For{$c = 1$ to $C$}
        \If{$\mathrm{log\_target}$}
            \State $\log p \gets P_{i, c}, \quad p \gets \exp(\log p)$
        \Else
            \State $p \gets P_{i, c}, \quad \log p \gets \log(\max(p, 10^{-15}))$
        \EndIf
        \State $\log q \gets \log Q_{i, c}$
        \If{$p > 0$}
            \State $kl_i \gets kl_i + p \cdot (\log p - \log q)$
        \EndIf
        \State Append $-p$ to $\mathbf{g}_i$
    \EndFor
    \State Append $kl_i$ to $\mathbf{l}$, append $\mathbf{g}_i$ to $\mathbf{G}$
\EndFor
\If{$\mathrm{red} = \text{batchmean}$}
    \State \Return $\frac{1}{B}\sum l_i, \quad \frac{1}{B}\mathbf{G}$
\ElsIf{$\mathrm{red} = \text{mean}$}
    \State \Return $\frac{1}{B \cdot C}\sum l_i, \quad \frac{1}{B \cdot C}\mathbf{G}$
\ElsIf{$\mathrm{red} = \text{sum}$}
    \State \Return $\sum l_i, \quad \mathbf{G}$
\Else
    \State \Return $\mathbf{l}, \quad \mathbf{G}$
\EndIf
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees}
\label{sec:guarantees}

\begin{theorem}[Gibbs' Inequality and Non-Negativity of Divergence]
\label{thm:gibbs_inequality}
\textbf{Assumptions:} Let $\mathbf{p}, \mathbf{q} \in \Delta^{C-1}$ with $\operatorname{supp}(\mathbf{p}) \subseteq \operatorname{supp}(\mathbf{q})$.
\textbf{Guarantee:} $D_{\mathrm{KL}}(P \parallel Q) \ge 0$, with equality $D_{\mathrm{KL}}(P \parallel Q) = 0$ if and only if $\mathbf{p} = \mathbf{q}$.
\textbf{Proof:}
Applying Jensen's inequality to the negative logarithm (which is strictly convex on $\mathbb{R}_{>0}$):
\begin{equation}
D_{\mathrm{KL}}(P \parallel Q) = \sum_{c=1}^C p_c \left(-\log \frac{q_c}{p_c}\right) \ge -\log\left(\sum_{c=1}^C p_c \frac{q_c}{p_c}\right) = -\log\left(\sum_{c=1}^C q_c\right) = -\log(1) = 0.
\end{equation}
Strict convexity ensures equality holds if and only if $\frac{q_c}{p_c} = 1$ for all $c \in \operatorname{supp}(\mathbf{p})$, meaning $\mathbf{p} = \mathbf{q}$.
\textbf{Limitations:} If $q_c = 0$ for any coordinate where $p_c > 0$, $D_{\mathrm{KL}}(P \parallel Q) = +\infty$.
\end{theorem}

\section{Complexity Analysis}
\label{sec:complexity}

\begin{itemize}
    \item \textbf{Time Complexity:} $\mathcal{O}(B \cdot C)$ arithmetic and logarithm evaluations.
    \item \textbf{Space Complexity:} $\mathcal{O}(B \cdot C)$ for gradient tensor storage.
\end{itemize}

\section{Worked Numerical Example}
\label{sec:worked_example}

Let $B = 1, C = 2, \mathbf{p} = [1.0, 0.0], \log\mathbf{q} = [\log(0.8), \log(0.2)] = [-0.223144, -1.609438]$:
\begin{enumerate}
    \item Element 1: $1.0(\log(1.0) - (-0.223144)) = 1.0(0.0 + 0.223144) = 0.223144$.
    \item Element 2: $p_2 = 0 \implies 0.0$.
    \item Loss: $\mathcal{L} = 0.223144$.
    \item Gradients: $\frac{\partial \mathcal{L}}{\partial \log\mathbf{q}} = [-p_1, -p_2] = [-1.0, 0.0]$.
\end{enumerate}
