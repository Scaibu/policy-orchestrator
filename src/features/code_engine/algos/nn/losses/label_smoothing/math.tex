\begin{abstract}
Standard one-hot targets encourage deep neural networks to produce overconfident logit representations, driving weight norms toward infinity and degrading expected calibration error (ECE). Label Smoothing Regularization (LSR) replaces hard Dirac distributions with a mixture of the true label and a uniform prior. This document provides the mathematical foundation, proves logit boundedness under finite loss, provides line-by-line pseudocode matching the reference implementation, and derives computational complexity.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\mathbf{z} \in \mathbb{R}^C$ denote the model logits, $\mathbf{p} = \operatorname{softmax}(\mathbf{z}) \in \Delta^{C-1}$ the predicted probabilities, and $y \in \{0, \dots, C-1\}$ the ground truth class.

\subsection{Soft Target Distribution Construction}
\paragraph{Intuitive Meaning:} Rather than forcing the model to be $100\%$ certain of the training label and $0\%$ certain of everything else, label smoothing reserves a small probability mass $\epsilon$ to be shared equally among all classes.

\paragraph{Formal Mathematical Definition:}
For smoothing factor $\epsilon \in [0, 1)$, the smoothed target probability vector $\mathbf{q} \in \Delta^{C-1}$ is:
\begin{equation}
\label{eq:smooth_target}
q_k = (1 - \epsilon)\delta_{k, y} + \frac{\epsilon}{C} = \begin{cases} 1 - \epsilon + \frac{\epsilon}{C} & \text{if } k = y, \\ \frac{\epsilon}{C} & \text{if } k \neq y. \end{cases}
\end{equation}
The label smoothed cross-entropy loss is the cross-entropy $H(\mathbf{q}, \mathbf{p})$:
\begin{equation}
\label{eq:smooth_loss}
\mathcal{L}_{\mathrm{LS}}(\mathbf{z}, y; \epsilon) = -\sum_{k=1}^C q_k \log p_k = (1 - \epsilon)(-\log p_y) + \frac{\epsilon}{C} \sum_{k=1}^C (-\log p_k).
\end{equation}

\paragraph{Worked Numerical Example:}
Let $C = 3, \epsilon = 0.1, y = 0$.
$q_0 = 1 - 0.1 + \frac{0.1}{3} = 0.9 + 0.033333 = 0.933333$.
$q_1 = q_2 = \frac{0.1}{3} = 0.033333$.

\subsection{Summary of Symbols}
\begin{itemize}
    \item $B \in \mathbb{N}$: Batch size.
    \item $C \in \mathbb{N}$: Total number of classes ($C \ge 2$).
    \item $\epsilon \in [0, 1)$: Label smoothing hyperparameter.
    \item $\mathbf{Z} \in \mathbb{R}^{B \times C}$: Unnormalized prediction logit matrix.
    \item $\mathbf{y} \in \{0, \dots, C-1\}^B$: True class index vector.
    \item $\mathbf{Q} \in \Delta^{C-1 \times B}$: Soft smoothed target matrix.
    \item $\mathbf{P} \in \Delta^{C-1 \times B}$: Normalized softmax prediction matrix.
    \item $\mathbf{G} \in \mathbb{R}^{B \times C}$: Logit gradient matrix ($\mathbf{P} - \mathbf{Q}$).
\end{itemize}

\section{Problem Definition and Core Concepts}
\label{sec:problem_definition}

\subsection{Gradient Mechanics Under Soft Targets}
Differentiating Equation~\eqref{eq:smooth_loss} with respect to logit $z_k$:
\begin{equation}
\frac{\partial \mathcal{L}_{\mathrm{LS}}}{\partial z_k} = p_k - q_k = p_k - (1 - \epsilon)\delta_{k, y} - \frac{\epsilon}{C}.
\end{equation}
The gradient vanishes when predicted probabilities $\mathbf{p}$ exactly match the smoothed targets $\mathbf{q}$.

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:label_smoothing} provides the complete algorithmic procedure matching \texttt{impl.py}.

\begin{algorithm}[H]
\caption{Label Smoothing Regularization Forward and Gradient Evaluation}
\label{alg:label_smoothing}
\begin{algorithmic}[1]
\Require Logits $\mathbf{Z} \in \mathbb{R}^{B \times C}$, targets $\mathbf{y} \in \mathbb{Z}^B$, $\epsilon \in [0, 1)$, reduction $\mathrm{red} \in \{\text{mean}, \text{sum}, \text{none}\}$.
\Ensure Loss $\mathcal{L}$, soft targets $\mathbf{Q}$, probabilities $\mathbf{P}$, gradients $\mathbf{G}$.
\State Initialize $\mathbf{Q} \gets [], \mathbf{P} \gets [], \mathbf{l} \gets [], \mathbf{G} \gets []$
\State $u \gets \frac{\epsilon}{C}, \quad u_{\mathrm{true}} \gets 1.0 - \epsilon + u$
\For{$i = 1$ to $B$}
    \State $m_i \gets \max_{c} Z_{i, c}$
    \State $S_i \gets \sum_{c=1}^C \exp(Z_{i, c} - m_i)$
    \State $\mathrm{lse}_i \gets m_i + \log(S_i)$
    \State $\mathbf{P}_i \gets [\exp(Z_{i, c} - \mathrm{lse}_i)]_{c=1}^C$
    \State $\mathbf{Q}_i \gets [u]_{c=1}^C$
    \State $Q_{i, y_i} \gets u_{\mathrm{true}}$
    \State $l_i \gets -\sum_{c=1}^C Q_{i, c} \cdot (Z_{i, c} - \mathrm{lse}_i)$
    \State $\mathbf{G}_i \gets [P_{i, c} - Q_{i, c}]_{c=1}^C$
    \State Append $\mathbf{Q}_i$ to $\mathbf{Q}$, $\mathbf{P}_i$ to $\mathbf{P}$, $l_i$ to $\mathbf{l}$, $\mathbf{G}_i$ to $\mathbf{G}$
\EndFor
\If{$\mathrm{red} = \text{mean}$}
    \State \Return $\frac{1}{B}\sum l_i, \quad \mathbf{Q}, \quad \mathbf{P}, \quad \frac{1}{B}\mathbf{G}$
\ElsIf{$\mathrm{red} = \text{sum}$}
    \State \Return $\sum l_i, \quad \mathbf{Q}, \quad \mathbf{P}, \quad \mathbf{G}$
\Else
    \State \Return $\mathbf{l}, \quad \mathbf{Q}, \quad \mathbf{P}, \quad \mathbf{G}$
\EndIf
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Logit Boundedness}
\label{sec:guarantees}

\begin{theorem}[Finite Optimal Logit Separation]
\label{thm:finite_logit_gap}
\textbf{Assumptions:} Let smoothing $\epsilon \in (0, 1)$ across $C \ge 2$ classes.
\textbf{Guarantee:} The optimal logit difference $z_y^* - z_k^*$ between the correct class $y$ and any incorrect class $k \neq y$ that minimizes $\mathcal{L}_{\mathrm{LS}}$ is finite:
\begin{equation}
\Delta z^* = z_y^* - z_k^* = \log\left( 1 + \frac{C(1 - \epsilon)}{\epsilon} \right) < \infty.
\end{equation}
\textbf{Proof:}
At the minimum of cross-entropy, the prediction matches the target distribution: $p_k^* = q_k$ for all $k$.
Taking ratios between target class $y$ and non-target class $k$:
\begin{equation}
\frac{p_y^*}{p_k^*} = \frac{\exp(z_y^*)}{\exp(z_k^*)} = \frac{q_y}{q_k} = \frac{1 - \epsilon + \epsilon / C}{\epsilon / C} = \frac{C(1 - \epsilon) + \epsilon}{\epsilon} = 1 + \frac{C(1 - \epsilon)}{\epsilon}.
\end{equation}
Taking the natural logarithm on both sides yields the exact finite bound $\Delta z^* < \infty$.
\textbf{Limitations:} Softened targets prevent the network from reaching near-zero cross-entropy loss, and confidence scores require post-hoc temperature scaling for risk estimation.
\end{theorem}

\section{Complexity Analysis}
\label{sec:complexity}

\begin{itemize}
    \item \textbf{Time Complexity:} $\mathcal{O}(B \cdot C)$ floating-point operations.
    \item \textbf{Space Complexity:} $\mathcal{O}(B \cdot C)$ for soft target and probability matrices.
\end{itemize}

\section{Worked Numerical Example}
\label{sec:worked_example}

Let $B = 1, C = 3, \epsilon = 0.1$, target $y = 0$, logits $\mathbf{z} = [1.0, 0.0, 0.0]$:
\begin{enumerate}
    \item Targets: $\mathbf{q} = [0.933333, 0.033333, 0.033333]$.
    \item $\operatorname{LSE} = 1.0 + \log(1 + 2e^{-1.0}) = 1.551445$.
    \item Probabilities: $\mathbf{p} = [0.576117, 0.211942, 0.211942]$.
    \item Gradients: $\mathbf{g} = \mathbf{p} - \mathbf{q} = [-0.357216, +0.178608, +0.178608]$.
\end{enumerate}

\section{Numerical Considerations and Edge Cases}
\label{sec:numerical}

\begin{itemize}
    \item \textbf{Distillation Incompatibility:} When combined with Knowledge Distillation, teacher soft probabilities already soften labels; $\epsilon$ should typically be set to $0$ to avoid excessive entropy dilation.
\end{itemize}
