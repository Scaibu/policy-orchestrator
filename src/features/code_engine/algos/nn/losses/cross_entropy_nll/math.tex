\begin{abstract}
This document provides the rigorous mathematical specification, information-theoretic formulation, and gradient propagation mechanics of Multiclass Cross-Entropy and Negative Log-Likelihood (NLL) with class weighting and index masking. Every mathematical concept is explained across three levels: intuitive meaning, formal formulation, and practical algorithmic application. We prove the strict convexity of the objective function, provide step-by-step pseudocode matching the reference implementation, derive operation-level complexity bounds, and demonstrate exact numeric validation.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\mathbb{R}$ denote the set of real numbers, $\mathbb{R}^C$ the $C$-dimensional vector space of unnormalized logits, and $\Delta^{C-1} = \{\mathbf{p} \in \mathbb{R}^C \mid p_c \ge 0, \sum_{c=1}^C p_c = 1\}$ the probability simplex.

\subsection{Coordinate and Log-Sum-Exp Mechanics}
\paragraph{Intuitive Meaning:} Logits represent unnormalized log-odds scores produced by a neural network. Softmax normalizes them into a valid probability distribution, and Cross-Entropy measures the distance between these predictions and the true one-hot target.

\paragraph{Formal Mathematical Definition:}
For logit vector $\mathbf{z} = [z_1, \dots, z_C]^\top \in \mathbb{R}^C$, the numerically stabilized Log-Sum-Exp operator is:
\begin{equation}
\operatorname{LSE}(\mathbf{z}) = \max(\mathbf{z}) + \log \sum_{c=1}^C \exp(z_c - \max(\mathbf{z})).
\end{equation}
The softmax probability vector $\mathbf{p} = [p_1, \dots, p_C]^\top \in \Delta^{C-1}$ is given by:
\begin{equation}
p_c = \exp(z_c - \operatorname{LSE}(\mathbf{z})).
\end{equation}

\paragraph{Worked Numerical Example:}
Let $\mathbf{z} = [2.0, 1.0, 0.1]^\top$.
$\max(\mathbf{z}) = 2.0$.
$\sum_{c=1}^3 \exp(z_c - 2.0) = \exp(0) + \exp(-1.0) + \exp(-1.9) = 1.0 + 0.367879 + 0.149569 = 1.517448$.
$\operatorname{LSE}(\mathbf{z}) = 2.0 + \log(1.517448) = 2.417031$.
$\mathbf{p} = [\exp(2.0 - 2.417031), \exp(1.0 - 2.417031), \exp(0.1 - 2.417031)]^\top = [0.6590, 0.2424, 0.0986]^\top$.

\subsection{Summary of Symbols}
\begin{itemize}
    \item $B \in \mathbb{N}$: Batch size (number of independent samples).
    \item $C \in \mathbb{N}$: Number of mutually exclusive target classes ($C \ge 2$).
    \item $\mathbf{Z} \in \mathbb{R}^{B \times C}$: Matrix of unnormalized logit predictions.
    \item $\mathbf{y} \in \{0, 1, \dots, C-1\}^B$: Ground truth class label vector.
    \item $\mathbf{w} \in \mathbb{R}_{\ge 0}^C$: Optional non-negative class weighting vector.
    \item $\mathrm{ignore\_index} \in \mathbb{Z}$: Target sentinel value excluded from loss calculation.
    \item $\mathbf{P} \in \Delta^{C-1 \times B}$: Predicted softmax probability matrix.
    \item $\mathbf{G} = \nabla_{\mathbf{Z}} \mathcal{L} \in \mathbb{R}^{B \times C}$: Matrix of analytic gradients with respect to logits.
\end{itemize}

\section{Problem Definition and Core Concepts}
\label{sec:problem_definition}

\subsection{Cross-Entropy and Empirical Kullback-Leibler Divergence}
\paragraph{Intuitive Meaning:} Cross-entropy penalizes the model in proportion to the negative logarithm of the probability assigned to the ground-truth class. If the model assigns probability $1.0$ to the true class, the loss is zero; if it assigns probability near zero, the loss approaches infinity.

\paragraph{Formal Specification:}
For sample $i$ with target class $y_i \in \{0, \dots, C-1\}$ and class weight $w_{y_i}$:
\begin{equation}
\label{eq:nll_def}
\mathcal{L}_i = -w_{y_i} \log p_{i, y_i} = w_{y_i} \left[ \operatorname{LSE}(\mathbf{z}_i) - z_{i, y_i} \right].
\end{equation}

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:cross_entropy} specifies the exact forward and gradient routine matching \texttt{impl.py}.

\begin{algorithm}[H]
\caption{Multiclass Cross-Entropy Forward and Analytic Gradient Evaluation}
\label{alg:cross_entropy}
\begin{algorithmic}[1]
\Require Logits matrix $\mathbf{Z} \in \mathbb{R}^{B \times C}$, targets $\mathbf{y} \in \mathbb{Z}^B$, class weights $\mathbf{w} \in \mathbb{R}^C$, $\mathrm{ignore\_index} \in \mathbb{Z}$, reduction $\mathrm{red} \in \{\text{mean}, \text{sum}, \text{none}\}$.
\Ensure Loss $\mathcal{L}$, probabilities $\mathbf{P} \in \mathbb{R}^{B \times C}$, gradients $\mathbf{G} \in \mathbb{R}^{B \times C}$, valid sample count $N_{\mathrm{valid}}$.
\State Initialize $\mathbf{P} \gets [], \mathbf{l} \gets [], \mathbf{G} \gets [], W_{\mathrm{total}} \gets 0.0, N_{\mathrm{valid}} \gets 0$
\For{$i = 1$ to $B$}
    \State $m_i \gets \max_{c} Z_{i, c}$
    \State $S_i \gets \sum_{c=1}^C \exp(Z_{i, c} - m_i)$
    \State $\mathrm{lse}_i \gets m_i + \log(S_i)$
    \State $\mathbf{P}_i \gets [\exp(Z_{i, c} - \mathrm{lse}_i)]_{c=1}^C$
    \If{$y_i = \mathrm{ignore\_index}$}
        \State $l_i \gets 0.0, \quad \mathbf{G}_i \gets [0.0]_{c=1}^C$
    \Else
        \State $w \gets w_{y_i}$ (or $1.0$ if $\mathbf{w} = \text{None}$)
        \State $l_i \gets w \cdot (\mathrm{lse}_i - Z_{i, y_i})$
        \State $\mathbf{G}_i \gets [w \cdot (P_{i, c} - \mathbb{I}_{\{c = y_i\}})]_{c=1}^C$
        \State $W_{\mathrm{total}} \gets W_{\mathrm{total}} + w, \quad N_{\mathrm{valid}} \gets N_{\mathrm{valid}} + 1$
    \EndIf
    \State Append $l_i$ to $\mathbf{l}$, append $\mathbf{G}_i$ to $\mathbf{G}$
\EndFor
\If{$\mathrm{red} = \text{mean}$}
    \State $norm \gets \max(W_{\mathrm{total}}, 1.0)$
    \State \Return $\mathcal{L} \gets \frac{1}{norm}\sum_{i=1}^B l_i, \quad \mathbf{P}, \quad \mathbf{G} \gets \frac{1}{norm}\mathbf{G}, \quad N_{\mathrm{valid}}$
\ElsIf{$\mathrm{red} = \text{sum}$}
    \State \Return $\mathcal{L} \gets \sum_{i=1}^B l_i, \quad \mathbf{P}, \quad \mathbf{G}, \quad N_{\mathrm{valid}}$
\Else
    \State \Return $\mathbf{l}, \quad \mathbf{P}, \quad \mathbf{G}, \quad N_{\mathrm{valid}}$
\EndIf
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Convexity Analysis}
\label{sec:guarantees}

\begin{theorem}[Strict Convexity and Uniqueness of the Gradient Zero-Crossing]
\label{thm:ce_convexity_rigorous}
\textbf{Assumptions:} Let $\mathbf{z} \in \mathbb{R}^C$ restricted to the zero-sum subspace $\mathbf{1}^\perp = \{\mathbf{z} \in \mathbb{R}^C \mid \sum_{c=1}^C z_c = 0\}$ with strictly positive class weight $w_y > 0$.
\textbf{Guarantee:} The cross-entropy loss $\mathcal{L}(\mathbf{z}, y)$ is strictly convex on $\mathbf{1}^\perp$, with Hessian $\mathbf{H} = w_y (\operatorname{diag}(\mathbf{p}) - \mathbf{p}\mathbf{p}^\top)$ having strictly positive eigenvalues on $\mathbf{1}^\perp$.
\textbf{Proof:}
For any non-zero test vector $\mathbf{v} \in \mathbf{1}^\perp \setminus \{\mathbf{0}\}$:
\begin{equation}
\mathbf{v}^\top \mathbf{H} \mathbf{v} = w_y \left[ \sum_{c=1}^C p_c v_c^2 - \left(\sum_{c=1}^C p_c v_c\right)^2 \right] = w_y \operatorname{Var}_{\mathbf{p}}(\mathbf{v}).
\end{equation}
Because $p_c = \exp(z_c - \operatorname{LSE}(\mathbf{z})) > 0$ for all $c \in \{1, \dots, C\}$, the probability measure $\mathbf{p}$ has full support on $\{1, \dots, C\}$. The variance $\operatorname{Var}_{\mathbf{p}}(\mathbf{v}) = 0$ if and only if $v_1 = v_2 = \dots = v_C = k$. However, since $\mathbf{v} \in \mathbf{1}^\perp$, $\sum_{c=1}^C v_c = C k = 0 \implies k = 0 \implies \mathbf{v} = \mathbf{0}$, which contradicts $\mathbf{v} \neq \mathbf{0}$. Therefore, $\mathbf{v}^\top \mathbf{H} \mathbf{v} > 0$ for all $\mathbf{v} \in \mathbf{1}^\perp \setminus \{\mathbf{0}\}$, proving strict convexity.
\textbf{Limitations:} On unconstrained $\mathbb{R}^C$, the loss is shift-invariant along $\mathbf{1}$ ($\mathcal{L}(\mathbf{z} + c\mathbf{1}) = \mathcal{L}(\mathbf{z})$), meaning the global minimizer for separable data lies at infinity ($\|\mathbf{z}\| \to \infty$).
\end{theorem}

\section{Complexity Analysis}
\label{sec:complexity}

\begin{itemize}
    \item \textbf{Time Complexity:} $\mathcal{O}(B \cdot C)$ floating-point operations for max extraction, exponentiation, sum-reduction, and gradient derivation.
    \item \textbf{Space Complexity:} $\mathcal{O}(B \cdot C)$ storage for probability and gradient tensors.
\end{itemize}

\section{Worked Numerical Example}
\label{sec:worked_example}

Let batch $B = 1, C = 3$, target $y = 0$, logits $\mathbf{z} = [2.0, 1.0, 0.1]$, weight $w_0 = 1.0$:
\begin{enumerate}
    \item $m = 2.0 \implies \mathbf{z} - m = [0.0, -1.0, -1.9]$.
    \item Sum of exponentials: $S = 1.0 + 0.367879 + 0.149569 = 1.517448$.
    \item Log-Sum-Exp: $\mathrm{lse} = 2.0 + \log(1.517448) = 2.417031$.
    \item Probabilities: $\mathbf{p} = [0.6590, 0.2424, 0.0986]$.
    \item Loss: $\mathcal{L} = 1.0 \times (2.417031 - 2.0) = 0.417031$.
    \item Gradients: $\nabla_{\mathbf{z}} \mathcal{L} = [0.6590 - 1.0, 0.2424 - 0.0, 0.0986 - 0.0] = [-0.3410, 0.2424, 0.0986]$.
\end{enumerate}

\section{Numerical Considerations and Edge Cases}
\label{sec:numerical}

\begin{itemize}
    \item \textbf{Zero Underflow Protection:} Log-probabilities are computed directly as $z_c - \operatorname{LSE}(\mathbf{z})$ rather than $\log(\operatorname{softmax}(\mathbf{z}))$, preventing underflow to $\log(0) = -\infty$.
    \item \textbf{Ignore Index Masking:} Samples matching $\mathrm{ignore\_index}$ produce strictly zero loss and zero gradient vectors without distorting the batch normalization denominator.
\end{itemize}
