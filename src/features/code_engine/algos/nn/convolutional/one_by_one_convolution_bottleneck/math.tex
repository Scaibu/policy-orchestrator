\begin{abstract}
This document provides the formal mathematical specification, channel manifold projection theory, and computational efficiency bounds of $1 \times 1$ Convolutions and 3-Stage Bottleneck Architectures. Directly evaluating $3 \times 3$ convolutions on wide high-dimensional channel tensors incurs extreme FLOP overhead ($\mathcal{O}(H W C_{\text{in}} C_{\text{out}} K^2)$). The 3-stage bottleneck topology uses $1 \times 1$ linear projections to compress channels into a low-dimensional manifold $C_{\text{mid}} \ll C_{\text{in}}$, performs spatial filtering on the compressed subspace, and expands channels back to $C_{\text{out}}$ via a second $1 \times 1$ projection. We derive the complete 3-stage mathematical formulation, prove FLOP reduction bounds, analyze operational complexities, trace a worked numerical example, and document residual integration guidelines.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $X \in \mathbb{R}^{C_{\text{in}} \times H \times W}$ denote the input activation tensor entering the bottleneck block.

\begin{itemize}
    \item $C_{\text{in}} \in \mathbb{N}_{\ge 1}$: Input channel count (e.g., $C_{\text{in}} = 256$).
    \item $C_{\text{mid}} \in \mathbb{N}_{\ge 1}$: Bottleneck reduced channel count (e.g., $C_{\text{mid}} = 64$).
    \item $C_{\text{out}} \in \mathbb{N}_{\ge 1}$: Output expanded channel count (e.g., $C_{\text{out}} = 256$).
    \item $W_{\text{red}} \in \mathbb{R}^{C_{\text{mid}} \times C_{\text{in}}}$: $1 \times 1$ channel reduction projection matrix.
    \item $W_{\text{spat}} \in \mathbb{R}^{C_{\text{mid}} \times C_{\text{mid}} \times K_h \times K_w}$: $3 \times 3$ spatial filtering kernel tensor.
    \item $W_{\text{exp}} \in \mathbb{R}^{C_{\text{out}} \times C_{\text{mid}}}$: $1 \times 1$ channel expansion projection matrix.
    \item $\gamma = C_{\text{mid}} / C_{\text{in}}$: Bottleneck compression factor ($\gamma = 0.25$ typical).
\end{itemize}

\section{Problem Definition and Core Concepts}

\subsection{Intuitive Meaning}
A $1 \times 1$ convolution is a position-wise multi-layer perceptron that pools and mixes information across channels without altering spatial dimensions ($H \times W$). In deep networks like ResNet-50/101/152, replacing a single heavy $3 \times 3$ convolution with a $1 \times 1 \to 3 \times 3 \to 1 \times 1$ bottleneck reduces FLOPs by more than $50\%$ while increasing non-linear representational capacity.

\subsection{Formal Mathematical Recurrences}

\begin{enumerate}
    \item \textbf{Stage 1 (Channel Reduction $1 \times 1$)}:
    \begin{equation}
    Z_{\text{red}}(c_m, h, w) = \sum_{c_i=0}^{C_{\text{in}}-1} W_{\text{red}}(c_m, c_i) X(c_i, h, w)
    \end{equation}
    \item \textbf{Stage 2 (Spatial Filtering $3 \times 3$)}:
    \begin{equation}
    Z_{\text{spat}}(c_m, h, w) = \sum_{c_k=0}^{C_{\text{mid}}-1} \sum_{i=0}^{K_h-1} \sum_{j=0}^{K_w-1} W_{\text{spat}}(c_m, c_k, i, j) Z_{\text{red, pad}}(c_k, h + i, w + j)
    \end{equation}
    \item \textbf{Stage 3 (Channel Expansion $1 \times 1$)}:
    \begin{equation}
    Y(c_o, h, w) = \sum_{c_m=0}^{C_{\text{mid}}-1} W_{\text{exp}}(c_o, c_m) Z_{\text{spat}}(c_m, h, w)
    \end{equation}
\end{enumerate}

\section{Algorithm Specification}

\begin{algorithm}[H]
\caption{3-Stage Bottleneck Convolutional Block}
\label{alg:bottleneck_block}
\begin{algorithmic}[1]
\Require Input $X \in \mathbb{R}^{C_{\text{in}} \times H \times W}$, reduction $W_{\text{red}}$, spatial $W_{\text{spat}}$, expansion $W_{\text{exp}}$, padding $p$.
\Ensure Output $Y \in \mathbb{R}^{C_{\text{out}} \times H_{\text{out}} \times W_{\text{out}}}$, compression ratio $\gamma$.
\State Allocate $Z_{\text{red}} \in \mathbb{R}^{C_{\text{mid}} \times H \times W}$
\For{$c_m \gets 0 \textbf{ to } C_{\text{mid}} - 1$}
    \For{$h \gets 0 \textbf{ to } H - 1$}
        \For{$w \gets 0 \textbf{ to } W - 1$}
            \State $Z_{\text{red}}[c_m][h][w] \gets \sum_{c_i} W_{\text{red}}[c_m][c_i] \cdot X[c_i][h][w]$
        \EndFor
    \EndFor
\EndFor
\State Pad $Z_{\text{red}}$ by $p$ to $Z_{\text{red, pad}}$
\State Allocate $Z_{\text{spat}} \in \mathbb{R}^{C_{\text{mid}} \times H_{\text{out}} \times W_{\text{out}}}$
\For{$c_m \gets 0 \textbf{ to } C_{\text{mid}} - 1$}
    \For{$h \gets 0 \textbf{ to } H_{\text{out}} - 1$}
        \For{$w \gets 0 \textbf{ to } W_{\text{out}} - 1$}
            \State $Z_{\text{spat}}[c_m][h][w] \gets \sum_{c_k, i, j} W_{\text{spat}}[c_m][c_k][i][j] \cdot Z_{\text{red, pad}}[c_k][h+i][w+j]$
        \EndFor
    \EndFor
\EndFor
\State Allocate $Y \in \mathbb{R}^{C_{\text{out}} \times H_{\text{out}} \times W_{\text{out}}}$
\For{$c_o \gets 0 \textbf{ to } C_{\text{out}} - 1$}
    \For{$h \gets 0 \textbf{ to } H_{\text{out}} - 1$}
        \For{$w \gets 0 \textbf{ to } W_{\text{out}} - 1$}
            \State $Y[c_o][h][w] \gets \sum_{c_m} W_{\text{exp}}[c_o][c_m] \cdot Z_{\text{spat}}[c_m][h][w]$
        \EndFor
    \EndFor
\EndFor
\State \Return $\{Y, Z_{\text{red}}, Z_{\text{spat}}, C_{\text{mid}} / C_{\text{in}}\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and FLOP Reduction}

\begin{theorem}[Computational FLOP Ratio of Bottleneck Block]
Let $C_{\text{in}} = C_{\text{out}} = C$, $C_{\text{mid}} = \gamma C$ with $K = 3$. The FLOP ratio $\eta$ of the 3-stage bottleneck compared to a direct two-layer $3 \times 3$ standard residual block ($2 \times 3^2 C^2 = 18 C^2$) is:
\begin{equation}
\eta = \frac{2 \gamma C^2 + 9 \gamma^2 C^2}{18 C^2} = \frac{2\gamma + 9\gamma^2}{18}
\end{equation}
For standard ResNet compression $\gamma = 0.25$:
\begin{equation}
\eta = \frac{2(0.25) + 9(0.0625)}{18} = \frac{0.50 + 0.5625}{18} = \frac{1.0625}{18} \approx 5.9\%
\end{equation}
achieving a nearly $17\times$ computational speedup for equivalent receptive field capacity.
\end{theorem}

\begin{proof}
Direct FLOP summation for the 3 stages:
\begin{align}
\text{FLOPs}_{\text{stage 1}} &= H W \cdot C_{\text{in}} \cdot C_{\text{mid}} = H W \gamma C^2 \\
\text{FLOPs}_{\text{stage 2}} &= H W \cdot C_{\text{mid}}^2 \cdot 9 = 9 H W \gamma^2 C^2 \\
\text{FLOPs}_{\text{stage 3}} &= H W \cdot C_{\text{mid}} \cdot C_{\text{out}} = H W \gamma C^2 \\
\text{Total FLOPs} &= H W (2\gamma C^2 + 9\gamma^2 C^2)
\end{align}
Comparing against two standard $3 \times 3$ conv layers requiring $2 \times 9 H W C^2 = 18 H W C^2$ yields the exact ratio $\eta = \frac{2\gamma + 9\gamma^2}{18}$.
\end{proof}

\subsection{Limitations}
\begin{itemize}
    \item \textbf{Information Bottleneck}: If $\gamma < 0.1$, excessive channel compression destroys critical feature subspaces before spatial filtering.
\end{itemize}

\section{Complexity Analysis}

\subsection{Time Complexity}
\begin{itemize}
    \item \textbf{Total Time Complexity}: $\Theta(H \cdot W \cdot C_{\text{mid}} \cdot (C_{\text{in}} + K_h K_w C_{\text{mid}} + C_{\text{out}}))$ FLOPs.
\end{itemize}

\subsection{Space Complexity}
\begin{itemize}
    \item \textbf{Storage}: $\mathcal{O}(H \cdot W \cdot (C_{\text{mid}} + C_{\text{out}}))$ memory.
\end{itemize}

\section{Exactness and Error Analysis}

The operation is mathematically exact.

\section{Worked Numerical Example}

Let $C_{\text{in}} = 2, C_{\text{mid}} = 1, C_{\text{out}} = 2, H = 1, W = 1, K = 1 \times 1, p = 0$.
\begin{equation}
X = [[[2.0]], [[4.0]]], \quad W_{\text{red}} = [[0.5, 0.5]], \quad W_{\text{spat}} = [[[[2.0]]]], \quad W_{\text{exp}} = [[1.0], [2.0]]
\end{equation}

\begin{enumerate}
    \item \textbf{Reduction}:
    $Z_{\text{red}} = 0.5(2.0) + 0.5(4.0) = 1.0 + 2.0 = 3.0$.
    \item \textbf{Spatial Filter}:
    $Z_{\text{spat}} = 2.0(3.0) = 6.0$.
    \item \textbf{Expansion}:
    $Y_0 = 1.0(6.0) = 6.0, Y_1 = 2.0(6.0) = 12.0 \implies Y = [[[6.0]], [[12.0]]]$.
\end{enumerate}

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Projection Shortcuts}: When input and output channels mismatch, a parallel $1 \times 1$ convolution on the shortcut path aligns feature dimensions.
\end{itemize}
