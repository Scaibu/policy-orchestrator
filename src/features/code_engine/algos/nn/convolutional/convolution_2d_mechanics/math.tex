\begin{abstract}
This document provides the formal mathematical specification, discrete cross-correlation formulation, and geometric transformation mechanics of the 2D Convolution operation. In deep neural architectures, convolutional layers extract local spatial features by sliding multi-channel filter banks over spatial activation maps. We derive the output spatial coordinate mapping equations under arbitrary stride and padding, prove translation equivariance under infinite-grid boundary conditions, analyze computational and memory complexities, trace a worked numerical example matching the reference implementation, and document border padding modes.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $X \in \mathbb{R}^{C_{\text{in}} \times H_{\text{in}} \times W_{\text{in}}}$ denote the 3D input activation tensor with $C_{\text{in}}$ channels, height $H_{\text{in}}$, and width $W_{\text{in}}$. Let $W \in \mathbb{R}^{C_{\text{out}} \times C_{\text{in}} \times K_h \times K_w}$ denote the 4D filter weight tensor.

\begin{itemize}
    \item $C_{\text{in}}, C_{\text{out}} \in \mathbb{N}_{\ge 1}$: Input and output channel counts.
    \item $H_{\text{in}}, W_{\text{in}} \in \mathbb{N}_{\ge 1}$: Input spatial grid dimensions.
    \item $K_h, K_w \in \mathbb{N}_{\ge 1}$: Vertical and horizontal kernel kernel extents.
    \item $s_h, s_w \in \mathbb{N}_{\ge 1}$: Stride sliding step factors.
    \item $p_h, p_w \in \mathbb{N}_{\ge 0}$: Symmetric zero-padding thicknesses.
    \item $b \in \mathbb{R}^{C_{\text{out}}}$: Additive channel bias vector.
    \item $Y \in \mathbb{R}^{C_{\text{out}} \times H_{\text{out}} \times W_{\text{out}}}$: Output feature map tensor.
\end{itemize}

\section{Problem Definition and Core Concepts}

\subsection{Intuitive Meaning}
Dense fully connected layers assign separate weights to every spatial position, causing parameter explosion and discarding spatial translation structure. The 2D convolution operation enforces local connectivity (receptive fields of size $K_h \times K_w$) and weight sharing across all spatial locations, creating representations that naturally detect local features (edges, textures, object parts) anywhere in the input field.

\subsection{Formal Mathematical Definition}
The output spatial grid resolution is determined by:
\begin{align}
H_{\text{out}} &= \left\lfloor \frac{H_{\text{in}} + 2 p_h - K_h}{s_h} \right\rfloor + 1 \\
W_{\text{out}} &= \left\lfloor \frac{W_{\text{in}} + 2 p_w - K_w}{s_w} \right\rfloor + 1
\end{align}
Let $X_{\text{pad}} \in \mathbb{R}^{C_{\text{in}} \times (H_{\text{in}}+2p_h) \times (W_{\text{in}}+2p_w)}$ be the zero-padded input. For each output channel $c_{\text{out}} \in \{0, \dots, C_{\text{out}}-1\}$ and coordinate $(h_{\text{out}}, w_{\text{out}})$:
\begin{equation}
Y_{c_{\text{out}}, h_{\text{out}}, w_{\text{out}}} = b_{c_{\text{out}}} + \sum_{c_{\text{in}}=0}^{C_{\text{in}}-1} \sum_{i=0}^{K_h-1} \sum_{j=0}^{K_w-1} W_{c_{\text{out}}, c_{\text{in}}, i, j} \cdot X_{\text{pad}}(c_{\text{in}}, h_{\text{out}} s_h + i, w_{\text{out}} s_w + j)
\end{equation}

\section{Algorithm Specification}

\begin{algorithm}[H]
\caption{2D Discrete Cross-Correlation Convolution Forward Pass}
\label{alg:conv2d}
\begin{algorithmic}[1]
\Require Input $X \in \mathbb{R}^{C_{\text{in}} \times H_{\text{in}} \times W_{\text{in}}}$, weights $W \in \mathbb{R}^{C_{\text{out}} \times C_{\text{in}} \times K_h \times K_w}$, bias $b \in \mathbb{R}^{C_{\text{out}}}$, strides $(s_h, s_w)$, padding $(p_h, p_w)$.
\Ensure Convolved tensor $Y \in \mathbb{R}^{C_{\text{out}} \times H_{\text{out}} \times W_{\text{out}}}$, dimensions $H_{\text{out}}, W_{\text{out}}$.
\State $H_{\text{out}} \gets \lfloor (H_{\text{in}} + 2p_h - K_h) / s_h \rfloor + 1$, $W_{\text{out}} \gets \lfloor (W_{\text{in}} + 2p_w - K_w) / s_w \rfloor + 1$
\State Construct zero-padded tensor $X_{\text{pad}} \in \mathbb{R}^{C_{\text{in}} \times (H_{\text{in}}+2p_h) \times (W_{\text{in}}+2p_w)}$
\State Allocate $Y \in \mathbb{R}^{C_{\text{out}} \times H_{\text{out}} \times W_{\text{out}}}$
\For{$c_{\text{out}} \gets 0 \textbf{ to } C_{\text{out}} - 1$}
    \For{$h \gets 0 \textbf{ to } H_{\text{out}} - 1$}
        \State $h_{\text{start}} \gets h \cdot s_h$
        \For{$w \gets 0 \textbf{ to } W_{\text{out}} - 1$}
            \State $w_{\text{start}} \gets w \cdot s_w$
            \State $\text{acc} \gets b[c_{\text{out}}]$
            \For{$c_{\text{in}} \gets 0 \textbf{ to } C_{\text{in}} - 1$}
                \For{$i \gets 0 \textbf{ to } K_h - 1$}
                    \For{$j \gets 0 \textbf{ to } K_w - 1$}
                        \State $\text{acc} \gets \text{acc} + W[c_{\text{out}}][c_{\text{in}}][i][j] \cdot X_{\text{pad}}[c_{\text{in}}][h_{\text{start}} + i][w_{\text{start}} + j]$
                    \EndFor
                \EndFor
            \EndFor
            \State $Y[c_{\text{out}}][h][w] \gets \text{acc}$
        \EndFor
    \EndFor
\EndFor
\State \Return $\{Y, H_{\text{out}}, W_{\text{out}}\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Translation Equivariance}

\begin{theorem}[Translation Equivariance of 2D Convolution]
Let $T_{\Delta}$ denote the spatial translation operator $(T_{\Delta} X)(c, y, x) = X(c, y - \Delta_y, x - \Delta_x)$ over the continuous plane $\mathbb{R}^2$ with stride $s_h = s_w = 1$. Then 2D convolution $\Phi_W$ commutes with translation:
\begin{equation}
\Phi_W(T_{\Delta} X) = T_{\Delta}(\Phi_W(X))
\end{equation}
\end{theorem}

\begin{proof}
Evaluating the output at coordinate $(y, x)$ with shifted input $X' = T_\Delta X$:
\begin{align}
\Phi_W(X')(y, x) &= \sum_{c_{\text{in}}} \int \int W(c_{\text{out}}, c_{\text{in}}, u, v) X'(c_{\text{in}}, y + u, x + v) \, du dv \\
&= \sum_{c_{\text{in}}} \int \int W(c_{\text{out}}, c_{\text{in}}, u, v) X(c_{\text{in}}, (y - \Delta_y) + u, (x - \Delta_x) + v) \, du dv \\
&= \Phi_W(X)(y - \Delta_y, x - \Delta_x) = (T_\Delta \Phi_W(X))(y, x)
\end{align}
proving exact translation equivariance.
\end{proof}

\subsection{Limitations}
\begin{itemize}
    \item \textbf{Scale and Rotation Non-Equivariance}: Standard convolutions are equivariant to spatial shifts, but not to spatial rotation or isotropic scaling.
\end{itemize}

\section{Complexity Analysis}

\subsection{Time Complexity}
\begin{itemize}
    \item \textbf{Multiply-Accumulate Operations}: For each output cell, $C_{\text{in}} K_h K_w$ multiplications and additions are performed.
    \item \textbf{Total Time Complexity}: $\Theta(C_{\text{out}} \cdot H_{\text{out}} \cdot W_{\text{out}} \cdot C_{\text{in}} \cdot K_h \cdot K_w)$ FLOPs.
\end{itemize}

\subsection{Space Complexity}
\begin{itemize}
    \item \textbf{Activation Storage}: $\mathcal{O}(C_{\text{out}} \cdot H_{\text{out}} \cdot W_{\text{out}})$ memory for the output tensor.
\end{itemize}

\section{Exactness and Error Analysis}

The operation computes exact inner products over floating-point arithmetic.

\section{Worked Numerical Example}

Let $C_{\text{in}} = 1, H_{\text{in}} = 2, W_{\text{in}} = 2, C_{\text{out}} = 1, K_h = 2, K_w = 2, s_h = s_w = 1, p_h = p_w = 0, b = [0.5]$.
\begin{equation}
X = [[[1.0, 2.0], [3.0, 4.0]]], \quad W = [[[[0.1, 0.2], [0.3, 0.4]]]]
\end{equation}
Spatial output shape: $H_{\text{out}} = (2 - 2)/1 + 1 = 1, W_{\text{out}} = (2 - 2)/1 + 1 = 1$.

\begin{enumerate}
    \item \textbf{Dot Product Calculation}:
    \begin{align}
    \text{Sum} &= 0.1(1.0) + 0.2(2.0) + 0.3(3.0) + 0.4(4.0) \\
    &= 0.10 + 0.40 + 0.90 + 1.60 = 3.00
    \end{align}
    \item \textbf{Add Bias}:
    \begin{equation}
    Y_{0, 0, 0} = 3.00 + 0.50 = 3.50
    \end{equation}
\end{enumerate}

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Zero Channels / Empty Input}: Dimensions $H_{\text{in}} + 2p_h < K_h$ fail validation, raising explicit `ValueError`.
\end{itemize}
