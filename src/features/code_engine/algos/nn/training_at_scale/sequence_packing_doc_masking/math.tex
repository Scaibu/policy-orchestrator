\section{Definitions and Cognitive Mental Models}

\subsection{Formal Mathematical Definition}
\textbf{Definition (Sequence Packing and Block-Diagonal Attention Masking):}
In standard language model training, training examples vary widely in token length. Padding shorter documents to a uniform sequence length $L_{\max}$ wastes a substantial fraction ($30\%\text{--}70\%$) of computational FLOPs on padding tokens whose loss is masked out. \textbf{Sequence Packing} concatenates multiple variable-length documents $\{D_1, D_2, \dots, D_m\}$ into a single packed sequence of length $L \le L_{\max}$ using bin-packing heuristics (such as First-Fit Decreasing, FFD).

To prevent cross-contamination and unintended causal attention across unrelated documents packed within the same sequence, \textbf{Document Masking} enforces a block-diagonal causal attention mask $M \in \{0, 1\}^{L \times L}$:
{\boldmath
\begin{equation}
M_{i, j} = \begin{cases}
1, & \text{if } j \le i \text{ and } \operatorname{doc}(i) = \operatorname{doc}(j) \\
0, & \text{otherwise}
\end{cases}
\end{equation}
}
where $\operatorname{doc}(i)$ denotes the document index owning token position $i$. Furthermore, positional embeddings (RoPE or absolute position IDs) are reset to zero at the start of each individual document:
{\boldmath
\begin{equation}
\operatorname{pos}(i) = i - \operatorname{offset}(\operatorname{doc}(i))
\end{equation}
}
ensuring that token positions reflect the document's internal structure rather than its arbitrary packed offset.

\subsection{Expanded Non-Technical Definition (Intuition for Anyone)}
\begin{enumerate}
    \item \textbf{Car-Pooling for Words}: Imagine a 50-passenger bus ($L_{\max}=2048$). If passenger A only needs 5 seats and passenger B only needs 10 seats, sending two separate buses wastes 85 empty seats. Sequence packing puts passenger A, B, and C on the same bus until all 50 seats are full!
    \item \textbf{Soundproof Partitions}: To prevent passengers from overhearing each other, you put soundproof glass dividers between the groups (document masking). Passenger B cannot attend to or talk to passenger A.
    \item \textbf{Zero Wasted Electricity}: Packing increases training efficiency by up to $2\times\text{--}3\times$, as nearly 100\% of GPU tensor core cycles process actual semantic content rather than useless padding tokens.
\end{enumerate}

\subsection{Child-Friendly Mental Model (Physical Metaphor)}
Imagine packing boxes for moving:
\begin{itemize}
    \item \textbf{Naive Padding}: Putting a tiny coffee cup in a giant refrigerator box and filling the rest with Styrofoam peanuts. You waste 100 boxes!
    \item \textbf{Sequence Packing}: Packing the coffee cup, a book, and a shoe tightly into one box so there is zero empty space.
\end{itemize}

\section{Core Mathematical Equations and Definitions}

\subsection{Block-Diagonal Attention Mask}
{\boldmath
\begin{equation}
A = \operatorname{Softmax}\left( \frac{Q K^T}{\sqrt{d}} + (M - 1) \cdot \infty \right) V
\end{equation}
}
\begin{itemize}
    \item \textbf{Formal Definition}: Attention operator with $-\infty$ masking on non-causal or cross-document entries.
    \item \textbf{What It Means}: Completely zeros out attention weights between different documents.
    \item \textbf{Why It Is Important}: Prevents information leakage and semantic confusion during multi-document training.
\end{itemize}

\subsection{Padding Efficiency Metric}
{\boldmath
\begin{equation}
\eta_{\text{pack}} = \frac{\sum_{k=1}^N |D_k|}{B \cdot L_{\max}}
\end{equation}
}
\begin{itemize}
    \item \textbf{Formal Definition}: Ratio of valid semantic tokens to total allocated sequence matrix cells across $B$ bins.
    \item \textbf{What It Means}: Measures how close the training run is to 100\% computational utilization.
    \item \textbf{Why It Is Important}: Primary KPI for batch throughput optimization.
\end{itemize}

\section{Rigorous Mathematical Analysis Checklist}

\begin{itemize}
    \item \textbf{Notation}: $D_k$ (documents with token lengths $|D_k|$), $L_{\max}$ (bin capacity), $B$ (total bins), $M \in \{0, 1\}^{L \times L}$ (mask).
    \item \textbf{Epistemic Status}: Standardized in Megatron-LM and FlashAttention (Dao et al., 2022). Universal practice in LLM pre-training.
    \item \textbf{Dependency Graph}: Document Lengths $\to \text{FFD Bin Packing} \to \text{Packed Bins} \to \text{Reset Pos IDs} \to \text{Block Mask}$.
    \item \textbf{Dimensions}: Sequence length $L_{\max} \in [2048, 8192]$, document counts $N$.
    \item \textbf{Regimes}: Instruction fine-tuning (SFT) where conversations have high length variance.
    \item \textbf{Invariants}: Cumulative lengths in any bin satisfy $\sum_{i \in \text{bin}} |D_i| \le L_{\max}$.
    \item \textbf{Isomorphisms}: Isomorphic to the classic 1D Bin Packing problem with item sizes $|D_i|$.
\end{itemize}

\section{Theoretical Theorems and Formal Proofs}

\begin{theorem}[Asymptotic Bound of First-Fit Decreasing Packing]
Let $\mathcal{D} = \{|D_1|, \dots, |D_N|\}$ be a collection of document token lengths with $|D_i| \le L_{\max}$. The number of sequence bins $B_{\text{FFD}}$ allocated by First-Fit Decreasing satisfies:
{\boldmath
\begin{equation}
B_{\text{FFD}} \le \frac{11}{9} B_{\text{OPT}} + \frac{6}{9}
\end{equation}
}
where $B_{\text{OPT}} = \lceil \frac{1}{L_{\max}} \sum_{i=1}^N |D_i| \rceil$ is the theoretical optimal bin count.
\end{theorem}

\begin{proof}
This follows directly from Johnson's theorem (1973) on the competitive ratio of First-Fit Decreasing for one-dimensional bin packing.
1. Documents are sorted in descending order of length: $|D_{(1)}| \ge |D_{(2)}| \ge \dots \ge |D_{(N)}|$.
2. Each document is greedily placed into the first open bin whose remaining capacity satisfies $C_{\text{rem}} \ge |D_{(i)}|$.
3. By the FFD packing lemma, at most one bin can be less than half full ($< L_{\max}/2$).
4. Integrating the weight functions across the packing intervals establishes the exact asymptotic competitive ratio:
{\boldmath
\begin{equation}
\limsup_{B_{\text{OPT}} \to \infty} \frac{B_{\text{FFD}}}{B_{\text{OPT}}} = \frac{11}{9} \approx 1.22
\end{equation}
}
Thus, FFD guarantees that total packed sequences are within $22\%$ of theoretical maximum density, providing near-optimal packing without combinatorial search overhead.
\end{proof}

\begin{theorem}[Strict Information Isolation Under Block-Diagonal Masking]
Let $i$ and $j$ be two token positions in a packed sequence belonging to distinct documents ($\operatorname{doc}(i) \neq \operatorname{doc}(j)$). Under the block-diagonal mask $M$, the gradient of token $i$'s output with respect to token $j$'s input embeddings is identically zero:
{\boldmath
\begin{equation}
\frac{\partial h_i}{\partial x_j} = \mathbf{0}
\end{equation}
}
\end{theorem}

\begin{proof}
Consider the attention weight between position $i$ and position $j$:
{\boldmath
\begin{equation}
A_{i, j} = \frac{\exp\left( \frac{q_i^T k_j}{\sqrt{d}} + (M_{i, j} - 1) \cdot \infty \right)}{\sum_{l=1}^L \exp\left( \frac{q_i^T k_l}{\sqrt{d}} + (M_{i, l} - 1) \cdot \infty \right)}
\end{equation}
}
When $\operatorname{doc}(i) \neq \operatorname{doc}(j)$, $M_{i, j} = 0$, giving $(0 - 1) \cdot \infty = -\infty$.
Taking the exponential:
{\boldmath
\begin{equation}
\exp\left( \frac{q_i^T k_j}{\sqrt{d}} - \infty \right) = 0 \implies A_{i, j} = 0
\end{equation}
}
The hidden representation $h_i$ at the output of the attention layer is:
{\boldmath
\begin{equation}
h_i = \sum_{l=1}^L A_{i, l} v_l = \sum_{l \in \operatorname{doc}(i), l \le i} A_{i, l} v_l
\end{equation}
}
Since $j \notin \operatorname{doc}(i)$, $h_i$ contains zero dependence on $v_j, k_j$, or $x_j$.
By the chain rule of differentiation:
{\boldmath
\begin{equation}
\frac{\partial h_i}{\partial x_j} = \sum_{l=1}^L \frac{\partial h_i}{\partial v_l} \frac{\partial v_l}{\partial x_j} + \sum_{l=1}^L \frac{\partial h_i}{\partial A_{i, l}} \frac{\partial A_{i, l}}{\partial x_j} = \mathbf{0}
\end{equation}
}
Because this holds for all layers by induction, cross-document representations remain strictly isolated throughout network depth.
\end{proof}

\section{Foundational Architectural Questions and Epistemic Meta-Questions}

\subsection{Architectural Question 1: Un-materialized Variable-Length Kernels}
\textbf{Question:} Does sequence packing require allocating full $L_{\max} \times L_{\max}$ attention masks in GPU memory?\\
\textbf{Answer:} No. Modern attention kernels (such as FlashAttention-2 varlen) do not materialize dense attention masks. Instead, they accept a flat 1D token tensor alongside a lightweight cumulative offsets array (`cu_seqlens` e.g. `[0, 300, 800, 1500, 2048]`). The GPU kernel limits its attention thread blocks exclusively within the boundaries of each document, saving both memory and FLOPs.

\textbf{Meta-Question:} Should loss be computed on separator tokens between packed documents?\\
\textbf{Answer:} Typically, separator tokens (e.g. `<|endoftext|>`) are supervised with next-token prediction to teach the model when to terminate generation, but subsequent document start tokens must have their loss masked to prevent predicting the first token of an unrelated document.

\subsection{Architectural Question 2: Positional ID Resetting}
\textbf{Question:} What happens if position IDs are not reset to zero at each document boundary?\\
\textbf{Answer:} If position IDs are continuous ($0$ to $2047$), the second document in a packed sequence begins at position ID $500$ instead of $0$. The model will incorrectly believe the conversation started 500 tokens in, distorting RoPE rotary angles and degrading prompt following. Resetting position IDs ensures every document is represented identically to its standalone form.

\section{Algorithmic Implementation Specification}

\begin{algorithm}[H]
\caption{Sequence Packing via First-Fit Decreasing and Boundary Generation}
\begin{algorithmic}[1]
\Require Document lengths $\mathbf{L}_{\text{doc}} \in \mathbb{N}^N$, maximum sequence length $L_{\max}$
\Ensure Packed bins $\mathcal{B}$, bin lengths $\mathbf{C}$, position IDs $\mathbf{P}$, document boundaries $\mathbf{E}$, efficiency $\eta$
\State Sort indices $\pi \gets \operatorname{ArgsortDescending}(\mathbf{L}_{\text{doc}})$
\State Allocate $\mathcal{B} \gets [\,], \quad \mathbf{R} \gets [\,]$ \Comment{Bins and remaining capacities}
\For{$idx \in \pi$}
    \State $\ell \gets \mathbf{L}_{\text{doc}}[idx], \quad \text{placed} \gets \text{False}$
    \For{$b = 0$ \textbf{to} $|\mathbf{R}| - 1$}
        \If{$\mathbf{R}[b] \ge \ell$}
            \State Append $idx$ to $\mathcal{B}[b], \quad \mathbf{R}[b] \gets \mathbf{R}[b] - \ell, \quad \text{placed} \gets \text{True}$
            \State \textbf{break}
        \EndIf
    \EndFor
    \If{\textbf{not} placed}
        \State Append $[idx]$ to $\mathcal{B}, \quad \text{Append } (L_{\max} - \ell) \text{ to } \mathbf{R}$
    \EndIf
\EndFor
\State Allocate $\mathbf{P} \gets [\,], \quad \mathbf{E} \gets [\,], \quad \mathbf{C} \gets [\,]$
\For{$b = 0$ \textbf{to} $|\mathcal{B}| - 1$}
    \State $\text{offset} \gets 0, \quad \text{pos\_bin} \gets [\,], \quad \text{bounds} \gets [\,]$
    \For{$doc \in \mathcal{B}[b]$}
        \State $\text{pos\_bin} \gets \text{pos\_bin} \cup [0, \dots, \mathbf{L}_{\text{doc}}[doc] - 1]$
        \State $\text{offset} \gets \text{offset} + \mathbf{L}_{\text{doc}}[doc]$
        \State Append $\text{offset}$ to $\text{bounds}$
    \EndFor
    \State Append $\text{pos\_bin}$ to $\mathbf{P}, \quad \text{Append } \text{bounds to } \mathbf{E}, \quad \text{Append } \text{offset to } \mathbf{C}$
\EndFor
\State $\eta \gets \left( \sum_{i=0}^{N-1} \mathbf{L}_{\text{doc}}[i] \right) / (|\mathcal{B}| \cdot L_{\max})$
\State \Return $(\mathcal{B}, \mathbf{C}, \mathbf{P}, \mathbf{E}, \eta)$
\end{algorithmic}
\end{algorithm}

\section{Big-$\Theta$ Complexity Analysis}

\begin{itemize}
    \item \textbf{Sorting Time}: $\Theta(N \log N)$ operations for descending length ordering.
    \item \textbf{Packing Placement Time}: $\mathcal{O}(N \cdot B)$ search through open bins.
    \item \textbf{Position \& Boundary Generation}: $\Theta(\sum |D_i|)$ linear token passes.
    \item \textbf{Total Time Complexity}: $\mathcal{O}(N \log N + N \cdot B + \text{TotalTokens})$.
    \item \textbf{Space Complexity}: $\Theta(\text{TotalTokens})$ to hold position IDs and packed indices.
\end{itemize}
