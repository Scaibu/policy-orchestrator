\begin{abstract}
This document presents the complete mathematical derivation and algorithmic specification of Numerically Stable Softmax, Log-Softmax, and Online Streaming Softmax (FlashAttention primitive). Evaluating naive softmax $p_i = \frac{e^{z_i}}{\sum_j e^{z_j}}$ triggers catastrophic floating-point overflow ($e^{z} \to \text{inf}$ for $z > 709$) or severe underflow ($e^z \to 0$ for $z < -709$). We prove the translation invariance theorem, derive the 3-pass and 1-pass online recurrences, provide the complete algorithmic pseudo-code, derive exact computational complexities, step through a numerical example, and analyze Jacobian dynamics.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $Z \in \mathbb{R}^{B \times K}$ denote a 2D matrix of unnormalized logit scores across $B$ batch items and $K$ discrete categorical classes.

\begin{itemize}
    \item $z \in \mathbb{R}^K$: Logit vector for a single batch item.
    \item $z_{\max} \triangleq \max_{k=1, \dots, K} z_k$: Maximum logit entry in the vector.
    \item $T > 0$: Thermodynamic temperature scaling factor ($T = 1.0$ by default).
    \item $p \in \Delta^{K-1} \triangleq \{p \in \mathbb{R}^K : p_k \ge 0, \sum_k p_k = 1\}$: Probability simplex vector.
    \item $\operatorname{LSE}(z) \triangleq \ln \left( \sum_{k=1}^K e^{z_k} \right)$: Log-Sum-Exp partition scalar.
\end{itemize}

\section{Problem Definition and Core Concepts}

\subsection{Intuition and Motivation}
Softmax converts unbounded real logits into a normalized probability distribution over $K$ classes. In IEEE-754 double precision, numbers exceeding $\approx 709.78$ overflow to $+\infty$. If any single logit $z_k \ge 710$, evaluating $e^{z_k}$ yields $\infty$, causing $\infty / \infty = \text{NaN}$. Stable softmax subtracts the maximum logit $z_{\max}$ from every component, ensuring the largest exponent evaluated is $e^0 = 1.0$.

\subsection{Formal Mathematical Definition}
\begin{definition}[Stable Softmax and Log-Softmax Formulations]
For logit vector $z \in \mathbb{R}^K$ and temperature $T > 0$, let $\tilde{z} = z / T$ and $m = \max_{k} \tilde{z}_k$.
\begin{align}
p_k &= \frac{e^{\tilde{z}_k - m}}{\sum_{j=1}^K e^{\tilde{z}_j - m}} \label{eq:stable_softmax} \\
\operatorname{LSE}(z) &= m + \ln\left( \sum_{j=1}^K e^{\tilde{z}_j - m} \right) \label{eq:lse_formula} \\
\ln p_k &= (\tilde{z}_k - m) - \ln\left( \sum_{j=1}^K e^{\tilde{z}_j - m} \right) = \tilde{z}_k - \operatorname{LSE}(z) \label{eq:log_softmax}
\end{align}
\end{definition}

\section{Algorithmic Specification}

The computational pipeline implemented in \texttt{NnAlgoSoftmaxStable} is detailed below.

\begin{algorithm}[H]
\caption{Numerically Stable Softmax and Online Streaming Execution}
\begin{algorithmic}[1]
\Require Logit batch $Z \in \mathbb{R}^{B \times K}$, temperature $T > 0$, mode $\in \{\text{standard\_stable}, \text{log\_softmax}, \text{online\_streaming}\}$
\Ensure Probability matrix $P \in \mathbb{R}^{B \times K}$, LSE vector of length $B$
\State $B \gets \text{rows}(Z), \quad K \gets \text{cols}(Z)$
\State $P \gets \text{new Matrix of shape } (B, K), \quad \text{lse\_vec} \gets \text{new Array of size } B$
\For{$b \gets 1 \textbf{ to } B$}
    \State $m \gets \max_{k=1}^K (Z[b][k] / T)$
    \State $S \gets \sum_{k=1}^K \exp(Z[b][k] / T - m)$
    \State $\text{lse} \gets m + \ln(S)$
    \State $\text{lse\_vec}[b] \gets \text{lse}$
    \For{$k \gets 1 \textbf{ to } K$}
        \If{$\text{mode} = \text{log\_softmax}$}
            \State $P[b][k] \gets (Z[b][k] / T) - \text{lse}$
        \Else
            \State $P[b][k] \gets \exp(Z[b][k] / T - m) / S$
        \EndIf
    \EndFor
\EndFor
\State \Return $\{\text{probabilities}: P, \text{log\_sum\_exp}: \text{lse\_vec}\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Translation Invariance}

\begin{theorem}[Exact Translation Invariance of Softmax]
For any logit vector $z \in \mathbb{R}^K$ and any constant scalar $c \in \mathbb{R}$:
\begin{equation}
\operatorname{Softmax}(z + c \mathbf{1}) = \operatorname{Softmax}(z)
\end{equation}
\end{theorem}

\begin{proof}
By direct substitution:
\begin{equation}
\frac{e^{(z_k + c)}}{\sum_{j=1}^K e^{(z_j + c)}} = \frac{e^c \cdot e^{z_k}}{e^c \sum_{j=1}^K e^{z_j}} = \frac{e^{z_k}}{\sum_{j=1}^K e^{z_j}}
\end{equation}
Setting $c = -m = -\max_k z_k$ preserves mathematical equality identically while bounding exponent arguments to $(-\infty, 0]$.
\end{proof}

\subsection{Limitations}
\begin{itemize}
    \item \textbf{Softmax Bottleneck}: For huge output vocabularies ($K > 250\text{k}$), evaluating full softmax over all tokens becomes compute/memory bound (solved by sampled softmax or hierarchical softmax).
\end{itemize}

\section{Complexity Analysis}

\subsection{Time and Space Complexity}
\begin{itemize}
    \item \textbf{Time}: 3 linear passes per row: finding max ($\mathcal{O}(K)$), sum of exp ($\mathcal{O}(K)$), and normalization ($\mathcal{O}(K)$). Total: $\Theta(BK)$ FLOPs.
    \item \textbf{Space}: $\Theta(BK)$ auxiliary memory for probabilities.
\end{itemize}

\section{Worked Numerical Example}

Consider logits $z = [1000.0, 1002.0, 999.0]$ at $T=1.0$.

\begin{enumerate}
    \item $m = \max(1000, 1002, 999) = 1002.0$.
    \item Shifted logits:
    \begin{align}
    \tilde{z}_1 - m &= 1000.0 - 1002.0 = -2.0 \\
    \tilde{z}_2 - m &= 1002.0 - 1002.0 = 0.0 \\
    \tilde{z}_3 - m &= 999.0 - 1002.0 = -3.0
    \end{align}
    \item Exponentials:
    $e^{-2} \approx 0.135335, \quad e^0 = 1.0, \quad e^{-3} \approx 0.049787$.
    \item Denominator Sum:
    $S = 0.135335 + 1.0 + 0.049787 = 1.185122$.
    \item Probabilities:
    \begin{align}
    p_1 &= 0.135335 / 1.185122 \approx 0.114195 \\
    p_2 &= 1.000000 / 1.185122 \approx 0.843795 \\
    p_3 &= 0.049787 / 1.185122 \approx 0.042010
    \end{align}
    Total $\sum p_k = 1.000000$.
\end{enumerate}

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Online Softmax (Milakov & Gimelshein 2018)}: Updates running maximum $m_{\text{new}} = \max(m_{\text{old}}, x_i)$ and rescales accumulator $S_{\text{new}} = S_{\text{old}} e^{m_{\text{old}} - m_{\text{new}}} + e^{x_i - m_{\text{new}}}$, enabling fused attention kernels (FlashAttention).
\end{itemize}
