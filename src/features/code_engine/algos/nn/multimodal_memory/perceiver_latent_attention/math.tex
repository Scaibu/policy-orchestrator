\begin{abstract}
Processing high-dimensional raw multimodal inputs (e.g., $M \sim 10^5$ raw bytes, pixels, or audio points) with full self-attention requires prohibitive $\mathcal{O}(M^2)$ memory and compute. Perceiver architectures introduce an asymmetric cross-attention bottleneck where a small learnable latent array of size $N \ll M$ queries the inputs, distilling the input into a compact representation and processing deep transformer blocks in latent space in linear $\mathcal{O}(NM)$ time. This document formalizes the asymmetric projection, complexity bounds, and numerical verification.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $\mathbf{X} \in \mathbb{R}^{M \times d}$ denote input arrays of length $M$, and let $\mathbf{L} \in \mathbb{R}^{N \times d}$ denote learnable latent bottleneck vectors with $N \ll M$.

\begin{equation}
\mathbf{L}_{\text{next}} = \operatorname{softmax}\left(\frac{\mathbf{L} \mathbf{X}^\top}{\sqrt{d}}\right) \mathbf{X}
\end{equation}

\section{Complexity Analysis}
\begin{itemize}
    \item \textbf{Cross-Attention Bottleneck:} $\Theta(N \cdot M \cdot d)$ operations instead of $\mathcal{O}(M^2 d)$.
    \item \textbf{Latent Self-Attention:} $\Theta(N^2 d)$ operations, independent of input size $M$.
\end{itemize}
