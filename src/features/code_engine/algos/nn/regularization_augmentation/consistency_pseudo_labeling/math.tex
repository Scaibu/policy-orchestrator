\begin{abstract}
This document provides the formal mathematical specification, cluster assumption analysis, and consistency regularization theory of FixMatch and Pseudo-Labeling. Semi-supervised learning leverages small labeled sets alongside vast unlabeled corpora by enforcing consistency across different stochastic perturbation manifolds. FixMatch generates artificial hard pseudo-labels from weakly augmented inputs when the maximum predicted class probability exceeds a high confidence threshold $\tau$, and penalizes cross-entropy error on heavily perturbed views of the same sample. We derive the analytical loss formulation, prove error propagation bounds, analyze time-space complexities, trace a worked example, and document confirmation bias safeguards.
\end{abstract}

\section{Notation and Mathematical Preliminaries}

Let $\mathcal{U} = \{u_1, \dots, u_B\}$ denote an unlabeled mini-batch of $B$ samples. Let $\alpha(\cdot)$ denote a weak augmentation morphism and $\mathcal{A}(\cdot)$ denote a strong augmentation operator.

\begin{itemize}
    \item $B \in \mathbb{N}_{\ge 1}$: Unlabeled batch size.
    \item $K \in \mathbb{N}_{\ge 1}$: Number of categorical target classes.
    \item $q_b = p_{\text{model}}(y \mid \alpha(u_b)) \in \Delta^{K-1}$: Predicted softmax probability distribution for weakly augmented sample $b$.
    \item $z_b = f_\theta(\mathcal{A}(u_b)) \in \mathbb{R}^K$: Unnormalized logit vector for strongly augmented sample $b$.
    \item $\hat{q}_b = \arg\max_k q_{b, k}$: Predicted argmax class index.
    \item $\tau \in (0, 1)$: Fixed confidence threshold ($\tau = 0.95$ default).
    \item $m_b = \mathbb{I}(\max_k q_{b, k} \ge \tau)$: Binary pseudo-label confidence mask.
    \item $\mathcal{L}_u$: Average unsupervised consistency loss.
\end{itemize}

\section{Problem Definition and Core Concepts}

\subsection{Intuitive Meaning}
Labeling massive datasets is expensive, but unlabeled data is plentiful. Under the smoothness and low-density separation assumptions, decision boundaries should pass through low-density regions and local perturbations of an example should not alter its semantic class. FixMatch uses a model's confident self-predictions on cleanly augmented views to supervise strongly perturbed views of the same underlying data point.

\subsection{Formal Mathematical Definition}
Given unlabeled sample $u_b$, the weak prediction is $q_b = \text{softmax}(f_\theta(\alpha(u_b)))$.
The artificial target distribution is the one-hot Dirac measure at the argmax:
\begin{equation}
\hat{p}_b = \text{one\_hot}(\arg\max_{k} q_{b, k})
\end{equation}
The consistency loss is evaluated only on confident samples ($\max_k q_{b, k} \ge \tau$):
\begin{equation}
\mathcal{L}_u = \frac{1}{B} \sum_{b=1}^B \mathbb{I}\left( \max_k q_{b, k} \ge \tau \right) \cdot \text{H}\left( \hat{p}_b, \, \text{softmax}(z_b) \right)
\end{equation}
where $\text{H}(p, q) = -\sum_k p_k \ln q_k = -\ln (\text{softmax}(z_b))_{\hat{q}_b}$.

\section{Algorithm Specification}

\begin{algorithm}[H]
\caption{FixMatch Confidence-Gated Consistency Loss}
\label{alg:fixmatch}
\begin{algorithmic}[1]
\Require Weak predictions $Q \in [0, 1]^{B \times K}$, Strong logits $Z \in \mathbb{R}^{B \times K}$, confidence threshold $\tau \in (0, 1)$.
\Ensure Pseudo-labels $\hat{y} \in \{-1, \dots, K-1\}^B$, masks $M \in \{0, 1\}^B$, sample losses $\ell \in \mathbb{R}^B$, mean loss $\mathcal{L}_u$.
\State Allocate $\hat{y} \gets -\mathbf{1}^B$, $M \gets \mathbf{0}^B$, $\ell \gets \mathbf{0}^B$
\For{$b \gets 1 \textbf{ to } B$}
    \State $p_{\text{max}} \gets \max_{k} Q_{b, k}$, $k^* \gets \arg\max_{k} Q_{b, k}$
    \If{$p_{\text{max}} \ge \tau$}
        \State $M[b] \gets 1.0$, $\hat{y}[b] \gets k^*$
        \State $z_{\text{max}} \gets \max_k Z_{b, k}$
        \State $\text{lse} \gets z_{\text{max}} + \ln \left( \sum_k e^{Z_{b, k} - z_{\text{max}}} \right)$
        \State $\ell[b] \gets -(Z_{b, k^*} - \text{lse})$
    \Else
        \State $M[b] \gets 0.0$, $\hat{y}[b] \gets -1$, $\ell[b] \gets 0.0$
    \EndIf
\EndFor
\State $\mathcal{L}_u \gets \frac{1}{B} \sum_{b=1}^B \ell[b]$
\State \Return $\{\hat{y}, M, \ell, \mathcal{L}_u, \text{mean}(M)\}$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees and Error Analysis}

\begin{theorem}[Confirmation Bias Bound under Hard Thresholding]
Let $\epsilon_{\text{weak}}$ denote the error rate of weak predictions when $p_{\text{max}} \ge \tau$. The probability of incorporating a corrupted pseudo-label into the training gradient decays exponentially with the threshold gap:
\begin{equation}
\mathbb{P}(\hat{q}_b \ne y^* \mid p_{\text{max}} \ge \tau) \le \frac{1 - \tau}{\tau} \cdot \frac{\mathbb{P}(y \ne y^*)}{\mathbb{P}(y = y^*)}
\end{equation}
\end{theorem}

\begin{proof}
By Bayes' rule and odds formulation:
\begin{equation}
\frac{\mathbb{P}(y \ne k^* \mid x)}{\mathbb{P}(y = k^* \mid x)} = \frac{1 - q_{b, k^*}}{q_{b, k^*}} \le \frac{1 - \tau}{\tau}
\end{equation}
For $\tau = 0.95$, $\frac{1 - 0.95}{0.95} = \frac{0.05}{0.95} \approx 0.0526$, ensuring that fewer than $5.3\%$ of masked labels can be erroneous in expectation, suppressing runaway confirmation bias.
\end{proof}

\subsection{Limitations}
\begin{itemize}
    \item \textbf{Class Imbalance Amplification}: In imbalanced domains, dominant classes reach threshold $\tau$ much faster than minority classes, starving minority classes of gradient signal.
\end{itemize}

\section{Complexity Analysis}

\subsection{Time Complexity}
\begin{itemize}
    \item \textbf{Argmax and LogSumExp}: Takes $\mathcal{O}(K)$ FLOPs per sample.
    \item \textbf{Total Time Complexity}: $\Theta(B \cdot K)$ FLOPs.
\end{itemize}

\subsection{Space Complexity}
\begin{itemize}
    \item \textbf{Storage}: $\mathcal{O}(B)$ floats to store binary masks and sample loss vectors.
\end{itemize}

\section{Exactness and Error Analysis}

The LogSumExp calculation uses stabilized subtraction $z_k - z_{\text{max}}$ to prevent floating-point overflow during exponentiation.

\section{Worked Numerical Example}

Let $B = 2, K = 3, \tau = 0.90$.
\begin{equation}
Q = \begin{bmatrix} 0.92 & 0.05 & 0.03 \\ 0.85 & 0.10 & 0.05 \end{bmatrix}, \quad Z = \begin{bmatrix} 3.0 & 1.0 & 0.0 \\ 2.0 & 4.0 & 1.0 \end{bmatrix}
\end{equation}

\begin{enumerate}
    \item \textbf{Sample 1}:
    \begin{itemize}
        \item Max prob: $\max(0.92, 0.05, 0.03) = 0.92 \ge 0.90 \implies M_1 = 1.0, \hat{y}_1 = 0$.
        \item Logits $Z_1 = [3.0, 1.0, 0.0] \implies e^{3-3} + e^{1-3} + e^{0-3} = 1 + 0.135335 + 0.049787 = 1.185122$.
        \item $\text{lse} = 3.0 + \ln(1.185122) = 3.0 + 0.169845 = 3.169845$.
        \item Loss $\ell_1 = -(3.0 - 3.169845) = 0.169845$.
    \end{itemize}
    \item \textbf{Sample 2}:
    \begin{itemize}
        \item Max prob: $\max(0.85, 0.10, 0.05) = 0.85 < 0.90 \implies M_2 = 0.0, \hat{y}_2 = -1, \ell_2 = 0.0$.
    \end{itemize}
    \item Unsupervised Loss: $\mathcal{L}_u = \frac{0.169845 + 0.0}{2} = 0.084923$.
    \item Mask Rate: $\frac{1 + 0}{2} = 0.50$.
\end{enumerate}

\section{Numerical Considerations and Edge Cases}

\begin{itemize}
    \item \textbf{Warmup Phase}: In initial epochs, mask rate may be $0$ until model predictions sharpen past threshold $\tau$.
\end{itemize}
