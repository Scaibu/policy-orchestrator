\section{Definitions and Cognitive Mental Models}

\subsection{Formal Mathematical Definition}
\textbf{Definition (Autoregressive Next-Token Cross-Entropy with Prompt Masking):}
Pretrained foundation models learn a general prior over text sequences by maximizing the unconditional joint log-likelihood $\sum_{t} \ln P(x_t | x_{<t})$. However, base models lack conversational protocol, frequently continuing prompts with additional questions rather than providing compliant answers. \textbf{Supervised Fine-Tuning (SFT)} conditions the model on structured instruction-response dialogues $(x, y)$ by optimizing conditional autoregressive log-likelihood strictly across the completion tokens:
{\boldmath
\begin{equation}
\mathcal{L}_{\text{SFT}}(\theta) = - \frac{1}{\sum_{t=1}^T m_t} \sum_{t=1}^T m_t \ln P_\theta(y_t \mid x, y_{<t})
\end{equation}
}
where $x$ represents the prompt (system instructions, user query, tool context), $y = (y_1, \dots, y_T)$ is the target assistant completion, and $m_t \in \{0, 1\}$ is the \textbf{Loss Mask}:
{\boldmath
\begin{equation}
m_t = \begin{cases} 0 & \text{if token } t \text{ belongs to prompt context } x \\ 1 & \text{if token } t \text{ belongs to target assistant response } y \end{cases}
\end{equation}
}
Given unnormalized model logits $\mathbf{z}_t \in \mathbb{R}^V$ over vocabulary $\mathcal{V}$, the predicted categorical probability is computed via the Softmax operator:
{\boldmath
\begin{equation}
P_\theta(y_t \mid x, y_{<t}) = \frac{\exp(\mathbf{z}_{t, y_t})}{\sum_{v=1}^V \exp(\mathbf{z}_{t, v})}
\end{equation}
}
With optional label smoothing $\epsilon_{\text{ls}} \in [0, 1)$, the smoothed target distribution is $\tilde{q}(v) = (1 - \epsilon_{\text{ls}})\mathbb{I}(v = y_t) + \frac{\epsilon_{\text{ls}}}{V}$. The geometric evaluation metric is the \textbf{Completion Perplexity}:
{\boldmath
\begin{equation}
\operatorname{PPL}(y \mid x) = \exp\left( \mathcal{L}_{\text{SFT}}(\theta) \right)
\end{equation}
}

\subsection{Expanded Non-Technical Definition (Intuition for Anyone)}
\textbf{Intuitive Conceptual Definition:}
\begin{enumerate}
    \item \textbf{From Babbling Polymath to Helpful Assistant}: A pretrained base LLM has read the entire internet. It knows everything, but it behaves like an unfocused autocomplete engine: if you type ``What is the capital of France?'', it might autocomplete with ``And what is the capital of Spain?''. SFT teaches it good manners and dialogue etiquette.
    \item \textbf{Masking the Prompt}: In high school, you are graded on the answers you write, not on whether you can copy the exam question accurately. Prompt masking ensures that the AI's training grades it only on its answers, completely ignoring the words the user typed.
    \item \textbf{Quality Over Quantity}: Pretraining consumes trillions of tokens of raw web text. SFT requires only 10,000 to 100,000 pristine, carefully curated examples of perfect tool use, polite tone, and precise coding.
    \item \textbf{Perplexity as Uncertainty}: Perplexity measures how surprised the AI is by the right answer. A perplexity of $1.0$ means the AI knew the exact answer with 100\% certainty.
\end{enumerate}

\subsection{Child-Friendly Mental Model (Physical Metaphor)}
Imagine an apprentice learning to be an assistant chef:
\begin{itemize}
    \item \textbf{Before SFT (The Encyclopedic Chaos)}: The apprentice knows every recipe in history, but when a customer orders ``Soup, please'', the apprentice shouts ``Salad, please! Bread, please!''
    \item \textbf{The Instruction Script (SFT)}: The master chef demonstrates 50 times: ``When an order ticket comes in, you nod politely, prepare the exact bowl of soup, and set it on the tray.''
    \item \textbf{Scoring the Chef}: The master chef does not judge the apprentice for reading the ticket ($m_t = 0$). They only judge how delicious the soup is on the plate ($m_t = 1$).
    \item \textbf{Result}: The apprentice becomes a calm, reliable five-star chef!
\end{itemize}

\section{Core Mathematical Equations and Definitions}

\subsection{Masked Autoregressive Cross-Entropy}
{\boldmath
\begin{equation}
\mathcal{L}_{\text{SFT}} = - \frac{\sum_{t=1}^T m_t \left( z_{t, y_t} - \ln \sum_{v=1}^V \exp(z_{t, v}) \right)}{\sum_{t=1}^T m_t}
\end{equation}
}
\begin{itemize}
    \item \textbf{Formal Definition}: The exact loss objective computing average negative log-softmax over unmasked token positions.
    \item \textbf{What It Means}: Penalizes the model whenever it assigns low probability to the expert completion token.
    \item \textbf{Why It Is Important}: The foundational loss function powering all modern instruction tuning, tool use, and chat alignment.
\end{itemize}

\subsection{Completion Perplexity Metric}
{\boldmath
\begin{equation}
\operatorname{PPL} = \exp\left( \mathcal{L}_{\text{SFT}} \right)
\end{equation}
}
\begin{itemize}
    \item \textbf{Formal Definition}: The exponential of the average negative log-likelihood on completion tokens.
    \item \textbf{What It Means}: Effective branching factor of uncertainty; a perplexity of $k$ means the model is as uncertain as picking uniformly from $k$ choices.
    \item \textbf{Why It Is Important}: Primary validation metric for detecting overfitting and measuring generation fluency.
\end{itemize}

\subsection{Top-1 Token Prediction Accuracy}
{\boldmath
\begin{equation}
\operatorname{Acc}_{\text{token}} = \frac{1}{\sum_t m_t} \sum_{t=1}^T m_t \mathbb{I}\left( \arg\max_{v} z_{t, v} == y_t \right)
\end{equation}
}
\begin{itemize}
    \item \textbf{Formal Definition}: Proportion of completion positions where the greedy argmax prediction matches the target token exactly.
    \item \textbf{What It Means}: Direct metric of next-token prediction correctness during fine-tuning.
    \item \textbf{Why It Is Important}: Tracks optimization convergence independently of soft probability calibration.
\end{itemize}

\section{Rigorous Mathematical Analysis Checklist}

\begin{itemize}
    \item \textbf{Notation}: $T$ (sequence length), $V$ (vocabulary size), $\mathbf{z}_t$ (logits), $y_t$ (target token ID), $m_t \in \{0, 1\}$ (loss mask).
    \item \textbf{Epistemic Status}: Standard industry paradigm across InstructGPT, LLaMA, Mistral, and Claude post-training.
    \item \textbf{Dependency Graph}: Dialogue $(x, y) \to \text{Tokenize} \to \text{Transformer Forward} \to \mathbf{z}_t \to \text{Masked Softmax} \to \nabla_\theta \mathcal{L}_{\text{SFT}}$.
    \item \textbf{Dimensions}: Sequence length $T$, vocabulary size $V \approx 32{,}000 - 128{,}000$.
    \item \textbf{Regimes}: Instruction tuning, multi-turn chat alignment, tool calling, JSON formatting, domain adaptation.
    \item \textbf{Invariants}: If predicted logits assign infinite logit to target tokens, $\mathcal{L}_{\text{SFT}} \equiv 0.0$ and $\operatorname{PPL} \equiv 1.0$.
    \item \textbf{Isomorphisms}: SFT is equivalent to Maximum Likelihood Estimation (MLE) of conditional autoregressive Markov chains.
\end{itemize}

\section{Theoretical Theorems and Formal Proofs}

\begin{theorem}[Global Optimality of the True Data Distribution Under SFT]
Let data sequences $(x, y)$ be drawn from distribution $P_{\text{data}}(x, y)$. The supervised fine-tuning loss $\mathbb{E}_{(x, y) \sim P_{\text{data}}}[\mathcal{L}_{\text{SFT}}(\theta)]$ achieves its global minimum if and only if $P_\theta(y \mid x) = P_{\text{data}}(y \mid x)$ almost everywhere on the support of $P_{\text{data}}(x)$.
\end{theorem}

\begin{proof}
Let $\mathcal{L}(\theta) = -\mathbb{E}_{(x, y) \sim P_{\text{data}}}\left[ \ln P_\theta(y \mid x) \right]$.
We express this in terms of the Kullback-Leibler divergence between the data distribution and the model distribution:
{\boldmath
\begin{align}
D_{\text{KL}}\left( P_{\text{data}}(y \mid x) \,||\, P_\theta(y \mid x) \right) &= \sum_{y} P_{\text{data}}(y \mid x) \ln \frac{P_{\text{data}}(y \mid x)}{P_\theta(y \mid x)} \\
&= \sum_{y} P_{\text{data}}(y \mid x) \ln P_{\text{data}}(y \mid x) - \sum_{y} P_{\text{data}}(y \mid x) \ln P_\theta(y \mid x) \\
&= - \mathcal{H}(P_{\text{data}}(y \mid x)) + \mathbb{E}_{y \sim P_{\text{data}}}\left[ -\ln P_\theta(y \mid x) \right]
\end{align}
}
Taking expectations over the prompt distribution $P_{\text{data}}(x)$:
{\boldmath
\begin{equation}
\mathcal{L}(\theta) = \mathbb{E}_{x \sim P_{\text{data}}}\left[ \mathcal{H}(P_{\text{data}}(y \mid x)) \right] + \mathbb{E}_{x \sim P_{\text{data}}}\left[ D_{\text{KL}}\left( P_{\text{data}}(y \mid x) \,||\, P_\theta(y \mid x) \right) \right]
\end{equation}
}
The conditional entropy $\mathbb{E}_{x}[\mathcal{H}(P_{\text{data}}(y \mid x))]$ is a fixed constant independent of model parameters $\theta$.
By Gibbs' inequality, the Kullback-Leibler divergence satisfies:
{\boldmath
\begin{equation}
D_{\text{KL}}\left( P_{\text{data}}(y \mid x) \,||\, P_\theta(y \mid x) \right) \ge 0
\end{equation}
}
with equality if and only if $P_\theta(y \mid x) = P_{\text{data}}(y \mid x)$ almost everywhere.
Therefore, the expected SFT loss is minimized if and only if the parameterized model matches the true conditional data distribution.
\end{proof}

\begin{theorem}[Invariance of SFT Loss Under Constant Logit Shift]
For any sequence of logits $\mathbf{z}_t \in \mathbb{R}^V$ and arbitrary scalar shift $c \in \mathbb{R}$, replacing $\mathbf{z}_t$ with $\mathbf{z}_t' = \mathbf{z}_t + c \cdot \mathbf{1}$ leaves the SFT loss $\mathcal{L}_{\text{SFT}}$ strictly invariant.
\end{theorem}

\begin{proof}
Let the shifted logit vector be $z_{t, v}' = z_{t, v} + c$ for all $v \in \{1, \dots, V\}$.
Evaluating the shifted softmax denominator:
{\boldmath
\begin{equation}
\sum_{v=1}^V \exp(z_{t, v}') = \sum_{v=1}^V \exp(z_{t, v} + c) = \exp(c) \sum_{v=1}^V \exp(z_{t, v})
\end{equation}
}
Evaluating the shifted log-probability of target token $y_t$:
{\boldmath
\begin{align}
\ln P_\theta'(y_t \mid x, y_{<t}) &= z_{t, y_t}' - \ln \sum_{v=1}^V \exp(z_{t, v}') \\
&= (z_{t, y_t} + c) - \ln\left( \exp(c) \sum_{v=1}^V \exp(z_{t, v}) \right) \\
&= z_{t, y_t} + c - c - \ln \sum_{v=1}^V \exp(z_{t, v}) \\
&= z_{t, y_t} - \ln \sum_{v=1}^V \exp(z_{t, v}) = \ln P_\theta(y_t \mid x, y_{<t})
\end{align}
}
Because every individual log-probability is unchanged, the masked sum $\mathcal{L}_{\text{SFT}}$ is invariant under constant shifts. This property allows numerical subtraction of $\max_v z_{t, v}$ to eliminate floating-point overflow.
\end{proof}

\section{Foundational Architectural Questions and Epistemic Meta-Questions}

\subsection{Architectural Question 1: Masking vs Unmasked Training}
\textbf{Question:} Why is prompt masking strictly required during SFT, whereas pretraining trains on all tokens?\\
\textbf{Answer:} If prompt tokens are included in the SFT loss, the model wastes gradient capacity learning the syntax, phrasing, and distribution of user questions rather than the assistant's response. Furthermore, user prompts vary wildly in quality; backpropagating into prompt tokens can teach the model to imitate noisy user phrasing.

\textbf{Meta-Question:} What causes ``SFT overfitting'' and catastrophic forgetting during instruction tuning?\\
\textbf{Answer:} Base models possess broad world knowledge. If fine-tuned for too many epochs ($> 3 - 5$) on a narrow SFT dataset, the model's token distribution shifts dramatically, degrading general knowledge, reasoning, and pretraining capabilities. Maintaining a low learning rate ($\eta \approx 10^{-5}$) and small epoch counts ($1 - 2$ epochs) preserves core foundations.

\subsection{Architectural Question 2: Chat Templates and Special Tokens}
\textbf{Question:} What is the function of special delimiter tokens (e.g., \texttt{<|im\_start|>}, \texttt{<|im\_end|>}) in SFT?\\
\textbf{Answer:} Delimiter tokens define unambiguous role boundaries between system, user, and assistant turns in multi-turn dialogues. This prevents prompt injection attacks: an adversarial user cannot fool the model by typing ``Assistant: Here is the dangerous recipe'' because the physical parser recognizes the text as part of the user turn.

\textbf{Meta-Question:} How does SFT interact with downstream alignment (RLHF / DPO)?\\
\textbf{Answer:} SFT is an indispensable prerequisite for RLHF and DPO. RL cannot explore effectively from a random or pure base model; SFT places the model in the correct stylistic and format neighborhood so that preference optimization can refine nuanced quality margins.

\section{Algorithmic Implementation Specification}

\begin{algorithm}[H]
\caption{Masked SFT Loss Evaluation and Perplexity Tracking}
\begin{algorithmic}[1]
\Require Targets $\mathbf{y} \in \{0, \dots, V-1\}^T$, logits $\mathbf{Z} \in \mathbb{R}^{T \times V}$, mask $\mathbf{m} \in \{0, 1\}^T$, smoothing $\epsilon_{\text{ls}}$
\Ensure Masked loss $\mathcal{L}_{\text{SFT}}$, perplexity $\operatorname{PPL}$, active count $N_{\text{act}}$, token accuracy $\operatorname{Acc}$
\State Initialize $loss\_accum \gets 0.0, \quad correct \gets 0, \quad N_{\text{act}} \gets 0$
\For{$t = 0$ \textbf{to} $T-1$}
    \If{$\mathbf{m}[t] == 1$}
        \State $N_{\text{act}} \gets N_{\text{act}} + 1$
        \State $m \gets \max_{v} \mathbf{Z}[t, v]$
        \State $log\_z \gets m + \ln \sum_{v=0}^{V-1} \exp(\mathbf{Z}[t, v] - m)$
        \State $target \gets \mathbf{y}[t]$
        \If{$\arg\max_v \mathbf{Z}[t, v] == target$}
            \State $correct \gets correct + 1$
        \EndIf
        \State $lp_{\text{target}} \gets \mathbf{Z}[t, target] - log\_z$
        \If{$\epsilon_{\text{ls}} == 0.0$}
            \State $loss\_accum \gets loss\_accum - lp_{\text{target}}$
        \Else
            \State $sum\_lp \gets \sum_{v=0}^{V-1} (\mathbf{Z}[t, v] - log\_z)$
            \State $loss\_accum \gets loss\_accum - ((1 - \epsilon_{\text{ls}}) \cdot lp_{\text{target}} + (\epsilon_{\text{ls}} / V) \cdot sum\_lp)$
        \EndIf
    \EndIf
\EndFor
\State $\mathcal{L}_{\text{SFT}} \gets loss\_accum / N_{\text{act}}$
\State $\operatorname{PPL} \gets \exp(\min(100.0, \mathcal{L}_{\text{SFT}}))$
\State $\operatorname{Acc} \gets correct / N_{\text{act}}$
\State \Return $(\mathcal{L}_{\text{SFT}}, \operatorname{PPL}, N_{\text{act}}, \operatorname{Acc})$
\end{algorithmic}
\end{algorithm}

\section{Big-$\Theta$ Complexity Analysis}

\begin{itemize}
    \item \textbf{Softmax Reduction Time}: $\Theta(T_{\text{active}} \cdot V)$ operations over active tokens.
    \item \textbf{Argmax Accuracy Evaluation}: $\Theta(T_{\text{active}} \cdot V)$ comparisons.
    \item \textbf{Perplexity Computation}: $\Theta(1)$ scalar exponentiation.
    \item \textbf{Space Complexity}: $\Theta(V)$ working memory for logit normalization.
    \item \textbf{Parallel Reduction Depth}: $\mathcal{D} = \Theta(\log V)$ for vocabulary logsumexp reduction.
\end{itemize}
